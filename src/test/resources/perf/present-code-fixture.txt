package com.serenity.ui.renderer

import java.util.concurrent.ConcurrentHashMap

/** Names the caches one render backend keeps for an owner. A key is compared by identity, so a backend declares exactly
  * one, and `kind` is how the caches come back with their own type.
  */
final class BackendCacheKey[A <: AnyRef](val kind: Class[A])

/** The caches render backends keep for one owner, each created the first time its backend asks for them. The owner
  * itself names no backend, so a launch that never draws with one never loads its classes.
  */
final class BackendCaches:

  private val slots = new ConcurrentHashMap[BackendCacheKey[?], AnyRef]()

  /** The caches of `key`, made by `make` if this is the first ask. Concurrent first asks make them once. */
  def getOrCreate[A <: AnyRef](key: BackendCacheKey[A])(make: => A): A =
    key.kind.cast(slots.computeIfAbsent(key, _ => make))

  def holds(key: BackendCacheKey[?]): Boolean = slots.containsKey(key)

/** A canvas-shaped [[Java2DRenderSurface]] that records caret fills instead of painting them, so the Swing window can
  * fill carets straight over its presented frame rather than compositing a full-window overlay image. Carets are only
  * ever drawn with `fillPixelRect`; the 1x1 backing image just satisfies the parent's constructor.
  */
final class CaretRecordingSurface private (
    metrics: CellMetrics,
    font: Font,
    logicalWidthPx: Int,
    logicalHeightPx: Int,
    deviceScaleX: Double,
    deviceScaleY: Double
) extends Java2DRenderSurface(
      new BufferedImage(1, 1, BufferedImage.TYPE_INT_ARGB),
      metrics,
      font,
      _ => (),
      logicalWidthPx = logicalWidthPx,
      logicalHeightPx = logicalHeightPx,
      deviceScaleX = deviceScaleX,
      deviceScaleY = deviceScaleY
    ):
  private val fills = new AtomicReference(Vector.empty[CaretRecordingSurface.CaretFill])

  override def fillPixelRect(xPx: Int, yPx: Int, widthPx: Int, heightPx: Int, color: RenderColor): Unit =
    val _ = fills.updateAndGet(
      _ :+ CaretRecordingSurface.CaretFill(PixelRect(xPx, yPx, widthPx.max(1), heightPx.max(1)), color.toAwt)
    )

  def recordedFills: List[CaretRecordingSurface.CaretFill] = fills.get().toList

object CaretRecordingSurface:

  final case class CaretFill(rect: PixelRect, color: Color)

  def forCanvas(metrics: CellMetrics, font: Font, canvas: javax.swing.JPanel): CaretRecordingSurface =
    val scale = Java2DRenderSurface.deviceScaleFor(canvas)
    new CaretRecordingSurface(
      metrics,
      font,
      Java2DRenderSurface.logicalCanvasDimension(canvas.getWidth, canvas.getPreferredSize.width),
      Java2DRenderSurface.logicalCanvasDimension(canvas.getHeight, canvas.getPreferredSize.height),
      scale.x,
      scale.y
    )

import java.awt.Font

import com.serenity.text.fallback.FontSlot
import com.serenity.ui.fonts.FontFallback

/** What the GUI cell grid needs to draw text its primary font has no glyph for: the primary, the chain behind it and
  * the cache that itemises (and remembers) a string's runs, so a frame does not itemise every line again. A terminal
  * has no fonts, so it passes none.
  */
final private[renderer] case class CellFallback(primary: Font, chain: FontFallback, cache: WrappedLineCache):

  /** `text` as `(from, until, family)` stretches, `family` being the fallback family that draws the stretch, empty when
    * the primary draws it. Empty when the primary draws all of `text`. Neutrals the primary covers stay on it, as every
    * one of them belongs in its own cell on a grid.
    */
  def stretchesOf(text: String): Vector[(Int, Int, Option[String])] =
    if text.isEmpty || cache.isPlainAscii(text) then Vector.empty
    else
      val runs = cache.fallbackRuns(primary, chain, text, stickyNeutrals = false)
      if runs.forall(_.slot == FontSlot.Primary) then Vector.empty
      else
        runs.map(run =>
          (run.from, run.until, Option.when(run.slot != FontSlot.Primary)(chain.slotFamilies(run.slot.index - 1)))
        )
package com.serenity.ui.renderer

import com.serenity.ui.layout.CharWidth

/** A string laid out on a character-cell grid: how many cells it takes and which stretches of it share a baseline x. A
  * narrow run is drawn as one string so the font's own shaping applies; a two-cell cluster (a wide character, an emoji
  * sequence) is drawn alone, with whatever extends it, on the cell its caret stop is, because its glyph's own advance
  * need not be two cells. How many cells a cluster takes is [[CharWidth.ofCluster]]'s answer, here as in the caret
  * stops and the terminal; a cluster is [[CharWidth.clusterEnd]]'s, which is also where the carets stop.
  */
private[ui] object CellRuns:

  /** The first codepoint with a wide East Asian Width: text below it is all narrow, which lets a string with no wide
    * character skip the cut.
    */
  private val FirstWide = 0x1100

  /** The first combining mark: text below it takes one cell a character. */
  private val FirstMark = 0x0300

  def cellsIn(text: String): Int =
    if text.forall(_ < FirstMark) then text.length
    else
      @annotation.tailrec
      def sum(index: Int, cells: Int): Int =
        if index >= text.length then cells
        else
          val end = CharWidth.clusterEnd(text, index)
          sum(end, cells + CharWidth.ofCluster(text, index, end))
      sum(0, 0)

  /** `text` cut into `(cell offset, piece)` where every two-cell cluster is a piece of its own and the narrow stretches
    * between are kept whole.
    */
  def pieces(text: String): Vector[(Int, String)] =
    if text.forall(_ < FirstWide) then Vector((0, text))
    else
      @annotation.tailrec
      def walk(
        index: Int,
        cell: Int,
        openFrom: Int,
        openCell: Int,
        found: Vector[(Int, String)]
      ): Vector[(Int, String)] =
        def closed: Vector[(Int, String)] =
          if openFrom >= 0 && openFrom < index then found :+ ((openCell, text.substring(openFrom, index))) else found
        if index >= text.length then closed
        else
          val end   = CharWidth.clusterEnd(text, index)
          val cells = CharWidth.ofCluster(text, index, end)
          if cells == 2 then walk(end, cell + 2, -1, 0, closed :+ ((cell, text.substring(index, end))))
          else if openFrom >= 0 && openFrom < index then walk(end, cell + cells, openFrom, openCell, found)
          else walk(end, cell + cells, index, cell, found)
      walk(0, 0, -1, 0, Vector.empty)

  /** `text` as one string per grapheme cluster, which is what a fallback font draws on its own cells: a mark or an
    * emoji extension drawn apart from its base would land on the cell after it. Marks with no base start a unit of
    * their own.
    */
  def clusters(text: String): Vector[String] =
    @annotation.tailrec
    def walk(index: Int, found: Vector[String]): Vector[String] =
      if index >= text.length then found
      else
        val end = CharWidth.clusterEnd(text, index)
        walk(end, found :+ text.substring(index, end))
    walk(0, Vector.empty)
package com.serenity.ui.renderer

import java.util.LinkedHashMap

import com.serenity.lsp.config.LanguageId
import com.serenity.lsp.model.SemanticToken
import com.serenity.state.models.{IndentationSettings, TextVisualLine}

/** Paints text onto a [[RenderSurface]] in the two ways a surface can accept it: the cell-grid path (`putString`,
  * counting screen columns) and the measured pixel path ([[renderMeasuredLine]], laying glyph runs out along a visual
  * line's own caret stops). Both share this object's run-splitting and codepoint width rules, so a line drawn either
  * way agrees with the other about where each character sits.
  */
object CharacterRenderer:

  /** A run of text to paint; `startX` is a *cell* column on the screen grid (a wide glyph occupies two of them). */
  final private case class TextRun(startX: Int, content: String)
  final private case class CollectedRuns(runs: List[TextRun], endX: Int)

  /** A grapheme cluster's boundaries within a line's text, as local character indices. */
  private[renderer] case class GraphemeSpan(startLocalIndex: Int, endLocalIndex: Int)

  private[renderer] def computeGraphemeSpans(text: String): Vector[GraphemeSpan] =
    @annotation.tailrec
    def collect(localIndex: Int, acc: List[GraphemeSpan]): Vector[GraphemeSpan] =
      if localIndex >= text.length then acc.reverse.toVector
      else
        val nextIndex = TextEditing.nextGraphemeBoundary(text, localIndex).min(text.length)
        collect(nextIndex, GraphemeSpan(localIndex, nextIndex) :: acc)

    collect(0, Nil)

  def renderString(
    surface: RenderSurface,
    x: Int,
    y: Int,
    content: String
  ): Unit =
    renderStringPlain(surface, x, y, content)

  def renderStringPlain(
    surface: RenderSurface,
    x: Int,
    y: Int,
    content: String,
    tabWidth: Int = IndentationSettings.DefaultTabSize
  ): Unit =
    val collectedRuns = collectPlainRuns(x, x, content, tabWidth)
    flushPlainRuns(surface, y, collectedRuns.runs)

  /** The characters of `visualLine` in buffer columns `[rangeStart, rangeEnd)` painted on the cell grid in the colours
    * and style the surface has, each on the cell the row's caret stops give it (so after a wide character, a tab or a
    * character a fallback font draws, it is still where the row measured it) and in the fallback font the row draws it
    * in. `originCell` is the cell of the row's first caret stop; cells outside `[leftCell, rightCell)` are not painted.
    */
  def renderCellRange(
    surface: RenderSurface,
    visualLine: TextVisualLine,
    rangeStart: Int,
    rangeEnd: Int,
    originCell: Int,
    y: Int,
    leftCell: Int,
    rightCell: Int,
    cellPx: Int,
    fallback: Option[CellFallback]
  ): Unit =
    val text      = visualLine.text
    val stretches = fallback.fold(Vector.empty)(_.stretchesOf(text))
    def cellOf(column: Int): Int =
      originCell + math.round(visualLine.xForColumn(column).getOrElse(visualLine.widthPx) / cellPx.toFloat)
    def familyAt(index: Int): Option[String] =
      stretches.collectFirst { case (from, until, family) if index >= from && index < until => family }.flatten
    @annotation.tailrec
    def paint(column: Int): Unit =
      val index = column - visualLine.startColumn
      if column < rangeEnd && index < text.length then
        if index < 0 then paint(column + 1)
        else
          val codePoint = text.codePointAt(index)
          val end       = CharWidth.clusterEnd(text, index)
          val cell      = cellOf(column)
          val cells     = cellOf(column + end - index) - cell
          if cell >= leftCell && cell < rightCell then
            val shown =
              if Character.isISOControl(codePoint) then " ".repeat(math.max(1, cells)) else text.substring(index, end)
            familyAt(index).fold(surface.putString(cell, y, shown))(family =>
              surface.text.withFallbackFamily(family, CharWidth.needsLigaturesToJoin(text, index, end))(
                withinCells(surface, cell, y, math.max(1, cells))(surface.putString(cell, y, shown))
              )
            )
          paint(column + end - index)
    paint(rangeStart)

  def renderChar(
    surface: RenderSurface,
    x: Int,
    y: Int,
    char: Char
  ): Unit =
    val displayChar = char match
      case '_'                           => '_'
      case '\t'                          => '\t'
      case c if c.isControl && c != '\t' => ' '
      case c                             => c

    surface.putString(x, y, displayChar.toString)

  def isVisibleChar(char: Char): Boolean =
    isVisibleCodePoint(char.toInt)

  def renderStyledString(
    surface: RenderSurface,
    x: Int,
    y: Int,
    content: String,
    theme: Theme,
    syntaxHighlightingEnabled: Boolean = true,
    language: Option[LanguageId] = None,
    styledSegments: Option[List[StyledText]] = None,
    semanticTokens: Option[List[SemanticToken]] = None,
    maxColumn: Option[Int] = None,
    highlightCache: com.serenity.ui.theme.ThemeHighlightCache = com.serenity.ui.theme.ThemeHighlightCache(),
    tabWidth: Int = IndentationSettings.DefaultTabSize,
    fallback: Option[CellFallback] = None
  ): Unit =
    styledSegments match
      case Some(styledTexts) =>
        renderStyledLine(surface, x, y, styledTexts, theme, maxColumn, tabWidth, fallback)
      case None if syntaxHighlightingEnabled =>
        val styledTexts = highlightCache.highlightLine(content, theme, language, semanticTokens)
        renderStyledLine(surface, x, y, styledTexts, theme, maxColumn, tabWidth, fallback)
      case None =>
        renderThemedPlain(surface, x, y, content, theme, tabWidth, maxColumn, fallback)

  def renderThemedPlain(
    surface: RenderSurface,
    x: Int,
    y: Int,
    content: String,
    theme: Theme,
    tabWidth: Int = IndentationSettings.DefaultTabSize,
    maxColumn: Option[Int] = None,
    fallback: Option[CellFallback] = None
  ): Unit =
    val collectedRuns = collectPlainRuns(x, x, content, tabWidth)
    renderThemedRuns(surface, y, collectedRuns.runs, theme, maxColumn, fallback)

  /** Render a visual line using pixel-precision caret stops.
    *
    * Walks the line by grapheme cluster (so a combining mark or an emoji sequence is never split across two runs),
    * groups consecutive clusters sharing the same effective fg/bg colour and style into runs, and calls
    * [[RenderSurface.drawRunPx]] for each. Callers must set the surface font before this call.
    *
    * @param xOriginPx
    *   pixel X of the pane's left edge
    * @param yPx
    *   pixel Y of the top of this visual line
    */
  def renderMeasuredLine(
    surface: RenderSurface,
    xOriginPx: Float,
    yPx: Int,
    lineHeightPx: Int,
    ascentPx: Int,
    visualLine: TextVisualLine,
    theme: Theme,
    syntaxHighlightingEnabled: Boolean = false,
    language: Option[LanguageId] = None,
    styledSegments: Option[List[StyledText]] = None,
    clipRightXPx: Option[Float] = None,
    semanticTokens: Option[List[SemanticToken]] = None,
    highlightCache: com.serenity.ui.theme.ThemeHighlightCache = com.serenity.ui.theme.ThemeHighlightCache(),
    graphemeCache: GraphemeSegmentationCache = GraphemeSegmentationCache()
  ): Unit =
    val text = visualLine.text
    if text.nonEmpty then
      val segments =
        styledSegments.getOrElse {
          if syntaxHighlightingEnabled then highlightCache.highlightLine(text, theme, language, semanticTokens)
          else List(StyledText(text, TextStyle.normal, theme.foreground, theme.background))
        }
      val runs = MeasuredLineRuns.of(visualLine, segments, graphemeCache.spansFor(text), theme)

      val shaped = surface.text.shapesStretchesOf(text)

      def paintRun(
        run: MeasuredRun,
        stretch: String,
        from: Int,
        until: Int,
        family: Option[String] = None
      ): Unit =
        val startXPx      = xOriginPx + run.minXPx
        val endXPx        = xOriginPx + run.maxXPx
        val clippedEndXPx = clipRightXPx.fold(endXPx)(_.min(endXPx))
        val widthPx       = clippedEndXPx - startXPx
        if widthPx > 0.0f then
          surface.setForegroundColor(run.foreground)
          surface.setBackgroundColor(run.background)
          withStyle(surface, run.style) {
            // A fallback glyph can stand taller than the row it sits in, and ink outside the row is not ink a repaint
            // of the rows beside it would put back, so it is cut to the row.
            val clipToRow = family.isDefined
            def draw(): Unit =
              if shaped then
                surface.text.drawShapedRunPx(
                  startXPx,
                  yPx,
                  widthPx,
                  lineHeightPx,
                  ascentPx,
                  stretch,
                  from,
                  until,
                  clipToRow
                )
              else surface.text.drawRunPx(startXPx, yPx, widthPx, lineHeightPx, ascentPx, run.text, clipToRow)
            family.fold(draw())(surface.text.withFallbackFamily(_)(draw()))
          }

      def paintSegment(segment: FontSegment): Unit =
        if shaped then
          MeasuredLineRuns
            .stretched(segment.runs, segment.text)
            .foreach(placed => paintRun(placed.run, placed.stretch, placed.from, placed.until, segment.family))
        else segment.runs.foreach(run => paintRun(run, run.text, 0, run.text.length, segment.family))

      if visualLine.fontRuns.isEmpty then
        if shaped then
          MeasuredLineRuns
            .stretched(runs, text)
            .foreach(placed => paintRun(placed.run, placed.stretch, placed.from, placed.until))
        else runs.foreach(run => paintRun(run, run.text, 0, run.text.length))
      else MeasuredLineRuns.byFont(visualLine, runs).foreach(paintSegment)

  private def renderStyledLine(
    surface: RenderSurface,
    x: Int,
    y: Int,
    styledTexts: List[com.serenity.ui.theme.StyledText],
    theme: Theme,
    maxColumn: Option[Int],
    tabWidth: Int,
    fallback: Option[CellFallback]
  ): Unit =
    styledTexts.foldLeft(x) { (currentX, styledText) =>
      val segmentTheme = theme.copy(
        foreground = styledText.foregroundColor,
        background = styledText.backgroundColor
      )
      val collectedRuns = collectPlainRuns(x, currentX, styledText.content, tabWidth)
      withStyle(surface, styledText.style) {
        renderThemedRuns(surface, y, collectedRuns.runs, segmentTheme, maxColumn, fallback)
      }
      collectedRuns.endX
    }: Unit

  private def withStyle(surface: RenderSurface, style: TextStyle)(render: => Unit): Unit =
    surface.enableStyle(style)
    try render
    finally surface.disableStyle(style)

  private def collectPlainRuns(originX: Int, startX: Int, content: String, tabWidth: Int): CollectedRuns =
    final case class PlainRunState(
        completed: List[TextRun],
        currentText: StringBuilder,
        currentStartX: Int,
        currentX: Int
    ):
      def flush: PlainRunState =
        if currentText.length > 0 then
          copy(
            completed = TextRun(currentStartX, currentText.toString) :: completed,
            currentText = StringBuilder()
          )
        else this

    val initial = PlainRunState(Nil, StringBuilder(), startX, startX)
    @annotation.tailrec
    def consume(index: Int, state: PlainRunState): PlainRunState =
      if index >= content.length then state.flush
      else
        val end = CharWidth.clusterEnd(content, index)
        val nextState = content.codePointAt(index) match
          case '\t' =>
            val flushed     = state.flush
            val spacesToAdd = TabStops.cellsAt(flushed.currentX - originX, tabWidth)
            val tabSpaces   = " " * spacesToAdd
            flushed.copy(
              completed = TextRun(flushed.currentX, tabSpaces) :: flushed.completed,
              currentStartX = flushed.currentX + spacesToAdd,
              currentX = flushed.currentX + spacesToAdd
            )
          case visible if isVisibleCodePoint(visible) =>
            val start = if state.currentText.length == 0 then state.currentX else state.currentStartX
            if end == index + Character.charCount(visible) then
              if Character.isBmpCodePoint(visible) then state.currentText.append(visible.toChar)
              else state.currentText.appendAll(Character.toChars(visible))
            else state.currentText.appendAll(content.substring(index, end))
            state.copy(currentStartX = start, currentX = state.currentX + CharWidth.ofCluster(content, index, end))
          case _ =>
            val flushed = state.flush
            flushed.copy(currentStartX = flushed.currentX)
        consume(end, nextState)

    val finalState = consume(0, initial)

    CollectedRuns(finalState.completed.reverse, finalState.currentX)

  private def isVisibleCodePoint(codePoint: Int): Boolean =
    !Character.isISOControl(codePoint)

  /** A run on the cell grid: the stretches the primary font draws as one string, and each character a fallback font
    * draws alone on the cell the caret stops put it on, since its glyph's advance has nothing to do with the grid.
    */
  private def flushRun(surface: RenderSurface, y: Int, run: TextRun, fallback: Option[CellFallback]): Unit =
    val stretches = fallback.fold(Vector.empty)(_.stretchesOf(run.content))
    if stretches.isEmpty then surface.putString(run.startX, y, run.content)
    else
      stretches.foreach { (from, until, family) =>
        val startCell = run.startX + CellRuns.cellsIn(run.content.substring(0, from))
        val stretch   = run.content.substring(from, until)
        family match
          case None => surface.putString(startCell, y, stretch)
          case Some(name) =>
            CellRuns.clusters(stretch).foldLeft(startCell) { (cell, shown) =>
              val cells = CellRuns.cellsIn(shown)
              surface.text.withFallbackFamily(name, CharWidth.needsLigaturesToJoin(shown, 0, shown.length)) {
                withinCells(surface, cell, y, cells)(surface.putString(cell, y, shown))
              }
              cell + cells
            }: Unit
      }

  /** A fallback glyph can stand taller than the row it sits in, and ink outside the row is not ink a repaint of the
    * rows beside it would put back, so it is cut to its own cells.
    */
  private def withinCells(surface: RenderSurface, cell: Int, y: Int, cells: Int)(draw: => Unit): Unit =
    surface.panelOutlines.fold(draw)(_.withRectClip(cell, y, cells, 1)(draw))

  private def flushPlainRuns(surface: RenderSurface, y: Int, runs: List[TextRun]): Unit =
    runs.foreach(flushRun(surface, y, _, None))

  private def renderThemedRuns(
    surface: RenderSurface,
    y: Int,
    runs: List[TextRun],
    theme: Theme,
    maxColumn: Option[Int],
    fallback: Option[CellFallback]
  ): Unit =
    surface.setForegroundColor(theme.foreground)
    surface.setBackgroundColor(theme.background)
    runs.flatMap(clipRunToColumn(_, maxColumn)).foreach(flushRun(surface, y, _, fallback))

  /** Cell-grid runs need no sub-character precision (unlike the measured pixel path's `clipRightXPx`), but a column is
    * a cell rather than a character: the content is truncated at the last glyph that fits whole, so a wide one is
    * dropped rather than half-drawn at the limit. `None` (the common case: word wrap on, or no pane-width limit given)
    * leaves every run untouched.
    */
  private def clipRunToColumn(run: TextRun, maxColumn: Option[Int]): Option[TextRun] =
    maxColumn match
      case None                               => Some(run)
      case Some(limit) if run.startX >= limit => None
      case Some(limit) =>
        val allowedCells = limit - run.startX
        if allowedCells >= run.content.length then Some(run)
        else
          @annotation.tailrec
          def fittingLength(index: Int, usedCells: Int): Int =
            if index >= run.content.length then index
            else
              val end   = CharWidth.clusterEnd(run.content, index)
              val width = CharWidth.ofCluster(run.content, index, end)
              if usedCells + width > allowedCells then index
              else fittingLength(end, usedCells + width)

          Some(run.copy(content = run.content.take(fittingLength(0, 0))))

/** Bounded, access-order-eviction memoization cache of [[CharacterRenderer.computeGraphemeSpans]]'s result, one
  * grapheme boundary walk per distinct line of text. Grapheme-cluster boundaries are a pure function of a line's text
  * content alone -- independent of theme or caret-stop layout -- so they're memoized the same way `ThemeHighlightCache`
  * caches syntax highlighting: same input, forever the same output, nothing to invalidate. This also removes the
  * redundant double walk that used to happen every render call, where `graphemeBounds` and `graphemeChars` each
  * independently re-ran `TextEditing.nextGraphemeBoundary` over the same text.
  *
  * Instance-scoped (issue #1677): one instance is created per render-owning entity (held on
  * [[com.serenity.state.manager.RenderCaches]], threaded through [[RenderContext]] to
  * [[CharacterRenderer.renderMeasuredLine]]) rather than a JVM-wide singleton, so two independently constructed
  * instances share no cache state.
  */
final class GraphemeSegmentationCache:
  import CharacterRenderer.GraphemeSpan

  private val MaxGraphemeSegmentationCacheEntries = 4096

  private val cache =
    new LinkedHashMap[String, Vector[GraphemeSpan]](16, 0.75f, true):
      override def removeEldestEntry(eldest: java.util.Map.Entry[String, Vector[GraphemeSpan]]): Boolean =
        size() > MaxGraphemeSegmentationCacheEntries

  private[renderer] def spansFor(text: String): Vector[GraphemeSpan] =
    cache.synchronized(Option(cache.get(text))) match
      case Some(cached) => cached
      case None =>
        val computed = CharacterRenderer.computeGraphemeSpans(text)
        cache.synchronized {
          val _ = cache.put(text, computed)
        }
        computed

object GraphemeSegmentationCache:
  def apply(): GraphemeSegmentationCache = new GraphemeSegmentationCache
package com.serenity.ui.renderer

import java.awt.Color

import scala.annotation.tailrec

import com.serenity.ui.layout.PixelRect

/** The opaque colour a clear painted, and the rectangles it painted it into.
  *
  * A run paints its background before its glyphs, which repeats the clear for every run whose background is the colour
  * the viewport was just cleared to. Over pixels nothing has touched since, that second fill changes nothing and can be
  * skipped; which pixels have been touched is [[PaintedPixels]]'s to say.
  *
  * Asked once per run per frame, so [[covers]] takes the run's edges as plain ints and allocates nothing.
  */
final private[renderer] case class CleanBackground(color: Color, cleared: List[PixelRect]):

  def covers(left: Int, top: Int, right: Int, bottom: Int, background: Color): Boolean =
    background == color && clearedAround(cleared, left, top, right, bottom)

  @tailrec
  private def clearedAround(rects: List[PixelRect], left: Int, top: Int, right: Int, bottom: Int): Boolean =
    rects match
      case Nil => false
      case rect :: rest =>
        (rect.xPx <= left && rect.yPx <= top && right <= rect.rightPx && bottom <= rect.bottomPx) ||
        clearedAround(rest, left, top, right, bottom)

private[renderer] object CleanBackground:

  /** Only an opaque colour replaces what is under it; a translucent one blends, so painting it twice differs. */
  def cleared(color: Color, rects: List[PixelRect]): Option[CleanBackground] =
    Option.when(color.getAlpha == 255)(CleanBackground(color, rects))

/** Row reuse for a multi-column (e-reader) pane.
  *
  * Row reuse is keyed by one row index per pane. A column-mode pane paints one snapshot per page column, so its rows
  * are numbered across the columns: column 0's rows first, then column 1's, and so on. Every consumer of
  * [[PaneFrameRecord]] -- damage translation, the comparison with the previous frame, the pixel bands kept -- then
  * works on a multi-column pane exactly as it does on a single-column one, and [[localRows]] hands each column back the
  * rows of its own it has to draw.
  */
private[renderer] object ColumnPaneRows:

  /** Every column's rows as one snapshot, the page's geometry taken from its first column; `None` for a pane not in
    * column mode.
    */
  def flatSnapshot(columns: Vector[ColumnSnapshotPlacement]): Option[TextLayoutSnapshot] =
    columns.headOption.map(first =>
      first.snapshot.copy(
        visualLines = columns.flatMap(_.snapshot.visualLines),
        columnBreaks = firstRows(columns).drop(1)
      )
    )

  /** The pane row each column's first row has, in column order. */
  def firstRows(columns: Vector[ColumnSnapshotPlacement]): Vector[Int] =
    columns.scanLeft(0)(_ + _.snapshot.visualLines.size).dropRight(1)

  /** Each column's share of the pane-wide `dirtyRows`, as that column's own row indices; `None` where the pane draws
    * every row.
    */
  def localRows(columns: Vector[ColumnSnapshotPlacement], dirtyRows: Option[Set[Int]]): Vector[Option[Set[Int]]] =
    columns.zip(firstRows(columns)).map { (column, first) =>
      val rowCount = column.snapshot.visualLines.size
      dirtyRows.map(_.collect { case row if row >= first && row < first + rowCount => row - first })
    }

  /** Each column's share of a pane-wide `rows` to repaint, as that column's own row indices. */
  def localRepaints(columns: Vector[ColumnSnapshotPlacement], rows: Repaint[Int]): Vector[Repaint[Int]] =
    localRows(columns, Option.unless(rows.everything)(rows.parts)).map(_.fold(Repaint.all[Int])(Repaint.only))

/** The pixels of a multi-column pane's line-number rails, which a frame repaints a row band at a time.
  *
  * A row owns its rail band from where its label starts to where the next row's does, so the ink a label spills into
  * the gap beside it stays with it. The band under the last row is owned by no row and is repainted every frame: it is
  * the one place a column that gained or lost a row changes without a row of it being dirty.
  */
private[renderer] object ColumnRails:

  def railRect(contentRect: LayoutRect, placement: ColumnSnapshotPlacement): LayoutRect =
    LayoutRect(
      contentRect.x + placement.xOffsetCells,
      contentRect.y,
      placement.gutterWidthCells,
      contentRect.height
    )

  def rowBands(context: RenderContext, rail: LayoutRect, snapshot: TextLayoutSnapshot): Vector[PixelRect] =
    val rows = RendererPaneSetup.paneRowRects(rail, context, snapshot)
    val top  = RendererChromePlan.pixelRectOf(context.cellMetrics)(rail).yPx
    rows.indices.toVector.map { index =>
      val row   = rows(index)
      val from  = if index == 0 then top.min(row.yPx) else row.yPx
      val until = rows.lift(index + 1).fold(row.bottomPx)(_.yPx.max(row.bottomPx))
      row.copy(yPx = from, heightPx = (until - from).max(0))
    }

  /** The bands of the rows `rows` leaves alone. */
  def keptBands(
    context: RenderContext,
    rail: LayoutRect,
    snapshot: TextLayoutSnapshot,
    rows: Repaint[Int]
  ): List[PixelRect] =
    if rows.everything then Nil
    else
      rowBands(context, rail, snapshot).toList.zipWithIndex.collect { case (band, row) if !rows.includes(row) => band }

  /** The rail pixels no row owns. */
  def unownedBands(context: RenderContext, rail: LayoutRect, snapshot: TextLayoutSnapshot): List[PixelRect] =
    PixelRect.uncoveredWithin(
      RendererChromePlan.pixelRectOf(context.cellMetrics)(rail),
      rowBands(context, rail, snapshot).toList
    )

/** The pre-edit's look while an input method composes (#2008): underlined, with the clause being converted on the
  * selection background. What the text already looked like (syntax colours, weight) is kept under it.
  */
private[renderer] object CompositionStyling:

  def overlay(segments: List[StyledText], row: TextVisualLine, span: Span, theme: Theme): List[StyledText] =
    val from  = (span.startColumn - row.startColumn).max(0)
    val until = (span.endColumn - row.startColumn).min(row.text.length)
    if row.bufferLine != span.line || from >= until then segments
    else
      val clauseEdges = span.clauses.flatMap(clause =>
        List(span.startColumn + clause.start - row.startColumn, span.startColumn + clause.end - row.startColumn)
      )
      val cuts = (from :: until :: clauseEdges).filter(index => index >= from && index <= until).toSet
      val pieces = segments
        .foldLeft((0, List.empty[(StyledText, Int)])) {
          case ((offset, acc), segment) =>
            (offset + segment.content.length, acc ++ piecesOf(segment, offset, cuts))
        }
        ._2
      pieces.map { (piece, start) =>
        if start >= from && start < until then styled(piece, row.startColumn + start, span, theme) else piece
      }

  private def piecesOf(segment: StyledText, offset: Int, cuts: Set[Int]): List[(StyledText, Int)] =
    val inside = cuts.filter(cut => cut > offset && cut < offset + segment.content.length).toList.sorted
    val points = offset :: inside ::: List(offset + segment.content.length)
    points.zip(points.drop(1)).collect {
      case (start, end) if end > start =>
        segment.copy(content = segment.content.substring(start - offset, end - offset)) -> start
    }

  private def styled(piece: StyledText, column: Int, span: Span, theme: Theme): StyledText =
    val selected = span.clauses.exists { clause =>
      clause.selected && column >= span.startColumn + clause.start && column < span.startColumn + clause.end
    }
    val underlined = piece.style.copy(isUnderlined = true)
    if selected then
      piece.copy(
        style = underlined,
        foregroundColor = theme.selection.foreground,
        backgroundColor = theme.selection.background
      )
    else piece.copy(style = underlined)

/** Pure accumulation logic standing in for the renderer's current `WeakHashMap[AnyRef, FrameRecord]`-based structural
  * diff (`frameRecords`/`publishedRecords`), once `#999` finishes wiring `planFrame` to consume `Damage` instead of
  * reconstructing and comparing a full frame description every frame. Landed here, tested standalone; `planFrame` does
  * not read this yet -- see `DamageProducer`'s doc comment for what still has to land before it safely can.
  *
  * Two independent accumulators mirror the two independent comparisons `planFrame` makes today:
  *
  *   - '''Per buffer''' (`accumulateBuffers`/`observeBufferDraw`): "what changed since this specific pixel buffer was
  *     last drawn into." Necessary because `SwingWindow.ReusableImagePool` (`SwingWindow.scala:650-665`) alternates
  *     between two images, so a buffer's own damage history spans however many frames it has been sitting unused, not
  *     just the one frame immediately prior -- accumulating via `Damage`'s `Monoid` across that whole span is exactly
  *     what a `Map[K, Damage]` keyed by buffer identity gives for free.
  *   - '''Screen''' (`accumulateScreen`/`observeScreenPublish`): "what changed since the currently-displayed image was
  *     published." Exactly one buffer is ever on screen at a time, so this needs only one `Damage` value, reset each
  *     time a frame is actually published rather than per buffer identity.
  *
  * Every operation here is a pure function over immutable values so the accumulation semantics can be proven correct in
  * isolation; the mutable, synchronized shell `frameRecords`/`publishedRecords` currently provide is a wiring concern
  * for whichever later change actually calls these from the renderer.
  */
object DamageAccumulator:

  /** Folds `damage` into every buffer identity already being tracked, via `Damage`'s `Monoid`. An identity not yet
    * tracked is left absent -- only [[observeBufferDraw]] starts tracking one, at the point it is first drawn into, so
    * accumulation before that point has nothing to attach to.
    */
  def accumulateBuffers[K](tracked: Map[K, Damage], damage: Damage): Map[K, Damage] =
    if damage == Damage.Nothing then tracked
    else tracked.view.mapValues(_ |+| damage).toMap

  /** The damage accumulated for `identity` since it was last drawn into (`Damage.Nothing` if never seen before), and
    * the updated tracking map with `identity` reset to `Damage.Nothing` -- and, if this is the first time `identity`
    * appears, now present so future [[accumulateBuffers]] calls reach it.
    */
  def observeBufferDraw[K](tracked: Map[K, Damage], identity: K): (Damage, Map[K, Damage]) =
    val damage = tracked.getOrElse(identity, Damage.Nothing)
    (damage, tracked.updated(identity, Damage.Nothing))

  /** Folds `damage` into the screen's accumulated damage since its last publish. */
  def accumulateScreen(current: Damage, damage: Damage): Damage =
    current |+| damage

  /** The damage accumulated since the screen was last published, and the reset (`Damage.Nothing`) value to store going
    * forward.
    */
  def observeScreenPublish(current: Damage): (Damage, Damage) =
    (current, Damage.Nothing)

/** The device pixels that hold a logical rect, clamped to the surface that holds them. */
final private[ui] case class DeviceArea(left: Int, top: Int, width: Int, height: Int):
  def isEmpty: Boolean = width <= 0 || height <= 0

private[ui] object DeviceArea:

  /** A fill at a fractional scale may land a device pixel outside the rect; at a whole scale it lands exactly, which
    * lets neighbouring rects be lifted and put back without touching each other.
    */
  def around(area: PixelRect, scaleX: Double, scaleY: Double, surfaceWidth: Int, surfaceHeight: Int): DeviceArea =
    val margin = if isWhole(scaleX) && isWhole(scaleY) then 0 else 1
    val left   = (math.floor(area.xPx * scaleX).toInt - margin).max(0)
    val top    = (math.floor(area.yPx * scaleY).toInt - margin).max(0)
    val right  = (math.ceil(area.rightPx * scaleX).toInt + margin).min(surfaceWidth)
    val bottom = (math.ceil(area.bottomPx * scaleY).toInt + margin).min(surfaceHeight)
    DeviceArea(left, top, (right - left).max(0), (bottom - top).max(0))

/** Draws a paragraph's drop cap glyph: the oversized first character a [[com.serenity.richtext.ParagraphRole.DropCap]]
  * paragraph singles out (see `RichTextStyling.dropCapSplitFontSpans`), spanning several of the paragraph's normal
  * visual lines. Kept as a sibling of `CharacterRenderer` rather than added to it (which is already at this codebase's
  * size ceiling) -- `RendererPaneContent` calls into this module once per drop-cap paragraph instead of growing either
  * of those files.
  *
  * Measured (GUI) drawing only: [[renderGlyphPx]] draws one pixel-run spanning the glyph's full multi-line height, the
  * same `surface.text.drawRunPx` primitive `CharacterRenderer.renderMeasuredLine` uses per run. The fixed-cell (TUI)
  * grid has no font-size concept to spread a glyph across rows with, so [[renderGlyphCell]] instead marks the one cell
  * the first character already occupies as bold in a highlighted colour -- never attempting multi-row spanning there,
  * per this codebase's TUI degradation convention.
  */
object DropCapRenderer:

  /** Draw the glyph as one oversized pixel run starting at the paragraph's first visual line and tall enough to cover
    * the whole glyph, whatever hangs below its last spanned line.
    *
    * @param xOriginPx
    *   left edge of the paragraph's normal text origin (the glyph draws flush with it; the body text wraps in beside it
    *   at the indent `DropCapLayout.leftInsetPx` reserves)
    * @param yTopPx
    *   top of the paragraph's first visual line
    * @param glyphHeightPx
    *   total pixel height the glyph covers (`DropCapGlyph.heightPx`)
    * @param glyphAscentPx
    *   how far below `yTopPx` the glyph's baseline sits (`DropCapGlyph.baselinePx`), which is the baseline of the last
    *   line the glyph spans
    * @param clipBottomPx
    *   the lowest pixel row the pane draws rows on: a glyph that would reach past it is cut there, since nothing below
    *   it is a row to repaint the glyph's rest from
    */
  def renderGlyphPx(
    surface: RenderSurface,
    xOriginPx: Float,
    yTopPx: Int,
    glyphWidthPx: Float,
    glyphHeightPx: Int,
    glyphAscentPx: Int,
    glyphText: String,
    glyphStyle: TextStyle,
    foreground: RenderColor,
    background: RenderColor,
    clipBottomPx: Int = Int.MaxValue
  ): Unit =
    val drawnHeightPx = glyphHeightPx.min(clipBottomPx - yTopPx)
    if glyphText.nonEmpty && glyphWidthPx > 0.0f && drawnHeightPx > 0 then
      surface.setForegroundColor(foreground)
      surface.setBackgroundColor(background)
      surface.enableStyle(glyphStyle)
      try
        surface.text.drawRunPx(
          xOriginPx,
          yTopPx,
          glyphWidthPx,
          drawnHeightPx,
          glyphAscentPx,
          glyphText,
          clipGlyphToRun = drawnHeightPx < glyphHeightPx
        )
      finally surface.disableStyle(glyphStyle)

  /** TUI fallback: highlight the first character's own cell as bold in `accentForeground`, with no attempt at multi-row
    * spanning (the fixed-cell grid has no font-size concept to do that with).
    */
  def renderGlyphCell(
    surface: RenderSurface,
    x: Int,
    y: Int,
    glyphText: String,
    accentForeground: RenderColor,
    background: RenderColor
  ): Unit =
    if glyphText.nonEmpty then
      surface.setForegroundColor(accentForeground)
      surface.setBackgroundColor(background)
      val boldStyle = TextStyle(isBold = true)
      surface.enableStyle(boldStyle)
      try surface.putString(x, y, glyphText)
      finally surface.disableStyle(boldStyle)

  /** Measured (GUI) home-line hook: paints the paragraph's glyph, fitted to the `dropCap.lines` lines it spans from
    * `lineTopPx` ([[DropCapGlyph]]), then returns the visual line and styled segments the caller's own ordinary
    * body-text draw call should paint instead of its originals -- both with the split-out first grapheme removed, so
    * the glyph is never drawn twice (see the type doc). Returns the inputs unchanged when the paragraph's first run is
    * empty or carries no drop cap glyph style.
    */
  def paintMeasuredHomeLine(
    surface: RenderSurface,
    document: RichTextDocument,
    dropCap: ParagraphRole.DropCap,
    visualLine: TextVisualLine,
    styledSegments: Option[List[StyledText]],
    baseFont: Font,
    frc: FontRenderContext,
    proseScale: Float,
    theme: Theme,
    glyphColour: RenderColor,
    xOriginPx: Float,
    lineTopPx: Int,
    memos: LayoutMemos,
    clipBottomPx: Int = Int.MaxValue,
    shape: LetterShape = LetterShape.Drop
  ): (TextVisualLine, Option[List[StyledText]]) =
    DropCapGlyph.forParagraph(
      baseFont,
      frc,
      Some(document),
      visualLine.bufferLine,
      dropCap,
      proseScale,
      memos,
      shape
    ) match
      case None => (visualLine, styledSegments)
      case Some(glyph) =>
        renderGlyphPx(
          surface,
          xOriginPx,
          lineTopPx,
          glyph.widthPx,
          glyph.heightPx + glyph.growthPx,
          glyph.baselinePx + glyph.growthPx,
          glyph.text,
          glyph.style,
          glyphColour,
          theme.background,
          clipBottomPx
        )
        (dropFirstGrapheme(visualLine, glyph.charCount), styledSegments.map(dropLeadingChars(_, glyph.charCount)))

  /** The colour the configured `document.drop_cap_colour` makes of the glyph, in the theme and heading settings `state`
    * carries.
    */
  private def glyphColourOf(state: AppState): RenderColor =
    state.persisted.config.documentConfig.dropCapColour.resolve(state.persisted.theme, state.effectiveHeadingSettings)

  /** Cell-grid (TUI) home-line hook: re-styles the one cell the caller's own ordinary text draw call already painted --
    * see [[renderGlyphCell]]'s own doc comment for why the fixed grid never attempts multi-row spanning.
    */
  def paintCellHomeLine(
    surface: RenderSurface,
    visualLine: TextVisualLine,
    x: Int,
    y: Int,
    theme: Theme,
    glyphColour: RenderColor
  ): Unit =
    visualLine.text.headOption.foreach { firstChar =>
      renderGlyphCell(surface, x, y, firstChar.toString, glyphColour, theme.background)
    }

  /** Drops the paragraph's split-out glyph character from a visual line's own drawable geometry: its text and the caret
    * stops at or past its new (shifted) start column -- the earlier stops belonged to the character the glyph itself
    * now paints. Keeping their `xPx` and the line's `xOffsetPx` rather than rebasing to 0 keeps the remaining text
    * exactly where `TextLayoutSnapshot` placed it: that layout takes the dropped character's width off the home row's
    * offset, so the text that remains starts at the indent every other spanned row starts at.
    */
  private def dropFirstGrapheme(visualLine: TextVisualLine, charCount: Int): TextVisualLine =
    val newStartColumn = visualLine.startColumn + charCount
    val row            = visualLine.row
    def keptFrom(xs: IArray[Float], offsets: IArray[Int]): (IArray[Float], IArray[Int]) =
      val kept = offsets.indices.filter(offsets(_) >= charCount)
      (IArray.from(kept.map(xs(_))), IArray.from(kept.map(offsets(_) - charCount)))
    val (caretXs, caretOffsets)     = keptFrom(row.caretXs, row.caretOffsets)
    val (landingXs, landingOffsets) = keptFrom(row.landingXs, row.landingOffsets)
    visualLine.copy(row =
      row.copy(
        text = row.text.drop(charCount),
        startColumn = newStartColumn,
        caretXs = caretXs,
        caretOffsets = caretOffsets,
        landingXs = landingXs,
        landingOffsets = landingOffsets
      )
    )

  /** Drops the same leading `count` characters from a parallel list of styled draw segments -- the coloured-text twin
    * of [[dropFirstGrapheme]], so the two stay aligned for the caller's draw call.
    */
  @annotation.tailrec
  private def dropLeadingChars(segments: List[StyledText], count: Int): List[StyledText] =
    segments match
      case StyledText(content, _, _, _) :: rest if count > 0 && content.length <= count =>
        dropLeadingChars(rest, count - content.length)
      case StyledText(content, style, fg, bg) :: rest if count > 0 =>
        StyledText(content.drop(count), style, fg, bg) :: rest
      case other => other

  /** Shared home-line detection for the two hooks below: the drop cap role (if any) `visualLine`'s own paragraph should
    * paint a glyph for, as the snapshot's role resolver resolved it (so honouring the `document.drop_caps_enabled`
    * config toggle). `None` when there's no rich document, the paragraph isn't a drop cap, or this isn't its home line
    * (`DropCapLayout.homeLineRole`).
    */
  private def homeLineRoleFor(snapshot: TextLayoutSnapshot, visualLine: TextVisualLine): Option[ParagraphRole.DropCap] =
    snapshot.richTextDocument.flatMap(doc =>
      DropCapLayout.homeLineRole(snapshot.roles.layoutRole(doc, visualLine.bufferLine), visualLine.startColumn)
    )

  /** `RendererPaneContent`'s one hook for the ordinary body-text draw call: on a drop cap paragraph's home line, when
    * painting through the measured (GUI) path (`measured`), this also paints the glyph itself (see
    * [[paintMeasuredHomeLine]]) and returns the split visual line/segments to draw instead. Every other case (cell/TUI
    * path, no drop cap, config toggle off, wrong line, no `FontRenderContext`) returns `(visualLine, styledSegments)`
    * unchanged -- the cell path's own glyph overlay is the separate, later [[paintCellHomeLineIfNeeded]] hook, since it
    * paints on top of the ordinary text instead of replacing it.
    */
  def adjustHomeLineDraw(
    snapshot: TextLayoutSnapshot,
    visualLine: TextVisualLine,
    styledSegments: Option[List[StyledText]],
    measured: Boolean,
    surface: RenderSurface,
    baseFont: Font,
    frc: Option[FontRenderContext],
    theme: Theme,
    glyphColour: RenderColor,
    xOriginPx: Float,
    lineTopPx: Int,
    memos: LayoutMemos,
    clipBottomPx: Int = Int.MaxValue
  ): (TextVisualLine, Option[List[StyledText]]) =
    (for
      _         <- Option.when(measured)(())
      document  <- snapshot.richTextDocument
      dropCap   <- homeLineRoleFor(snapshot, visualLine)
      renderFrc <- frc
    yield paintMeasuredHomeLine(
      surface,
      document,
      dropCap,
      visualLine,
      styledSegments,
      baseFont,
      renderFrc,
      snapshot.proseScale,
      theme,
      glyphColour,
      xOriginPx,
      lineTopPx,
      memos,
      clipBottomPx,
      snapshot.roles.shapeAt(document, visualLine.bufferLine)
    )).getOrElse((visualLine, styledSegments))

  /** `RendererPaneContent`'s single call for the measured (GUI) path: computes the drop-cap-adjusted line/segments
    * ([[adjustHomeLineDraw]]) and paints them via `CharacterRenderer.renderMeasuredLine` -- keeping that whole call out
    * of `RendererPaneContent`, which is already at this codebase's size ceiling.
    */
  def renderMeasuredHomeAware(
    snapshot: TextLayoutSnapshot,
    state: AppState,
    visualLine: TextVisualLine,
    styledSegments: Option[List[StyledText]],
    context: RenderContext,
    buffer: Buffer,
    xOriginPx: Float,
    lineTopPx: Int,
    lineHeightPx: Int,
    ascentPx: Int,
    clipRightXPx: Float,
    semanticTokens: Option[List[SemanticToken]],
    clipBottomPx: Int = Int.MaxValue
  ): Unit =
    val theme = state.persisted.theme
    val (drawLine, drawSegments) = adjustHomeLineDraw(
      snapshot,
      visualLine,
      styledSegments,
      true,
      context.surface,
      context.fontForBuffer(buffer),
      context.surface.text.fontRenderContext,
      theme,
      glyphColourOf(state),
      xOriginPx,
      lineTopPx,
      context.caches.wrappedLines.layoutMemos,
      clipBottomPx
    )
    CharacterRenderer.renderMeasuredLine(
      context.surface,
      xOriginPx,
      lineTopPx,
      lineHeightPx,
      ascentPx,
      drawLine,
      theme,
      state.syntaxHighlightingEnabled,
      buffer.document.language,
      drawSegments,
      clipRightXPx = Some(clipRightXPx),
      semanticTokens = semanticTokens,
      highlightCache = context.caches.themeHighlightCache,
      graphemeCache = context.caches.graphemeSegmentationCache
    )

  /** `RendererPaneContent`'s post-draw hook for the cell/TUI path: paints the drop cap glyph overlay
    * ([[paintCellHomeLine]]) on a drop cap paragraph's home line, a no-op everywhere else.
    */
  def paintCellHomeLineIfNeeded(
    snapshot: TextLayoutSnapshot,
    state: AppState,
    visualLine: TextVisualLine,
    surface: RenderSurface,
    x: Int,
    y: Int
  ): Unit =
    homeLineRoleFor(snapshot, visualLine).foreach(_ =>
      paintCellHomeLine(surface, visualLine, x, y, state.persisted.theme, glyphColourOf(state))
    )

/** Makes the first use of a font, at a render context, happen once and on one thread before anything is measured or
  * shaped with it. AWT's first use of a font is not safe against another thread's first use of it: readings taken while
  * two threads warm the same font have been seen to differ from the settled ones, and every long-lived cache (shaped
  * text, line extents, glyph advances) would keep the wrong one. The warm-up runs under one lock and a thread that
  * finds the font already settled takes no lock and allocates nothing, so only the first use of each font waits.
  *
  * Every path that measures or shapes text calls [[settle]] with the font it is about to use; a path that does not
  * (painting through `Graphics2D` directly) can still race a measurement that is the font's first use.
  */
final class FontFirstUse:
  import FontFirstUse.*

  private val settled = new ConcurrentHashMap[Font, java.util.Map[FontRenderContext, java.lang.Boolean]]()
  private val lock    = new Object

  def isSettled(font: Font, frc: FontRenderContext): Boolean =
    settled.getOrDefault(font, NoContexts).containsKey(frc)

  def settle(font: Font, frc: FontRenderContext): Unit = settle(font, frc, warmUp)

  private[ui] def settle(font: Font, frc: FontRenderContext, firstUse: (Font, FontRenderContext) => Unit): Unit =
    if !isSettled(font, frc) then
      lock.synchronized {
        if !isSettled(font, frc) then
          firstUse(font, frc)
          if settled.size >= MaxSettled then settled.clear()
          val _ = settled
            .computeIfAbsent(font, _ => new ConcurrentHashMap[FontRenderContext, java.lang.Boolean]())
            .put(frc, java.lang.Boolean.TRUE)
      }

  def settleAll(fonts: Iterable[Font], frc: FontRenderContext): Unit = fonts.foreach(settle(_, frc))

object FontFirstUse:

  private val MaxSettled = 4096

  private val NoContexts: java.util.Map[FontRenderContext, java.lang.Boolean] = java.util.Map.of()

  private val WarmUpText = "Hoxi AVATAR fi To"

  /** One for the process: the fonts AWT has warmed are the process's. */
  val shared: FontFirstUse = new FontFirstUse()

  def settle(font: Font, frc: FontRenderContext): Unit = shared.settle(font, frc)

  def settleAll(fonts: Iterable[Font], frc: FontRenderContext): Unit = shared.settleAll(fonts, frc)

  def isSettled(font: Font, frc: FontRenderContext): Boolean = shared.isSettled(font, frc)

  def warmUp(font: Font, frc: FontRenderContext): Unit =
    val chars = WarmUpText.toCharArray
    val _     = font.getLineMetrics(WarmUpText, frc)
    val _     = font.getStringBounds(WarmUpText, frc)
    val nomnl = font.createGlyphVector(frc, chars)
    val _     = (0 until nomnl.getNumGlyphs).map(nomnl.getGlyphMetrics(_).getAdvance)
    val laid  = font.layoutGlyphVector(frc, chars, 0, chars.length, Font.LAYOUT_LEFT_TO_RIGHT)
    val _     = laid.getGlyphPositions(0, laid.getNumGlyphs + 1, new Array[Float](2 * (laid.getNumGlyphs + 1)))
    val text  = AttributedString(WarmUpText)
    text.addAttribute(TextAttribute.FONT, font)
    val layout = TextLayout(text.getIterator, frc)
    val _      = layout.getAdvance
package com.serenity.ui.renderer

import java.awt.Font
import java.awt.font.FontRenderContext
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.atomic.{AtomicLong, AtomicReference}

/** What was read off a font by shaping a few characters of it, kept by the owner that asked: whether its layout advance
  * is the sum of its glyphs', whether its tables substitute glyphs, whether its advances are uniform, how a pair of
  * characters kerns, whether a row may be cut before a character, and whether it is measured glyph by glyph.
  *
  * Every kind of probe shares one bounded memo. Two generations stand in for an eviction order, as in
  * [[ShapedTextCache]]: a hit in the older one moves the answer to the newer, and when the newer fills the older is
  * dropped, so the memo holds at most twice `capacityPerGeneration` answers however many fonts an owner meets. An
  * answer that was dropped is read off the font again.
  */
final class FontProbes(capacityPerGeneration: Int = FontProbes.DefaultCapacity):
  import FontProbes.*

  private val generations = AtomicReference(Generations(newMap(), newMap()))
  private val probes      = AtomicLong(0)

  /** How many probes ran, as opposed to being answered from what an earlier one found. */
  def probesRun: Long = probes.get

  private def lookup(key: Key): Option[AnyRef] =
    val held = generations.get()
    Option(held.current.get(key)).orElse(Option(held.previous.get(key)).map(remember(key, _)))

  private def flag(key: Key)(answer: => Boolean): Boolean =
    lookup(key) match
      case Some(found: java.lang.Boolean) => found
      case _ =>
        val _        = probes.incrementAndGet()
        val measured = answer
        val _        = remember(key, measured: java.lang.Boolean)
        measured

  private def kerning(key: Key)(answer: => Float): Float =
    lookup(key) match
      case Some(found: java.lang.Float) => found
      case _ =>
        val _        = probes.incrementAndGet()
        val measured = answer
        val _        = remember(key, measured: java.lang.Float)
        measured

  private def remember(key: Key, answer: AnyRef): AnyRef =
    val held = generations.get()
    if held.current.size < capacityPerGeneration then
      val _ = held.current.put(key, answer)
    else
      val next = Generations(newMap(), held.current)
      val _    = next.current.put(key, answer)
      val _    = generations.compareAndSet(held, next)
    answer

  def layoutAgreement(key: Probe)(read: Probe => Boolean): Boolean =
    flag(Key(Kind.LayoutAgreement, key.font, key.frc, NoChar, NoChar))(read(key))

  def substitution(key: Probe)(read: Probe => Boolean): Boolean =
    flag(Key(Kind.Substitution, key.font, key.frc, NoChar, NoChar))(read(key))

  def uniformAdvance(key: Probe)(read: Probe => Boolean): Boolean =
    flag(Key(Kind.UniformAdvance, key.font, key.frc, NoChar, NoChar))(read(key))

  def pairKerning(key: PairProbe)(read: PairProbe => Float): Float =
    kerning(Key(Kind.PairKerning, key.font, key.frc, key.first, key.second))(read(key))

  def barrier(key: BarrierProbe)(read: BarrierProbe => Boolean): Boolean =
    flag(Key(Kind.Barrier, key.font, key.frc, key.next, NoChar))(read(key))

  def measuredLayout(key: Probe)(read: Probe => Boolean): Boolean =
    flag(Key(Kind.MeasuredLayout, key.font, key.frc, NoChar, NoChar))(read(key))

object FontProbes:

  val DefaultCapacity: Int = 4096

  final case class Probe(font: Font, frc: FontRenderContext)

  final case class PairProbe(font: Font, frc: FontRenderContext, first: Char, second: Char)

  final case class BarrierProbe(font: Font, frc: FontRenderContext, next: Char)

  private enum Kind:
    case LayoutAgreement, Substitution, UniformAdvance, PairKerning, Barrier, MeasuredLayout

  private val NoChar = '\u0000'

  final private case class Key(kind: Kind, font: Font, frc: FontRenderContext, first: Char, second: Char)

  private type Answers = ConcurrentHashMap[Key, AnyRef]

  final private case class Generations(current: Answers, previous: Answers)

  private def newMap(): Answers = new ConcurrentHashMap[Key, AnyRef]()

/** Feeds [[FrameCounters]] from the two ends of [[RendererFramePlanner.renderFrame]]: what the plan decided before the
  * frame drew, and what the surface did once it flushed.
  */
private[renderer] object FrameCounting:

  /** `plan` is `None` for a frame that repaints everything, chrome and the whole cell grid included. */
  def planned(counters: FrameCounters, plan: Option[FramePlan], viewport: ViewportSize, metrics: CellMetrics): Unit =
    if counters.isEnabled then
      counters.add(FrameCounter.ChromePainted, if plan.forall(_.chrome.paintsAny) then 1L else 0L)
      counters.add(FrameCounter.PresentAreaPx, presentedArea(plan.flatMap(_.repaintRegion), viewport, metrics))

  def overlayModelBuilt(counters: FrameCounters): Unit = counters.add(FrameCounter.OverlayModelBuilds, 1L)

  /** `dirtyRows` is the dirty-region contract of [[RendererPaneContent]]: `None` draws all `total` rows. */
  def rowsPainted(counters: FrameCounters, total: Int, dirtyRows: Option[Set[Int]]): Unit =
    if counters.isEnabled then counters.rows(total, dirtyRows.fold(total)(_.size))

  /** Closes the frame once its surface has flushed, since the bytes a terminal wrote are known only then. The surface
    * is drained even with counting off, so what it gathered then is not charged to the first frame counted.
    */
  def flushed(counters: FrameCounters, surface: RenderSurface): Unit =
    surface.activity.drainInto(counters)
    counters.frameFinished()

  private def presentedArea(region: Option[List[PixelRect]], viewport: ViewportSize, metrics: CellMetrics): Long =
    region.fold(viewport.width.toLong * metrics.charWidth * viewport.height * metrics.lineHeight)(
      _.foldLeft(0L)((area, rect) => area + rect.widthPx.toLong * rect.heightPx)
    )

/** A panel painted through a layer buffer: the cells it is placed in, the pixels it paints, and the pixels its buffer
  * covers.
  *
  * `painted` is what the panel can change on the frame; `bounds` is that with a margin, so a buffer cut to it still
  * holds everything the panel draws.

final private[renderer] case class PanelSite(
    id: SurfaceId,
    rect: LayoutRect,
    painted: PixelRect,
    bounds: PixelRect,
    content: Any
)

private[renderer] object PanelSite:
  private val MarginCells = 2

  def apply(
    id: SurfaceId,
    rect: LayoutRect,
    verticalOffsetRows: Double,
    metrics: CellMetrics,
    window: PixelRect,
    content: Any
  ): PanelSite =
    val left    = metrics.toPixelX(rect.x)
    val top     = metrics.toPixelY(rect.y)
    val width   = metrics.toPixelX(rect.right) - left
    val height  = metrics.toPixelY(rect.bottom) - top
    val shift   = math.ceil(math.abs(FloatingSurfaceGeometry.signedRowOffsetPixels(verticalOffsetRows, metrics))).toInt
    val painted = PixelRect(left, top - shift, width, height + 2 * shift)
    val marginX = MarginCells * metrics.charWidth
    val marginY = MarginCells * metrics.lineHeight
    val inflated =
      PixelRect(
        painted.xPx - marginX,
        painted.yPx - marginY,
        painted.widthPx + 2 * marginX,
        painted.heightPx + 2 * marginY
      )
    PanelSite(id, rect, painted, inflated.intersection(window).getOrElse(window), content)

    FrameCounting.overlayModelBuilt(context.caches.frameCounters)

      context.surface.viewportHeight * context.cellMetrics.lineHeight

    val panels = FramePanels(overlays, Map.empty, Set.empty)

      overlay.surfaceId.map(

/** Which panels a frame can leave exactly as the surface already holds them. */
private[renderer] object PanelReuse:

  final case class Frame(viewportWidth: Int, viewportHeight: Int, cursorVisible: Boolean)

  /** A panel is kept when what `holds` says the surface holds is current, and no other layer has repainted, or will
    * repaint, the pixels it covers: nothing in `repainted` touches it, no other panel shares them, and no other
    * floating panel stood there last frame (what it left behind would be kept with it).
    */
  def kept(
    panels: FramePanels,
    holds: PanelSite => Boolean,
    isDirty: SurfaceId => Boolean,
    repainted: List[PixelRect],
    previousFloating: Map[SurfaceId, PixelRect]
  ): Set[SurfaceId] =
    val sites = panels.sites.values.toList
    sites.collect {
      case site
          if !isDirty(site.id) && holds(site) && !repainted.exists(_.intersects(site.painted)) &&
            !sites.exists(other => other.id != site.id && other.painted.intersects(site.painted)) &&
            !previousFloating.exists((id, rect) => id != site.id && rect.intersects(site.painted)) =>
        site.id
    }.toSet

/** A glyph's own advance, which `GlyphVector.getGlyphMetrics` also reports but only after measuring the glyph's
  * outline: a native call per glyph on macOS, which made measuring a page of text there cost minutes. A vector of the
  * one glyph ends where its advance does, and the answer is a function of font, render context and glyph alone, so it
  * is kept. A run or a stretch asks for its font's [[GlyphAdvance.Table]] once and reads every glyph from that; the
  * hash lookup on font and render context is paid per run, not per glyph. `maxTables` only stops a long session of
  * ever-different fonts from growing the map without limit.
  */
final class GlyphAdvance(maxTables: Int = GlyphAdvance.DefaultMaxTables):
  import GlyphAdvance.*

  private val tables    = new ConcurrentHashMap[TableKey, Table]()
  private val measuredN = AtomicLong(0)

  /** How many glyphs were measured, as opposed to served from a table. */
  def measurements: Long = measuredN.get()

  def table(font: Font, frc: FontRenderContext): Table =
    val key = TableKey(font, frc)
    if tables.size >= maxTables && !tables.containsKey(key) then tables.clear()
    tables.computeIfAbsent(key, key => Table(key.font, key.frc, measuredN))

  def of(font: Font, frc: FontRenderContext, glyphCode: Int): Float = table(font, frc).of(glyphCode)

object GlyphAdvance:

  val DefaultMaxTables: Int = 256

  private val SlotBits = 8
  private val Slots    = 1 << SlotBits
  private val Probes   = 8

  final private case class TableKey(font: Font, frc: FontRenderContext)

  /** The advances of one font's glyphs at one render context, in a fixed array of slots. A glyph lives in one of the
    * [[Probes]] slots from its hash on, and a slot holds the glyph code and its advance in one `Long`, so a reader sees
    * both or neither. A glyph with no room in its probes takes the place of the first, and is measured again when it is
    * next asked for: that costs time and never correctness.
    */
  final class Table private[GlyphAdvance] (font: Font, frc: FontRenderContext, measuredN: AtomicLong):
    private val slots = new AtomicLongArray(Slots)

    def of(glyphCode: Int): Float =
      val home = (glyphCode * 0x9e3779b1) >>> (32 - SlotBits)

      @annotation.tailrec
      def probe(step: Int): Float =
        val slot = (home + step) & (Slots - 1)
        val held = slots.get(slot)
        if held != 0L && (held >>> 32).toInt == glyphCode then java.lang.Float.intBitsToFloat(held.toInt)
        else if held == 0L || step == Probes - 1 then remember(if held == 0L then slot else home, glyphCode)
        else probe(step + 1)

      probe(0)

    def advancesOf(glyphCodes: Array[Int]): Array[Float] =
      val advances = new Array[Float](glyphCodes.length)

      @annotation.tailrec
      def fill(glyph: Int): Array[Float] =
        if glyph == glyphCodes.length then advances
        else
          advances(glyph) = of(glyphCodes(glyph))
          fill(glyph + 1)

      fill(0)

    private def remember(slot: Int, glyphCode: Int): Float =
      val advance = measure(glyphCode)
      slots.set(slot, (glyphCode.toLong << 32) | (java.lang.Float.floatToRawIntBits(advance) & 0xffffffffL))
      advance

    private def measure(glyphCode: Int): Float =
      val _ = measuredN.incrementAndGet()
      FontFirstUse.settle(font, frc)
      font.createGlyphVector(frc, Array(glyphCode)).getGlyphPosition(1).getX.toFloat

  private val cellGridHeightPx = (effectiveLogicalHeightPx / metrics.lineHeight) * metrics.lineHeight

  g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON)
  g.setRenderingHint(RenderingHints.KEY_FRACTIONALMETRICS, RenderingHints.VALUE_FRACTIONALMETRICS_ON)
  g.translate(-imageOriginPx.xPx.toDouble, -imageOriginPx.yPx.toDouble)
  g.scale(deviceScaleX, deviceScaleY)
  g.setFont(font)

  /** The FontRenderContext this surface uses for text layout. Exposed so that TextLayoutSnapshot and other measurement
    * code can use the identical FRC, preventing cursor drift on proportional fonts.
    */
  private val renderContext: FontRenderContext = g.getFontRenderContext()

  private val fgRef                   = AtomicReference(Color.WHITE)
  private val bgRef                   = AtomicReference(Color.BLACK)
  private val lastPaintColorRef       = AtomicReference(Color.WHITE)
  private val baseFontRef             = AtomicReference(font)
  private val logicalPixelRowOverride = AtomicReference[Option[(Int, Int)]](None)

  private val painted       = PaintedPixels()
  private val underlineRef  = AtomicBoolean(false)
  private val userClipDepth = AtomicInteger(0)

  override def setFont(newFont: FontSpec): Unit =
    baseFontRef.set(newFont.toAwt)
    g.setFont(newFont.toAwt)

  override def fontRenderContext: Option[FontRenderContext] = Some(renderContext)

  override def drawRunPx(
    xPx: Float,
    yPx: Int,
    bgWidthPx: Float,
    lineHeightPx: Int,
    ascentPx: Int,
    s: String,
    clipGlyphToRun: Boolean = false
  ): Unit =
    paintRun(xPx, yPx, bgWidthPx, lineHeightPx, ascentPx, s.nonEmpty, clipGlyphToRun)(
      g.drawString(s, xPx, (yPx + ascentPx).toFloat)
    )

  override def drawShapedRunPx(
    xPx: Float,
    yPx: Int,
    bgWidthPx: Float,
    lineHeightPx: Int,
    ascentPx: Int,
    stretch: String,
    from: Int,
    until: Int,
    clipGlyphToRun: Boolean
  ): Unit =
    paintRun(xPx, yPx, bgWidthPx, lineHeightPx, ascentPx, from < until, clipGlyphToRun) {
      val shaped = activity.counting(shapedText)(shapedText.stretchOf(stretch, g.getFont, renderContext))
      shaped.glyphsOf(from, until).foreach(g.drawGlyphVector(_, xPx, (yPx + ascentPx).toFloat))
    }

  override def withFallbackFamily(family: String, ligaturesOn: Boolean)(draw: => Unit): Unit =
    styledFonts.drawingIn(g, family, ligaturesOn)(draw)

  override def shapesStretchesOf(text: String): Boolean =
    ShapedTextCache.worthShaping(baseFontRef.get(), text)

  private def paintRun(
    xPx: Float,
    yPx: Int,
    bgWidthPx: Float,
    lineHeightPx: Int,
    ascentPx: Int,
    hasText: Boolean,
    clipGlyphToRun: Boolean
  )(drawGlyphs: => Unit): Unit =
    val boundedLeft   = runLeft(xPx)
    val boundedRight  = runRight(xPx, bgWidthPx)
    val boundedTop    = runTop(yPx)
    val boundedBottom = runBottom(yPx, lineHeightPx)

    if boundedLeft < boundedRight && boundedTop < boundedBottom then
      activity.runDrawn()
      val boundedWidth  = boundedRight - boundedLeft
      val boundedHeight = boundedBottom - boundedTop
      if painted.needsBackground(boundedLeft, boundedTop, boundedRight, boundedBottom, bgRef.get()) then
        fillBackground(bgRef.get(), boundedLeft, boundedTop, boundedWidth, boundedHeight)
        painted.ranOver(boundedLeft, boundedTop, boundedRight, boundedBottom)
      if hasText then
        g.setColor(fgRef.get())
        val savedClip = if userClipDepth.get() == 0 then None else Option(g.getClip)
        try
          if clipGlyphToRun then g.clipRect(boundedLeft, boundedTop, boundedWidth, boundedHeight)
          else g.clipRect(0, 0, cellGridWidthPx, cellGridHeightPx)
          drawGlyphs
        finally Java2DRenderSurface.restoreClip(g, savedClip)
        drawUnderline(boundedLeft, boundedWidth, yPx + ascentPx)
        painted.ranOver(boundedLeft, boundedTop, boundedRight, boundedBottom)

  private def runLeft(xPx: Float): Int = math.floor(xPx.toDouble).toInt.max(0).min(cellGridWidthPx)

  private def runRight(xPx: Float, bgWidthPx: Float): Int =
    math.ceil((xPx + bgWidthPx).toDouble).toInt.max(0).min(cellGridWidthPx)

  private def runTop(yPx: Int): Int = yPx.max(0).min(cellGridHeightPx)

  private def runBottom(yPx: Int, lineHeightPx: Int): Int = (yPx + lineHeightPx).max(0).min(cellGridHeightPx)

  /** An underline is a rule under the run rather than a font attribute, so an underlined run draws with the font of a
    * plain one. Its thickness never drops below one device pixel.
    */
  private def drawUnderline(leftPx: Int, widthPx: Int, baselinePx: Int): Unit =
    if underlineRef.get() && widthPx > 0 then
      val underline = styledFonts.underlineOf(g.getFont)
      val thickness = underline.thicknessPx.max((1.0 / deviceScaleY).toFloat)
      g.setColor(fgRef.get())
      g.fill(new Rectangle2D.Float(leftPx.toFloat, baselinePx + underline.offsetPx, widthPx.toFloat, thickness))

  override def drawLabelPx(
    xPx: Float,
    yPx: Int,
    bgWidthPx: Float,
    lineHeightPx: Int,
    ascentPx: Int,
    s: String
  ): Unit =
    LabelAnchor.of(g.getTransform, xPx, yPx + ascentPx) match
      case Some(anchor) if s.nonEmpty && userClipDepth.get() == 0 =>
        drawRunPx(xPx, yPx, bgWidthPx, lineHeightPx, ascentPx, "")
        blitLabel(anchor, s)
        val left = runLeft(xPx)
        drawUnderline(left, runRight(xPx, bgWidthPx) - left, yPx + ascentPx)
        painted.ranOver(left, runTop(yPx), runRight(xPx, bgWidthPx), runBottom(yPx, lineHeightPx))
      case _ => drawRunPx(xPx, yPx, bgWidthPx, lineHeightPx, ascentPx, s)

  private def blitLabel(anchor: LabelAnchor, s: String): Unit =
    val sprite = labelSprites.spriteFor(
      LabelSprites.Key(g.getFont, anchor.scaleX, anchor.scaleY, fgRef.get().getRGB, s, anchor.phaseSixteenths),
      renderContext
    )
    val device = image.createGraphics()
    try
      device.setComposite(g.getComposite)
      device.clipRect(0, 0, cellGridWidthPx * anchor.scaleX, cellGridHeightPx * anchor.scaleY)
      val _ = device.drawImage(
        sprite.image,
        anchor.deviceX - sprite.originX,
        anchor.deviceBaseline - sprite.baseline,
        Java2DRenderSurface.NoOpImageObserver
      )
    finally device.dispose()

  def setForegroundColor(color: RenderColor): Unit = fgRef.set(Java2DRenderSurface.awtColor(color, fgRef.get()))
  def setBackgroundColor(color: RenderColor): Unit = bgRef.set(Java2DRenderSurface.awtColor(color, bgRef.get()))
  def getBackgroundColor: RenderColor              = RenderColor.fromAwt(bgRef.get())

  private def paintColor(color: RenderColor): Color =
    val awt = Java2DRenderSurface.awtColor(color, lastPaintColorRef.get())
    lastPaintColorRef.set(awt)
    awt

  /** The backing image doubles as the persistence key: whoever hands the same image back next frame gets the pixels
    * this frame leaves behind. Only surfaces built with `contentPersists` advertise it, because an image the caller
    * intends to hand out once carries no promise about what it will contain next time.
    */
  override def persistentContentKey: Option[SurfaceContentIdentity] =
    Option.when(contentPersists)(SurfaceContentIdentity(image))

  /** Fill `px, py, pw, ph` with `color`, guaranteeing the pixels actually become exactly `color` -- including fully
    * transparent (alpha 0, the "use the terminal/desktop's own backdrop" sentinel, #1240) -- rather than the no-op the
    * default SRC_OVER composite gives a zero-alpha fill against a buffer that may already carry opaque pixels from a
    * previous frame (this surface's backing image can be pooled/reused across frames, see `contentPersists`). Restores
    * whatever composite (e.g. an active `setAlpha` translucency scale) was in effect before the fill, so this only
    * affects the one fill call, never anything drawn after it. A frame image has no alpha, and the canvas shows black
    * through a transparent pixel, so it gets that black.
    */
  private def fillBackground(color: Color, px: Int, py: Int, pw: Int, ph: Int): Unit =
    if color.getAlpha == 0 then
      val savedComposite = g.getComposite
      g.setComposite(AlphaComposite.Src)
      g.setColor(if image.getColorModel.hasAlpha then color else Color.BLACK)
      g.fillRect(px, py, pw, ph)
      g.setComposite(savedComposite)
    else
      g.setColor(color)
      g.fillRect(px, py, pw, ph)

  override def clearViewport(color: RenderColor): Unit =
    setBackgroundColor(color)
    fillBackground(bgRef.get(), 0, 0, effectiveLogicalWidthPx, effectiveLogicalHeightPx)
    painted.cleared(
      bgRef.get(),
      scala.collection.immutable.List(PixelRect(0, 0, effectiveLogicalWidthPx, effectiveLogicalHeightPx)),
      clipped = userClipDepth.get() > 0
    )

  override def clearViewportExcept(color: RenderColor, preserved: scala.collection.immutable.List[PixelRect]): Unit =
    if preserved.isEmpty then clearViewport(color)
    else
      setBackgroundColor(color)
      val background = bgRef.get()
      val uncovered =
        PixelRect.uncoveredWithin(PixelRect(0, 0, effectiveLogicalWidthPx, effectiveLogicalHeightPx), preserved)
      uncovered.foreach(rect => fillBackground(background, rect.xPx, rect.yPx, rect.widthPx, rect.heightPx))
      painted.cleared(background, uncovered, clipped = userClipDepth.get() > 0)

  def putString(x: Int, y: Int, s: String): Unit =
    if s.nonEmpty then
      val px      = metrics.toPixelX(x)
      val py      = pixelYForRow(y)
      val widthPx = CellRuns.cellsIn(s) * metrics.charWidth
      // The background covers every cell the string takes, which a wide character makes more than its length.
      fillBackground(bgRef.get(), px, py, widthPx, metrics.lineHeight)
      painted.touched(px, py, widthPx, metrics.lineHeight)
      g.setColor(fgRef.get())
      // A narrow stretch is drawn as one shaped string so font features like ligatures can apply; a wide character is
      // drawn alone on its own cell, since its glyph's advance need not be two of them.
      CellRuns
        .pieces(s)
        .foreach((cell, piece) => g.drawString(piece, px + cell * metrics.charWidth, py + metrics.ascent))
      drawUnderline(px, widthPx, py + metrics.ascent)

  def fillRect(x: Int, y: Int, width: Int, height: Int, char: Char): Unit =
    val px = metrics.toPixelX(x)
    val py = pixelYForRow(y)
    val pw = width * metrics.charWidth
    val ph = height * metrics.lineHeight
    fillBackground(bgRef.get(), px, py, pw, ph)
    painted.touched(px, py, pw, ph)
    if char != ' ' then
      g.setColor(fgRef.get())
      (0 until height).foreach { row =>
        (0 until width).foreach { col =>
          g.drawString(char.toString, metrics.toPixelX(x + col), metrics.toPixelY(y + row) + metrics.ascent)
        }
      }

  override def withLogicalPixelRow(cellRow: Int, pixelY: Int)(render: => Unit): Unit =
    val previous = logicalPixelRowOverride.getAndSet(Some(cellRow -> pixelY))
    try render
    finally logicalPixelRowOverride.set(previous)

  override def withPixelTranslation(xPx: Double, yPx: Double)(render: => Unit): Unit =
    val savedTransform = g.getTransform
    try
      g.translate(xPx, yPx)
      painted.withTranslation(xPx, yPx)(render)
    finally g.setTransform(savedTransform)

  private def pixelYForRow(row: Int): Int =
    logicalPixelRowOverride
      .get()
      .collect { case (cellRow, pixelY) if cellRow == row => pixelY }
      .getOrElse(metrics.toPixelY(row))

  def enableStyle(style: TextStyle): Unit =
    underlineRef.set(style.isUnderlined)
    g.setFont(styledFonts.fontFor(baseFontRef.get(), style))

  def disableStyle(style: TextStyle): Unit =
    underlineRef.set(false)
    g.setFont(baseFontRef.get())

  override def setAlpha(alpha: Float): Unit =
    val clamped = alpha.max(0f).min(1f)
    painted.setTranslucent(clamped < 1f)
    g.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, clamped))

  override def strokeRect(x: Int, y: Int, width: Int, height: Int, color: RenderColor, strokeWidth: Float): Unit =
    val px          = metrics.toPixelX(x)
    val py          = metrics.toPixelY(y)
    val pw          = width * metrics.charWidth
    val ph          = height * metrics.lineHeight
    val inset       = math.ceil(strokeWidth / 2).toInt
    val savedStroke = g.getStroke
    painted.touched(px, py, pw, ph)
    g.setColor(paintColor(color))
    g.setStroke(new BasicStroke(strokeWidth, BasicStroke.CAP_SQUARE, BasicStroke.JOIN_MITER))
    g.drawRect(px + inset, py + inset, pw - 2 * inset, ph - 2 * inset)
    g.setStroke(savedStroke)

  override def withRectClip(x: Int, y: Int, width: Int, height: Int)(render: => Unit): Unit =
    val px        = metrics.toPixelX(x)
    val py        = metrics.toPixelY(y)
    val pw        = width * metrics.charWidth
    val ph        = height * metrics.lineHeight
    val savedClip = g.getClip
    userClipDepth.incrementAndGet()
    try
      g.clip(new Rectangle2D.Double(px, py, pw, ph))
      render
    finally
      userClipDepth.decrementAndGet()
      g.setClip(savedClip)

  override def fillPixelRect(
    xPx: Int,
    yPx: Int,
    widthPx: Int,
    heightPx: Int,
    color: RenderColor
  ): Unit =
    painted.touched(xPx, yPx, widthPx.max(1), heightPx.max(1))
    g.setColor(paintColor(color))
    g.fillRect(xPx, yPx, widthPx.max(1), heightPx.max(1))

  override def drawImage(image: RenderImage, x: Int, y: Int, width: Int, height: Int): Unit =
    val px        = metrics.toPixelX(x)
    val py        = metrics.toPixelY(y)
    val pw        = width * metrics.charWidth
    val ph        = height * metrics.lineHeight
    val savedClip = g.getClip
    painted.touched(px, py, pw, ph)
    g.clipRect(px, py, pw, ph)
    g.drawImage(image.toAwt, px, py, pw, ph, Java2DRenderSurface.NoOpImageObserver)
    g.setClip(savedClip)

  /** Blit a whole-surface layer buffer straight onto this surface's backing image at device resolution, 1:1, bypassing
    * `g`'s cell-grid geometry and its `deviceScale` transform entirely. The layer buffer is produced by [[forLayer]] at
    * exactly this surface's own backing dimensions (`deviceImageDimension(logicalSize, deviceScale)`), so a raw
    * device-pixel copy reproduces it exactly -- see [[PixelDrawing.compositeFullSurfaceLayer]] for why routing it
    * through [[drawImage]] instead would shrink it a little per composite.
    */
  override def compositeFullSurfaceLayer(layerImage: RenderImage): Unit =
    blitLayer(layerImage, DeviceOrigin.Zero)

  override def compositeLayer(layerImage: RenderImage, bounds: PixelRect): Unit =
    blitLayer(layerImage, DeviceOrigin.of(bounds, deviceScaleX, deviceScaleY))

  private def blitLayer(layerImage: RenderImage, origin: DeviceOrigin): Unit =
    activity.layerComposited()
    painted.forgetClear()
    val rawGraphics = image.createGraphics()
    try
      val _ = rawGraphics.drawImage(layerImage.toAwt, origin.xPx, origin.yPx, Java2DRenderSurface.NoOpImageObserver)
    finally rawGraphics.dispose()

  /** In device pixels on an untransformed graphics, because `copyArea` is not implemented for a scaled one. A copy onto
    * an overlapping area runs bottom-up when it moves down, so no source row is overwritten before it is read.
    */
  override def copyPixels(from: PixelRect, dyPx: Int): Unit =
    painted.forgetClear()
    val raw = image.createGraphics()
    try
      raw.setComposite(AlphaComposite.Src)
      raw.copyArea(
        math.round(from.xPx * deviceScaleX).toInt,
        math.round(from.yPx * deviceScaleY).toInt,
        math.round(from.widthPx * deviceScaleX).toInt,
        math.round(from.heightPx * deviceScaleY).toInt,
        0,
        math.round(dyPx * deviceScaleY).toInt
      )
    finally raw.dispose()

  def hideCursor(): Unit = ()

  def viewportWidth: Int                 = effectiveLogicalWidthPx / metrics.charWidth
  def viewportHeight: Int                = effectiveLogicalHeightPx / metrics.lineHeight
  override def devicePixelScaleX: Double = deviceScaleX
  override def devicePixelScaleY: Double = deviceScaleY

  def flush(): Unit =
    g.dispose()
    onFlush(image)

object Java2DRenderSurface:

  final private[serenity] case class DeviceScale(x: Double, y: Double)

  /** `Graphics2D` can only express "no clip" as a null `Shape`. */
  @SuppressWarnings(Array("org.wartremover.warts.Null"))
  private def restoreClip(g: Graphics2D, saved: Option[Shape]): Unit = g.setClip(saved.orNull)

  /** `Graphics2D.drawImage`'s `ImageObserver` callback exists for images that may still be loading asynchronously (e.g.
    * from a URL); every image this class ever draws is an already-fully-materialised `BufferedImage`, so the callback
    * can never fire and a `null` observer (Java's usual shorthand for "don't call me back") is equivalent to this
    * no-op. Returning `false` tells the (never-invoked) caller not to bother scheduling further notifications.
    */
  private[serenity] val NoOpImageObserver: ImageObserver = (_, _, _, _, _, _) => false

  /** Reuses `current` when it already carries `color`, so a renderer re-setting the same theme colour on every run does
    * not allocate a fresh `java.awt.Color` each time.
    */
  private def awtColor(color: RenderColor, current: Color): Color =
    if current.getRGB == color.argb then current else color.toAwt

  /** Build a frame surface over an image supplied by `acquireImage`.
    *
    * `contentPersists` says the acquired image is recycled rather than freshly allocated, so whatever was drawn into
    * that same image instance previously is still there. Callers pass a pooled acquirer together with `true`; a
    * single-use image must stay `false` so nothing downstream tries to reuse pixels that were never kept.
    *
    * The image is `TYPE_INT_RGB`: the window is opaque, and Java2D blends glyphs into it markedly faster than into
    * ARGB.
    *
    * Every frame built over the same `canvas` shares one [[RenderSurface.layerCacheOwner]].
    */
  def forFrame(
    metrics: CellMetrics,
    font: Font,
    canvas: javax.swing.JPanel,
    onFlush: BufferedImage => Unit,
    acquireImage: (Int, Int, Int) => BufferedImage,
    textCaches: SurfaceTextCaches,
    contentPersists: Boolean = true
  ): Java2DRenderSurface =
    val logicalWidth  = logicalCanvasDimension(canvas.getWidth, canvas.getPreferredSize.width)
    val logicalHeight = logicalCanvasDimension(canvas.getHeight, canvas.getPreferredSize.height)
    val scale         = deviceScaleFor(canvas)
    val image = acquireImage(
      deviceImageDimension(logicalWidth, scale.x),
      deviceImageDimension(logicalHeight, scale.y),
      BufferedImage.TYPE_INT_RGB
    )
    new Java2DRenderSurface(
      image,
      metrics,
      font,
      onFlush,
      logicalWidthPx = logicalWidth,
      logicalHeightPx = logicalHeight,
      deviceScaleX = scale.x,
      deviceScaleY = scale.y,
      contentPersists = contentPersists,
      layerCacheOwnerOverride = Some(ScreenIdentity(canvas)),
      textCaches = textCaches
    )

  def forImage(
    image: BufferedImage,
    metrics: CellMetrics,
    font: Font,
    canvas: javax.swing.JPanel,
    onFlush: BufferedImage => Unit,
    textCaches: SurfaceTextCaches
  ): Java2DRenderSurface =
    val logicalWidth  = logicalCanvasDimension(canvas.getWidth, canvas.getPreferredSize.width)
    val logicalHeight = logicalCanvasDimension(canvas.getHeight, canvas.getPreferredSize.height)
    val scale         = deviceScaleFor(canvas)
    new Java2DRenderSurface(
      image,
      metrics,
      font,
      onFlush,
      logicalWidthPx = logicalWidth,
      logicalHeightPx = logicalHeight,
      deviceScaleX = scale.x,
      deviceScaleY = scale.y,
      textCaches = textCaches
    )

  /** Build a layer surface from numbers alone -- no `JPanel` required, unlike [[forFrame]]/[[forImage]]. Stage 1 of
    * #1100 flagged those two as "tied to a Swing `JPanel` for device-scale/logical-size derivation" as the open design
    * problem blocking per-surface buffering; this resolves it by deriving the same inputs from an existing
    * [[Java2DRenderSurface]] that already computed them (see [[Java2DRenderSurface.newLayerSurface]]) instead of from a
    * canvas. The image starts fully transparent: `recycled` is cleared and reused when its size matches, otherwise a
    * new image is allocated.
    */
  def forLayer(
    metrics: CellMetrics,
    font: Font,
    logicalWidthPx: Int,
    logicalHeightPx: Int,
    deviceScaleX: Double,
    deviceScaleY: Double,
    onFlush: BufferedImage => Unit,
    textCaches: SurfaceTextCaches,
    recycled: Option[BufferedImage] = None,
    activity: SurfaceActivity = SurfaceActivity(),
    bounds: Option[PixelRect] = None
  ): Java2DRenderSurface =
    LayerSurfaces.build(
      metrics,
      font,
      logicalWidthPx,
      logicalHeightPx,
      deviceScaleX,
      deviceScaleY,
      onFlush,
      textCaches,
      recycled,
      activity,
      bounds
    )

  private[serenity] def deviceImageDimension(logicalDimensionPx: Int, deviceScale: Double): Int =
    math.ceil(logicalDimensionPx.max(1) * deviceScale.max(1.0)).toInt.max(1)

  private[serenity] def logicalCanvasDimension(currentPx: Int, preferredPx: Int): Int =
    if currentPx > 0 then currentPx
    else preferredPx.max(1)

  private[serenity] def deviceScaleFor(canvas: javax.swing.JPanel): DeviceScale =
    Option(canvas.getGraphicsConfiguration)
      .map(_.getDefaultTransform)
      .map(transform => DeviceScale(transform.getScaleX.max(1.0), transform.getScaleY.max(1.0)))
      .getOrElse(DeviceScale(1.0, 1.0))

import java.awt.geom.AffineTransform

/** Where a label's ink lands in device pixels: its origin on the pixel grid plus the sixteenth of a pixel it starts
  * past it, and the integral device scale it was drawn at.
  */
final private[ui] case class LabelAnchor(
    deviceX: Int,
    deviceBaseline: Int,
    phaseSixteenths: Int,
    scaleX: Int,
    scaleY: Int
)

private[ui] object LabelAnchor:

  /** Only an integral device scale puts a label's baseline on a whole device pixel, which a sprite needs: it is blitted
    * 1:1, never resampled.
    */
  def of(transform: AffineTransform, xPx: Float, baselinePx: Int): Option[LabelAnchor] =
    val deviceBaseline = transform.getTranslateY + transform.getScaleY * baselinePx
    val axisAligned    = transform.getShearX == 0.0 && transform.getShearY == 0.0
    Option.when(
      axisAligned && isWhole(transform.getScaleX) && isWhole(transform.getScaleY) && transform.getScaleX >= 1.0 &&
        transform.getScaleY >= 1.0 && isWhole(deviceBaseline)
    ) {
      val sixteenths = math.round((transform.getTranslateX + transform.getScaleX * xPx) * 16.0)
      LabelAnchor(
        Math.floorDiv(sixteenths, 16L).toInt,
        math.rint(deviceBaseline).toInt,
        Math.floorMod(sixteenths, 16L).toInt,
        transform.getScaleX.toInt,
        transform.getScaleY.toInt
      )
    }

  def isWhole(value: Double): Boolean = math.abs(value - math.rint(value)) < 1e-6
package com.serenity.ui.renderer

import java.awt.font.FontRenderContext
import java.awt.image.BufferedImage
import java.awt.{Color, Font, RenderingHints}
import java.util.concurrent.atomic.{AtomicInteger, AtomicLong}

/** The ink of short labels that repeat verbatim frame after frame -- line numbers -- rendered once and blitted.
  *
  * Laying a label out costs a `TextLayout` per draw, and a gutter redraws a few dozen of them whenever its rows change.
  * What a label looks like is fixed by its font, device scale, colour, text and the sub-pixel phase it starts at, so
  * that tuple keys a cache of its rendered ink. The cache holds at most `capacity` sprites, because the keys of a
  * scroll through a long document are unbounded.
  */
final class LabelSprites(capacity: Int):
  import LabelSprites.*

  private val cache       = new BoundedRefCache[Key, Sprite](new AtomicInteger(capacity))
  private val rasterisedN = new AtomicLong(0)

  def size: Int = cache.snapshot.size

  /** How many sprites have been rendered, as opposed to served from the cache. */
  def rasterised: Long = rasterisedN.get()

  def spriteFor(key: Key, fontRenderContext: FontRenderContext): Sprite =
    cache.get(key).getOrElse {
      val sprite = rasterise(key, fontRenderContext)
      cache.put(key, sprite)
      sprite
    }

  /** Room for an em of ink on every side of the text, so accents, italics and descenders are never clipped. */
  private def rasterise(key: Key, fontRenderContext: FontRenderContext): Sprite =
    val _        = rasterisedN.incrementAndGet()
    val emPx     = math.ceil(key.font.getSize2D.toDouble * math.max(key.scaleX, key.scaleY)).toInt.max(1)
    val inkWidth = math.ceil(key.font.getStringBounds(key.text, fontRenderContext).getWidth * key.scaleX).toInt
    val image    = new BufferedImage(inkWidth + 2 * emPx + 1, 3 * emPx, BufferedImage.TYPE_INT_ARGB)
    val g        = image.createGraphics()
    try
      g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON)
      g.setRenderingHint(RenderingHints.KEY_FRACTIONALMETRICS, RenderingHints.VALUE_FRACTIONALMETRICS_ON)
      g.scale(key.scaleX.toDouble, key.scaleY.toDouble)
      g.setFont(key.font)
      g.setColor(new Color(key.argb, true))
      g.drawString(
        key.text,
        ((emPx + key.phaseSixteenths / 16.0) / key.scaleX).toFloat,
        (2.0 * emPx / key.scaleY).toFloat
      )
    finally g.dispose()
    Sprite(image, originX = emPx, baseline = 2 * emPx)

/** Stable identity for one independently-composited surface within a frame -- the base editor content, the chrome
  * (gutter/line numbers/spacers), the pinned-panel stack, the floating-overlay stack, or the modal layer. Distinct from
  * [[SurfaceContentIdentity]]/[[ScreenIdentity]], which identify pixel buffers and screens for cross-frame damage
  * bookkeeping; a `LayerId` identifies a *role* in a frame's paint order, stable across frames even as the pixels
  * behind it are redrawn.
  */
final case class LayerId(value: String)

/** Per-layer compositing parameters: how a layer's own rendered content is combined onto the frame it sits on. `alpha`
  * is the only knob today -- [[RendererFramePlanner]] still paints every layer directly into the shared frame surface
  * rather than into a buffer of its own, so blur/tint/shadow have nowhere to attach yet. This is the seam #1086
  * (backdrop blur), #1087 (per-surface glass) and #1091 (soft per-surface shadows) extend once each layer owns its own
  * buffer.
  */
final case class LayerEffect(alpha: Float):
  require(alpha >= 0.0f && alpha <= 1.0f, s"LayerEffect.alpha must be in [0, 1], was $alpha")

object LayerEffect:

  /** No compositing adjustment: full opacity, nothing else applied. Every layer in today's frame uses this except the
    * modal backdrop's dimming.
    */
  val identity: LayerEffect = LayerEffect(1.0f)

/** One layer in a frame's composite stack: its position in paint order, the effect blended in as it is composited, and
  * the damage it has accumulated since it was last (re)rendered.
  *
  * A layer whose `damage` is [[Damage.Nothing]] does not need to be redrawn this frame -- once a layer owns a buffer
  * that persists across frames, the compositor can reuse whatever it drew last time instead of re-rendering. Nothing in
  * [[RendererFramePlanner]] owns such a per-layer buffer yet (every layer still paints straight into the shared frame
  * surface, so skipping a layer would just leave a hole), which is why [[LayerCompositor.dirtyLayers]] is not yet
  * called from the render path -- it is exercised by its own tests, ready for the per-layer-buffer follow-up this
  * issue's own performance note asks for.
  */
final case class Layer(id: LayerId, zOrder: Int, effect: LayerEffect, damage: Damage)

/** The compositing seam #1100 introduces: today's single-pass paint sequence (spacer columns, line numbers, gutter,
  * editor panes, pinned panels, floating panels, modal) expressed as an explicit, ordered stack of named layers rather
  * than an implicit sequence of calls. `RendererFramePlanner.renderFrame` builds that stack and paints each layer's
  * already-written function in the order [[orderedForComposite]] resolves, so the visual stacking order lives in one
  * place as data instead of being hard-coded by call sequence -- and the modal backdrop's alpha dimming, the one place
  * today's renderer already blends a layer, goes through [[withEffect]] rather than raw calls to `Effects.setAlpha`.
  */
object LayerCompositor:

  /** Layers in back-to-front paint order -- lower `zOrder` paints first, so later layers land on top. Stable for equal
    * `zOrder` values: layers that tie keep the order they were given in, so the stack's declared order controls
    * ambiguous cases rather than an unspecified sort.
    */
  def orderedForComposite(layers: List[Layer]): List[Layer] =
    layers.sortBy(_.zOrder)

  /** Layers that must be (re)rendered before this frame's composite -- every layer except one with empty damage, whose
    * last-rendered content is still correct and can simply be recomposited unchanged. See [[Layer]]'s doc comment for
    * why nothing calls this from [[RendererFramePlanner]] yet.
    */
  def dirtyLayers(layers: List[Layer]): List[Layer] =
    layers.filterNot(layer => layer.damage == Damage.Nothing)

  /** Run `paint` with `effect`'s alpha applied to `surface`, restoring full opacity afterwards -- even if `paint`
    * throws, so one layer's effect never bleeds into the next layer painted on the same surface. A surface without an
    * [[Effects]] capability (e.g. a headless test double) simply runs `paint` at whatever alpha it already had, since
    * there is no alpha to set.
    */
  def withEffect(surface: RenderSurface)(effect: LayerEffect)(paint: => Unit): Unit =
    surface.effects match
      case None => paint
      case Some(fx) =>
        fx.setAlpha(effect.alpha)
        try paint
        finally fx.setAlpha(LayerEffect.identity.alpha)

  def build(
    metrics: CellMetrics,
    font: Font,
    logicalWidthPx: Int,
    logicalHeightPx: Int,
    deviceScaleX: Double,
    deviceScaleY: Double,
    onFlush: BufferedImage => Unit,
    textCaches: SurfaceTextCaches,
    recycled: Option[BufferedImage],
    activity: SurfaceActivity,
    bounds: Option[PixelRect]
  ): Java2DRenderSurface =

    val width = bounds.fold(Java2DRenderSurface.deviceImageDimension(logicalWidthPx, deviceScaleX))(

      .map(clearedForReuse)
      .getOrElse {
        activity.layerAllocated(width.toLong * height)
        new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB)

      image,
      metrics,
      font,
      onFlush,
      logicalWidthPx = logicalWidthPx,
      logicalHeightPx = logicalHeightPx,
      deviceScaleX = deviceScaleX,
      deviceScaleY = deviceScaleY,
      contentPersists = false,
      imageOriginPx = origin,
      textCaches = textCaches,
      activity = activity
    )

  private def clearedForReuse(image: BufferedImage): BufferedImage =
    val graphics = image.createGraphics()
    try
      graphics.setComposite(AlphaComposite.Clear)
      graphics.fillRect(0, 0, image.getWidth, image.getHeight)
    finally graphics.dispose()
    image

import scala.annotation.tailrec

import com.serenity.state.models.TextVisualLine

/** A stretch of a measured visual line drawn in one call: one colour pair, one style, and its pixel extent measured
  * from the line's left edge.
  */
final private[renderer] case class MeasuredRun(
    text: String,
    foreground: RenderColor,
    background: RenderColor,
    style: TextStyle,
    minXPx: Float,
    maxXPx: Float
)

/** A [[MeasuredRun]] with the stretch of text it shares a style, and so a font, with its neighbours: `stretch` is that
  * text and the run is `stretch.substring(from, until)`.
  */
final private[renderer] case class StretchedRun(run: MeasuredRun, stretch: String, from: Int, until: Int)

/** The part of a measured line drawn in one font: `family` is the fallback family, `None` for the line's own font. Its
  * `runs` are the line's runs cut to the segment, and `text` is the segment, which is what a stretch is shaped from so
  * that shaping never reaches across a font boundary.
  */
final private[renderer] case class FontSegment(family: Option[String], runs: Vector[MeasuredRun], text: String)

/** Splits a measured visual line into [[MeasuredRun]]s wherever a grapheme cluster's effective colours or style differ
  * from the cluster before it, so no cluster is ever split. Runs a full frame of prose through it, so it walks arrays
  * instead of building per-character collections.
  */
private[renderer] object MeasuredLineRuns:

  final private case class Attributes(foreground: RenderColor, background: RenderColor, style: TextStyle)

  def of(
    visualLine: TextVisualLine,
    segments: List[StyledText],
    spans: Vector[GraphemeSpan],
    theme: Theme
  ): Vector[MeasuredRun] =
    val text        = visualLine.text
    val bounds      = ClusterBounds(visualLine, spans)
    val styled      = segments.toArray
    val segmentEnds = styled.scanLeft(0)(_ + _.content.length).drop(1)
    val unstyled    = Attributes(theme.foreground, theme.background, TextStyle.normal)
    val attributesBySegment =
      styled.map(segment => Attributes(segment.foregroundColor, segment.backgroundColor, segment.style))
    val runs = Vector.newBuilder[MeasuredRun]

    @tailrec
    def segmentAt(localIndex: Int, from: Int): Int =
      if from < segmentEnds.length && segmentEnds(from) <= localIndex then segmentAt(localIndex, from + 1) else from

    def attributesOf(segment: Int): Attributes =
      if segment < attributesBySegment.length then attributesBySegment(segment) else unstyled

    def emit(start: Int, end: Int, attributes: Attributes, runText: java.lang.StringBuilder, boundsFrom: Int): Int =
      val (minXPx, maxXPx, next) = bounds.extent(start, end, boundsFrom)
      runs += MeasuredRun(
        runText.toString,
        attributes.foreground,
        attributes.background,
        attributes.style,
        minXPx,
        maxXPx
      )
      next

    def textOf(span: GraphemeSpan): java.lang.StringBuilder =
      new java.lang.StringBuilder().append(text, span.startLocalIndex, span.endLocalIndex)

    @tailrec
    def group(
      spanIndex: Int,
      segment: Int,
      start: Int,
      end: Int,
      attributes: Attributes,
      runText: java.lang.StringBuilder,
      boundsFrom: Int
    ): Unit =
      if spanIndex >= spans.length then
        val _ = emit(start, end, attributes, runText, boundsFrom)
      else
        val span           = spans(spanIndex)
        val spanSegment    = segmentAt(span.startLocalIndex, segment)
        val spanAttributes = attributesOf(spanSegment)
        if spanAttributes == attributes then
          val _ = runText.append(text, span.startLocalIndex, span.endLocalIndex)
          group(spanIndex + 1, spanSegment, start, span.endLocalIndex, attributes, runText, boundsFrom)
        else
          val next = emit(start, end, attributes, runText, boundsFrom)
          group(
            spanIndex + 1,
            spanSegment,
            span.startLocalIndex,
            span.endLocalIndex,
            spanAttributes,
            textOf(span),
            next
          )

    spans.headOption.foreach { first =>
      val segment = segmentAt(first.startLocalIndex, 0)
      group(1, segment, 0, first.endLocalIndex, attributesOf(segment), textOf(first), 0)
    }
    runs.result()

  /** `runs` cut at the boundaries of the line's font runs. The line's own font fills the gaps between them. A run cut
    * in two keeps the line's caret stop at the cut as the edge of both halves.
    */
  def byFont(visualLine: TextVisualLine, runs: Vector[MeasuredRun]): Vector[FontSegment] =
    val text    = visualLine.text
    val offsets = runs.scanLeft(0)(_ + _.text.length)
    val fonts   = visualLine.fontRuns

    def xAt(offset: Int): Float =
      visualLine.row.xForColumn(visualLine.startColumn + offset).fold(0.0f)(_ + visualLine.xOffsetPx)

    def cut(from: Int, until: Int): Vector[MeasuredRun] =
      runs.indices.flatMap { index =>
        val low  = math.max(offsets(index), from)
        val high = math.min(offsets(index + 1), until)
        Option.when(low < high) {
          val run = runs(index)
          run.copy(
            text = run.text.substring(low - offsets(index), high - offsets(index)),
            minXPx = if low == offsets(index) then run.minXPx else xAt(low),
            maxXPx = if high == offsets(index + 1) then run.maxXPx else xAt(high)
          )
        }
      }.toVector

    def segment(from: Int, until: Int, slot: Int): FontSegment =
      FontSegment(
        Option.when(slot > 0)(visualLine.fallbackFamilies.lift(slot - 1)).flatten,
        cut(from, until),
        text.substring(from, until)
      )

    val declared = Vector.tabulate(fonts.length / 3)(run => (fonts(3 * run), fonts(3 * run + 1), fonts(3 * run + 2)))
    val (covered, segments) = declared.foldLeft((0, Vector.empty[FontSegment])) {
      case ((reached, found), (from, until, slot)) =>
        val gap = Option.when(from > reached)(segment(reached, from, 0))
        (until, found ++ gap :+ segment(from, until, slot))
    }
    segments ++ Option.when(covered < text.length)(segment(covered, text.length, 0))

  /** The stretches of `[from, until)` (offsets into the row's text) that one font draws, each with the fallback family
    * that draws it: the line's own font between the fallback runs.
    */
  def fontPieces(visualLine: TextVisualLine, from: Int, until: Int): Vector[(Int, Int, Option[String])] =
    val fonts    = visualLine.fontRuns
    val declared = Vector.tabulate(fonts.length / 3)(run => (fonts(3 * run), fonts(3 * run + 1), fonts(3 * run + 2)))
    def familyOf(slot: Int): Option[String] =
      Option.when(slot > 0)(visualLine.fallbackFamilies.lift(slot - 1)).flatten
    val cuts = declared
      .flatMap((low, high, slot) =>
        val start = math.max(low, from)
        val end   = math.min(high, until)
        Option.when(start < end)((start, end, familyOf(slot)))
      )
    val (reached, pieces) = cuts.foldLeft((from, Vector.empty[(Int, Int, Option[String])])) {
      case ((at, found), (start, end, family)) =>
        (end, found ++ Option.when(start > at)((at, start, None)) :+ ((start, end, family)))
    }
    pieces ++ Option.when(reached < until)((reached, until, None))

  /** Places each run in its style stretch -- the consecutive runs of one style, which draw in one font and so can be
    * shaped as one text -- where `runs` are the runs of `text`.
    */
  def stretched(runs: Vector[MeasuredRun], text: String): Vector[StretchedRun] =
    val offsets = runs.scanLeft(0)(_ + _.text.length)
    if runs.headOption.forall(first => runs.forall(_.style == first.style)) then
      Vector.tabulate(runs.length)(index => StretchedRun(runs(index), text, offsets(index), offsets(index + 1)))
    else mixedStretches(runs, text, offsets)

  private def mixedStretches(runs: Vector[MeasuredRun], text: String, offsets: Vector[Int]): Vector[StretchedRun] =
    val firsts = runs.indices.foldLeft(Vector.empty[Int]) { (found, index) =>
      found :+ (if index > 0 && runs(index - 1).style == runs(index).style then found(index - 1) else index)
    }
    val stretchTexts = runs.indices
      .groupMapReduce(firsts(_))(index => offsets(index + 1))(math.max)
      .map((first, end) => first -> text.substring(offsets(first), end.min(text.length)))
    runs.indices.map { index =>
      val first = firsts(index)
      StretchedRun(
        runs(index),
        stretchTexts.getOrElse(first, runs(index).text),
        offsets(index) - offsets(first),
        offsets(index + 1) - offsets(first)
      )
    }.toVector

  /** The pixel extent of each grapheme cluster whose two boundaries both have a caret stop, in line order. A cluster
    * whose stops are missing has no extent of its own.
    */
  final private class ClusterBounds(
      starts: Array[Int],
      ends: Array[Int],
      startXs: Array[Float],
      endXs: Array[Float],
      count: Int
  ):

    /** The extent of the clusters overlapping `[start, end)`, `(0, 0)` when there are none, and where the next, later
      * run may begin searching. Clusters are in ascending order, so the overlapping ones are contiguous.
      */
    def extent(start: Int, end: Int, from: Int): (Float, Float, Int) =
      val first = firstEndingAfter(start, from)
      @tailrec
      def widen(index: Int, minXPx: Float, maxXPx: Float, found: Boolean): (Float, Float) =
        if index >= count || starts(index) >= end then if found then (minXPx, maxXPx) else (0.0f, 0.0f)
        else if found then widen(index + 1, minXPx.min(startXs(index)), maxXPx.max(endXs(index)), found = true)
        else widen(index + 1, startXs(index), endXs(index), found = true)
      val (minXPx, maxXPx) = widen(first, 0.0f, 0.0f, found = false)
      (minXPx, maxXPx, first)

    @tailrec
    private def firstEndingAfter(start: Int, from: Int): Int =
      if from < count && ends(from) <= start then firstEndingAfter(start, from + 1) else from

  private object ClusterBounds:

    def apply(visualLine: TextVisualLine, spans: Vector[GraphemeSpan]): ClusterBounds =
      val measured = visualLine.row
      val offset   = visualLine.xOffsetPx
      val starts   = new Array[Int](spans.length)
      val ends     = new Array[Int](spans.length)
      val startXs  = new Array[Float](spans.length)
      val endXs    = new Array[Float](spans.length)
      @tailrec
      def collect(spanIndex: Int, stopIndex: Int, count: Int): Int =
        if spanIndex >= spans.length || stopIndex + 1 >= measured.caretCount then count
        else
          val span = spans(spanIndex)
          if measured.caretColumn(stopIndex) == visualLine.startColumn + span.startLocalIndex &&
              measured.caretColumn(stopIndex + 1) == visualLine.startColumn + span.endLocalIndex
          then
            starts(count) = span.startLocalIndex
            ends(count) = span.endLocalIndex
            startXs(count) = measured.caretXs(stopIndex) + offset
            endXs(count) = measured.caretXs(stopIndex + 1) + offset
            collect(spanIndex + 1, stopIndex + 1, count + 1)
          else collect(spanIndex, stopIndex + 1, count)
      new ClusterBounds(starts, ends, startXs, endXs, collect(0, 0, 0))

/** How far a string advances in a font, measured once and remembered.
  *
  * Overlay rows size their background and place their caret from the advance of their text, which a `TextLayout`
  * measures. A modal repaints the same rows every frame, so the measurement is keyed by exactly what determines it --
  * text, font and render context -- and a repeat is a lookup. The cache holds at most `capacity` advances, because the
  * texts typed into a search or prompt box are unbounded. Measuring shapes through `shapedText`, so the painter that
  * shares it finds the text already shaped.
  */

  import MeasuredTextAdvances.*

  private val cache     = new BoundedRefCache[Key, Float](new AtomicInteger(capacity))
  private val measuredN = AtomicLong(0)

  /** How many advances were measured, as opposed to served from the cache. */
  def measurements: Long = measuredN.get()

    val key = Key(text, font, fontRenderContext)
    cache.get(key).getOrElse {
      val _ = measuredN.incrementAndGet()
      val advance =
        TextLayoutSnapshot.caretXsForText(text, font, shapedText, fontRenderContext).lastOption.getOrElse(0.0f)
      cache.put(key, advance)
      advance
    }

    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    row.segments match
      case label :: description :: shortcut :: Nil =>
        val (labelWidth, descriptionWidth, shortcutWidth) = priorityThreeColumnWidths(width)
        renderColumnCell(surface, x, y, labelWidth, label, theme, defaultForeground, defaultBackground, font)
        renderColumnCell(
          surface,
          x + labelWidth + 1,
          y,
          descriptionWidth,
          description,
          theme,
          defaultForeground,
          defaultBackground,
          font
        )
        renderColumnCell(
          surface,
          x + labelWidth + descriptionWidth + 2,
          y,
          shortcutWidth,
          shortcut,
          theme,
          defaultForeground,
          defaultBackground,
          font,
          alignRight = true
        )
      case _ =>
        renderColumnRow(surface, x, y, width, row, theme, defaultForeground, defaultBackground, font)

  def threeColumnWidths(width: Int): (Int, Int, Int) =
    val safeWidth      = math.max(0, width)
    val preferredLabel = math.min(22, math.max(8, safeWidth / 3))
    val preferredValue = math.min(18, math.max(8, safeWidth / 4))
    val (labelWidth, valueWidth) =
      if preferredLabel + preferredValue + 2 <= safeWidth then (preferredLabel, preferredValue)
      else (math.min(22, safeWidth / 3), math.min(18, safeWidth / 4))
    (labelWidth, math.max(0, safeWidth - labelWidth - valueWidth - 2), valueWidth)

  private def priorityThreeColumnWidths(width: Int): (Int, Int, Int) =
    val safeWidth      = math.max(0, width)
    val preferredLabel = math.min(36, math.max(8, (safeWidth * 3) / 5))
    val preferredValue = math.min(18, math.max(8, safeWidth / 5))
    val (labelWidth, valueWidth) =
      if preferredLabel + preferredValue + 2 <= safeWidth then (preferredLabel, preferredValue)
      else (math.min(22, safeWidth / 3), math.min(18, safeWidth / 4))
    (labelWidth, math.max(0, safeWidth - labelWidth - valueWidth - 2), valueWidth)

  private def fourColumnWidths(width: Int): (Int, Int, Int, Int) =
    val safeWidth       = math.max(0, width)
    val labelWidth      = math.min(28, math.max(8, (safeWidth * 2) / 5))
    val valueWidth      = math.min(12, math.max(0, safeWidth / 5))
    val scopeWidth      = math.min(10, math.max(0, safeWidth / 8))
    val breadcrumbWidth = math.max(0, safeWidth - labelWidth - valueWidth - scopeWidth - 3)
    (labelWidth, valueWidth, scopeWidth, breadcrumbWidth)

  private def twoColumnWidths(width: Int): (Int, Int) =
    val safeWidth      = math.max(0, width)
    val preferredLabel = math.min(22, math.max(8, safeWidth / 3))
    val labelWidth     = if preferredLabel + 1 <= safeWidth then preferredLabel else math.min(22, safeWidth / 3)
    (labelWidth, math.max(0, safeWidth - labelWidth - 1))

  private def renderColumnCell(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    segment: OverlaySegment,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font,
    alignRight: Boolean = false
  ): Unit =
    val text    = fitCellText(segment.text, width)
    val renderX = if alignRight then x + math.max(0, width - text.length) else x
    OverlaySegmentRowRenderer.renderSegmentText(
      surface,
      renderX,
      y,
      text.length,
      text,
      segment,
      theme,
      defaultForeground,
      defaultBackground,
      font = font
    )

  def fitCellText(text: String, width: Int): String =
    if width <= 0 then ""
    else if text.length <= width then text
    else if width <= 3 then text.take(width)
    else text.take(width - 3) + "..."
package com.serenity.ui.renderer

import com.serenity.ui.color.RenderColor
import com.serenity.ui.layout.*
import com.serenity.ui.theme.Theme

/** Paints one row of a floating surface: the row's background, then its glyphs in whatever arrangement its
  * [[OverlayRowLayout]] calls for, then its caret. Split out of [[TextOverlayRenderer]], which owns the surface as a
  * whole -- its frame, border, material and content resolution -- and delegates each row here. Sits alongside
  * [[OverlaySegmentRowRenderer]] and [[OverlayColumnRowRenderer]], which it dispatches to for the multi-segment row
  * shapes.
  */
private[renderer] object OverlayRowPainter:

  def renderRow(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    cursorVisible: Boolean,
    defaultForeground: Option[RenderColor],
    defaultBackground: Option[RenderColor],
    font: java.awt.Font,
    cellMetrics: CellMetrics,
    advances: MeasuredTextAdvances,
    textInsetPx: Double,
    pixelY: Option[Int] = None,
    pixelHeight: Option[Int] = None
  ): Unit =
    surface.text.withLogicalPixelRow(y, pixelY.getOrElse(cellMetrics.toPixelY(y))) {
      renderRowAt(
        surface,
        x,
        y,
        width,
        row,
        theme,
        cursorVisible,
        defaultForeground,
        defaultBackground,
        font,
        cellMetrics,
        advances,
        textInsetPx,
        pixelY,
        pixelHeight
      )
    }

  private def renderRowAt(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    cursorVisible: Boolean,
    defaultForeground: Option[RenderColor],
    defaultBackground: Option[RenderColor],
    font: java.awt.Font,
    cellMetrics: CellMetrics,
    advances: MeasuredTextAdvances,
    textInsetPx: Double,
    pixelY: Option[Int],
    pixelHeight: Option[Int]
  ): Unit =
    val rowView = scrolledRowView(row, width)
    val baseFg  = defaultForeground.getOrElse(theme.panel.foreground)
    val baseBg  = defaultBackground.getOrElse(theme.panel.background)
    val rowBackground =
      rowView.row.backgroundColor
        .map(_.withAlpha(baseBg.alpha))
        .getOrElse(
          if rowView.row.selected then theme.highlighted.background.withAlpha(baseBg.alpha)
          else baseBg
        )
    val rowForeground =
      rowView.row.foregroundColor
        .map(_.withAlpha(baseFg.alpha))
        .getOrElse(if rowView.row.selected then theme.highlighted.foreground.withAlpha(baseFg.alpha) else baseFg)
    val rowLeftXPx  = cellMetrics.toPixelX(x)
    val rowRightXPx = cellMetrics.toPixelX(x + width)

    surface.setForegroundColor(rowForeground)
    surface.setBackgroundColor(rowBackground)
    if rowView.row.selected then
      pixelHeight.foreach { height =>
        surface.pixels.fillPixelRect(
          xPx = rowLeftXPx,
          yPx = pixelY.getOrElse(cellMetrics.toPixelY(y)),
          widthPx = rowRightXPx - rowLeftXPx,
          heightPx = height,
          color = rowBackground
        )
      }
    if rowView.row.selected then surface.enableStyle(theme.focusStyle)
    CharacterRenderer.renderStringPlain(surface, x, y, " " * width)

    // Only the glyphs move. The row's background fill and its blanking pass above stay full-bleed, so a selected row
    // still spans the whole content rect and the row remains one uninterrupted click target -- the inset is about
    // where text sits inside the row, not about making the row smaller. A terminal's `withPixelTranslation` is the
    // identity, so this is structurally a no-op there rather than a rounding that happens to vanish.
    surface.pixels.withPixelTranslation(textInsetPx, 0.0) {
      renderRowGlyphs(
        surface,
        x,
        y,
        width,
        rowView,
        theme,
        cursorVisible,
        rowForeground,
        rowBackground,
        font,
        cellMetrics,
        advances,
        pixelY
      )
    }

  private def renderRowGlyphs(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    rowView: OverlayRowView,
    theme: Theme,
    cursorVisible: Boolean,
    rowForeground: RenderColor,
    rowBackground: RenderColor,
    font: java.awt.Font,
    cellMetrics: CellMetrics,
    advances: MeasuredTextAdvances,
    pixelY: Option[Int]
  ): Unit =
    val rowLeftXPx  = cellMetrics.toPixelX(x)
    val rowRightXPx = cellMetrics.toPixelX(x + width)

    rowView.row.layout match
      case OverlayRowLayout.Plain =>
        // `leadingPadding` indents a Plain row (e.g. the settings-surface group-preview rows) by shrinking its
        // available width and shifting its start column, rather than baking literal spaces into `plainText` --
        // consistent with how `renderCompactDistributedRow` already honors it for the toolbar's row shape.
        val pad           = rowView.row.leadingPadding.max(0).min(width)
        val indentedX     = x + pad
        val indentedWidth = width - pad
        if rowView.row.segments.nonEmpty then
          OverlaySegmentRowRenderer.renderInlineSegments(
            surface,
            indentedX,
            y,
            indentedWidth,
            rowView.row,
            theme,
            rowForeground,
            rowBackground,
            font
          )
        else if rowView.useMeasuredCursor && shouldUseMeasuredCursor(surface) then
          renderMeasuredPlainRow(
            surface,
            indentedX,
            pixelY.getOrElse(cellMetrics.toPixelY(y)),
            indentedWidth,
            rowView.row.plainText,
            font,
            cellMetrics,
            advances,
            rowRightXPx
          )
        else CharacterRenderer.renderStringPlain(surface, indentedX, y, rowView.row.plainText.take(indentedWidth))
      case OverlayRowLayout.Distributed =>
        OverlaySegmentRowRenderer.renderDistributedRow(
          surface,
          x,
          y,
          width,
          rowView.row,
          theme,
          rowForeground,
          rowBackground,
          font
        )
      case OverlayRowLayout.Split =>
        OverlaySegmentRowRenderer.renderSplitRow(
          surface,
          x,
          y,
          width,
          rowView.row,
          theme,
          rowForeground,
          rowBackground,
          font
        )
      case OverlayRowLayout.Columns =>
        OverlayColumnRowRenderer.renderColumnRow(
          surface,
          x,
          y,
          width,
          rowView.row,
          theme,
          rowForeground,
          rowBackground,
          font
        )
      case OverlayRowLayout.PriorityColumns =>
        OverlayColumnRowRenderer.renderPriorityColumnRow(
          surface,
          x,
          y,
          width,
          rowView.row,
          theme,
          rowForeground,
          rowBackground,
          font
        )

    if cursorVisible then
      rowView.row.cursorColumn
        .flatMap(cursorColumn => cursorPlacement(rowView.row, x, width, cursorColumn, rowView.useMeasuredCursor))
        .foreach { placement =>
          if placement.useMeasured && shouldUseMeasuredCursor(surface) then
            renderMeasuredCursor(
              surface,
              placement.x,
              pixelY.getOrElse(cellMetrics.toPixelY(y)),
              placement.textBeforeCursor,
              theme,
              font,
              cellMetrics,
              advances,
              rowLeftXPx,
              rowRightXPx
            )
          else if placement.cellColumn >= 0 && placement.cellColumn < width then
            surface.setForegroundColor(theme.background)
            surface.setBackgroundColor(theme.cursor)
            CharacterRenderer.renderChar(surface, placement.x + placement.cellColumn, y, ' ')
        }

    if rowView.row.selected then surface.disableStyle(theme.focusStyle)

  final private case class OverlayRowView(row: OverlayRow, useMeasuredCursor: Boolean)

  final private case class CursorPlacement(x: Int, textBeforeCursor: String, useMeasured: Boolean = false):
    def cellColumn: Int =
      textBeforeCursor.length

  private def cursorPlacement(
    row: OverlayRow,
    x: Int,
    width: Int,
    cursorColumn: Int,
    useMeasuredCursor: Boolean
  ): Option[CursorPlacement] =
    row.layout match
      case OverlayRowLayout.Plain =>
        Some(CursorPlacement(x, row.plainText.take(cursorColumn.max(0).min(row.plainText.length)), useMeasuredCursor))
      case OverlayRowLayout.Split =>
        splitCursorPlacement(row, x, width, cursorColumn)
      case OverlayRowLayout.Columns | OverlayRowLayout.PriorityColumns =>
        columnCursorPlacement(row, x, width)
      case OverlayRowLayout.Distributed =>
        None

  private def splitCursorPlacement(
    row: OverlayRow,
    x: Int,
    width: Int,
    cursorColumn: Int
  ): Option[CursorPlacement] =
    row.segments match
      case left :: rightSegments if rightSegments.nonEmpty =>
        val rightTexts     = rightSegments.map(_.text)
        val rightGroupText = rightTexts.mkString(" ")
        val rightStartCol  = left.text.length + 1
        val rightStartX =
          if row.cursorColumn.nonEmpty then x + rightStartCol
          else
            val rightGroupWidth = math.min(width, rightGroupText.length)
            x + math.max(0, width - rightGroupWidth)
        if cursorColumn <= left.text.length then
          Some(CursorPlacement(x, left.text.take(cursorColumn.max(0).min(left.text.length)), useMeasured = true))
        else
          val localColumn = (cursorColumn - rightStartCol).max(0).min(rightGroupText.length)
          Some(CursorPlacement(rightStartX, rightGroupText.take(localColumn), useMeasured = true))
      case _ =>
        Some(CursorPlacement(x, row.plainText.take(cursorColumn.max(0).min(row.plainText.length))))

  private def columnCursorPlacement(row: OverlayRow, x: Int, width: Int): Option[CursorPlacement] =
    row.segments match
      case _ :: _ :: value :: Nil if value.selected =>
        val (labelWidth, hintWidth, valueWidth) = OverlayColumnRowRenderer.threeColumnWidths(width)
        val valueText                           = OverlayColumnRowRenderer.fitCellText(value.text, valueWidth)
        val valueX = x + labelWidth + hintWidth + 2 + math.max(0, valueWidth - valueText.length)
        Some(CursorPlacement(valueX, valueText, useMeasured = true))
      case _ =>
        row.cursorColumn.map(cursorColumn =>
          CursorPlacement(x, row.plainText.take(cursorColumn.max(0).min(row.plainText.length)))
        )

  private def scrolledRowView(row: OverlayRow, width: Int): OverlayRowView =
    val useMeasuredCursor = row.cursorColumn.nonEmpty
    val scrollOffset =
      row.layout match
        case OverlayRowLayout.Plain | OverlayRowLayout.Split =>
          row.cursorColumn match
            case Some(cursorColumn) if row.plainText.length > width =>
              math.max(0, math.min(cursorColumn - width + 1, row.plainText.length - width))
            case _ =>
              0
        case OverlayRowLayout.Columns | OverlayRowLayout.PriorityColumns | OverlayRowLayout.Distributed =>
          0

    if scrollOffset == 0 then OverlayRowView(row, useMeasuredCursor)
    else
      val visibleText = row.plainText.slice(scrollOffset, scrollOffset + width)
      OverlayRowView(
        row.copy(
          plainText = visibleText,
          cursorColumn = row.cursorColumn.map(_ - scrollOffset).filter(_ >= 0),
          segments = Nil,
          layout = OverlayRowLayout.Plain
        ),
        useMeasuredCursor
      )

  // #1105: drawRunPx is a no-op on a surface with no FontRenderContext (a terminal), so the measured path can never be
  // taken there regardless of what the font alone would call for (ligatures, proportional advances, ...). Every real
  // (GUI) surface reports a FontRenderContext unconditionally, so this was already the de facto behaviour for GUI mode
  // -- the font-only checks previously OR'd in here never had the chance to matter on a real surface, only on a
  // cell-only one, where they were exactly the bug: they could force the measured (dropped) path even with no
  // FontRenderContext to measure against.
  private def shouldUseMeasuredCursor(surface: RenderSurface): Boolean =
    surface.text.fontRenderContext.nonEmpty

  private def renderMeasuredPlainRow(
    surface: RenderSurface,
    x: Int,
    yPx: Int,
    width: Int,
    text: String,
    font: java.awt.Font,
    cellMetrics: CellMetrics,
    advances: MeasuredTextAdvances,
    maxRightXPx: Int
  ): Unit =
    val visibleText = text.take(width)
    if visibleText.nonEmpty then
      val frc        = surface.text.fontRenderContext.getOrElse(TextLayoutSnapshot.defaultFontRenderContext())
      val advancePx  = advances.advanceOf(visibleText, font, frc)
      val textXPx    = cellMetrics.toPixelX(x).toFloat
      val maxWidthPx = math.max(1.0f, maxRightXPx.toFloat - textXPx)
      val widthPx    = advancePx.max(1.0f).min(maxWidthPx)
      surface.text.drawRunPx(textXPx, yPx, widthPx, cellMetrics.lineHeight, cellMetrics.ascent, visibleText)

  private def renderMeasuredCursor(
    surface: RenderSurface,
    x: Int,
    yPx: Int,
    textBeforeCursor: String,
    theme: Theme,
    font: java.awt.Font,
    cellMetrics: CellMetrics,
    advances: MeasuredTextAdvances,
    minXPx: Int,
    maxRightXPx: Int
  ): Unit =
    val frc          = surface.text.fontRenderContext.getOrElse(TextLayoutSnapshot.defaultFontRenderContext())
    val advancePx    = advances.advanceOf(textBeforeCursor, font, frc)
    val rawWidthPx   = math.max(2, math.round(cellMetrics.charWidth * 0.12f))
    val caretWidthPx = math.min(rawWidthPx, math.max(1, maxRightXPx - minXPx))
    val unclampedXPx = cellMetrics.toPixelX(x) + math.round(advancePx)
    val xPx          = math.max(minXPx, math.min(unclampedXPx, maxRightXPx - caretWidthPx))
    surface.pixels.fillPixelRect(xPx, yPx, caretWidthPx, cellMetrics.lineHeight, theme.cursor)
package com.serenity.ui.renderer

import java.awt.Font

/** Paints the segment-oriented [[OverlayRowLayout]] variants -- `Distributed`, `Split`, and `Plain`'s inline-segment
  * path -- extracted from [[TextOverlayRenderer]] alongside [[OverlayColumnRowRenderer]] to keep each row-layout family
  * in one place.
  */
object OverlaySegmentRowRenderer:

  /** The foreground a tone paints in, or `None` for the surface's own default. */
  def toneForeground(tone: OverlayTone, theme: Theme): Option[RenderColor] =
    tone match
      case OverlayTone.Normal => None
      case OverlayTone.Muted  => Some(theme.muted)
      case OverlayTone.Error  => Some(theme.error.foreground)
      case OverlayTone.Accent => Some(theme.accent)

  /** The background a tone paints in, or `None` for the surface's own default. */
  def toneBackground(tone: OverlayTone, theme: Theme): Option[RenderColor] =
    tone match
      case OverlayTone.Error                                           => Some(theme.error.background)
      case OverlayTone.Normal | OverlayTone.Muted | OverlayTone.Accent => None

  def renderDistributedRow(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    val segments = row.segments
    if segments.isEmpty then CharacterRenderer.renderStringPlain(surface, x, y, row.plainText.take(width))
    else if segments.exists(_.allocatedWidth.nonEmpty) then
      renderCompactDistributedRow(surface, x, y, width, row, theme, defaultForeground, defaultBackground, font)
    else
      val baseCellWidth = width / segments.length
      val remainder     = width % segments.length

      val _ = segments.zipWithIndex.foldLeft(x) {
        case (cursorX, (segment, index)) =>
          val cellWidth = baseCellWidth + (if index < remainder then 1 else 0)
          renderSegmentCell(
            surface,
            cursorX,
            y,
            cellWidth,
            segment,
            theme,
            defaultForeground,
            defaultBackground,
            font
          )
          cursorX + cellWidth
      }
      ()

  def renderCompactDistributedRow(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    val segments = row.segments
    if segments.isEmpty then CharacterRenderer.renderStringPlain(surface, x, y, row.plainText.take(width))
    else
      val startX = x + row.leadingPadding.max(0).min(width)
      val _ = segments.zipWithIndex.foldLeft(startX) {
        case (cursorX, (segment, index)) =>
          val remainingWidth = (x + width - cursorX).max(0)
          val cellWidth      = segment.allocatedWidth.getOrElse(0).min(remainingWidth)
          renderSegmentCell(
            surface,
            cursorX,
            y,
            cellWidth,
            segment,
            theme,
            defaultForeground,
            defaultBackground,
            font
          )
          val afterCell = cursorX + cellWidth
          val afterSeparator =
            if segment.trailingSeparator && afterCell < x + width then
              surface.setForegroundColor(defaultForeground)
              surface.setBackgroundColor(defaultBackground)
              CharacterRenderer.renderChar(surface, afterCell, y, '│')
              afterCell + 1
            else afterCell
          if index < segments.length - 1 && afterSeparator < x + width then afterSeparator + 1
          else afterSeparator
      }
      ()

  def renderSplitRow(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    row.segments match
      case left :: rightSegments if rightSegments.nonEmpty =>
        if row.cursorColumn.nonEmpty then
          renderEditableSplitRow(
            surface,
            x,
            y,
            width,
            left,
            rightSegments,
            theme,
            defaultForeground,
            defaultBackground,
            font
          )
        else
          val rightTexts      = rightSegments.map(_.text)
          val rightGroupText  = rightTexts.mkString(" ")
          val rightGroupWidth = math.min(width, rightGroupText.length)
          val leftMaxWidth    = math.max(0, width - rightGroupWidth - 1)
          val leftText        = left.text.take(leftMaxWidth)

          renderSegmentText(
            surface,
            x,
            y,
            leftText.length,
            leftText,
            left,
            theme,
            defaultForeground,
            defaultBackground,
            font
          )

          val rightStartX = x + math.max(0, width - rightGroupWidth)
          val _ = rightSegments.foldLeft(rightStartX) { (cursorX, segment) =>
            val text = segment.text.take(math.max(0, x + width - cursorX))
            renderSegmentText(
              surface,
              cursorX,
              y,
              text.length,
              text,
              segment,
              theme,
              defaultForeground,
              defaultBackground,
              font
            )
            cursorX + text.length + 1
          }
      case _ =>
        CharacterRenderer.renderStringPlain(surface, x, y, row.plainText.take(width))

  def renderEditableSplitRow(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    left: OverlaySegment,
    rightSegments: List[OverlaySegment],
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    val leftText = left.text.take(width)
    renderSegmentText(
      surface,
      x,
      y,
      leftText.length,
      leftText,
      left,
      theme,
      defaultForeground,
      defaultBackground,
      font
    )

    val firstRightX = x + leftText.length + 1
    val _ = rightSegments.foldLeft(firstRightX) { (cursorX, segment) =>
      val remainingWidth = math.max(0, x + width - cursorX)
      val text           = segment.text.take(remainingWidth)
      renderSegmentText(
        surface,
        cursorX,
        y,
        text.length,
        text,
        segment,
        theme,
        defaultForeground,
        defaultBackground,
        font
      )
      cursorX + text.length + 1
    }

  def renderInlineSegments(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    row: OverlayRow,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    val rightEdge = x + width
    val _ = row.segments.foldLeft(x) { (cursorX, segment) =>
      val remainingWidth = math.max(0, rightEdge - cursorX)
      val text           = segment.text.take(remainingWidth)
      renderSegmentText(
        surface,
        cursorX,
        y,
        text.length,
        text,
        segment,
        theme,
        defaultForeground,
        defaultBackground,
        font
      )
      val nextX = cursorX + text.length
      if nextX < rightEdge then nextX + 1 else nextX
    }

  def renderSegmentCell(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,
    segment: OverlaySegment,
    theme: Theme,
    defaultForeground: RenderColor,
    defaultBackground: RenderColor,
    font: Font
  ): Unit =
    val iconWidth   = segment.inlineIcon.map(_.length).getOrElse(0).min(width)
    val iconGap     = if iconWidth > 0 && width > iconWidth && segment.text.nonEmpty then 1 else 0
    val text        = segment.text.take(math.max(0, width - iconWidth - iconGap))
    val renderWidth = iconWidth + iconGap + text.length
    val leftPad     = math.max(0, (width - renderWidth) / 2)
    val renderX     = x + leftPad
    renderSegmentText(
      surface,
      renderX,
      y,
      renderWidth,
      text,
      segment,
      theme,
      defaultForeground,
      defaultBackground,
      font
    )

  def renderSegmentText(
    surface: RenderSurface,
    x: Int,
    y: Int,
    width: Int,

        segment.backgroundColor

          .getOrElse(
            if segment.selected then theme.highlighted.background.withAlpha(defaultBackground.alpha)
            else toneBackground(segment.tone, theme).fold(defaultBackground)(_.withAlpha(defaultBackground.alpha))
          )
      val segmentForeground =
        segment.foregroundColor
          .map(_.withAlpha(defaultForeground.alpha))
          .getOrElse(
            if segment.selected then theme.highlighted.foreground.withAlpha(defaultForeground.alpha)
            else toneForeground(segment.tone, theme).fold(defaultForeground)(_.withAlpha(defaultForeground.alpha))
          )

      surface.setBackgroundColor(segmentBackground)
      val inlineIcon = segment.inlineIcon.filter(_ => width > 0)
      inlineIcon.foreach { icon =>
        segment.inlineIconFontFamily.foreach(family =>

        )
        CharacterRenderer.renderStringPlain(surface, x, y, icon.take(width))
        if segment.inlineIconFontFamily.nonEmpty then surface.text.setFont(FontSpec.fromAwt(font))
      }
      val iconWidth = inlineIcon.map(_.length.min(width)).getOrElse(0)
      val iconGap   = if iconWidth > 0 && width > iconWidth && segmentText.nonEmpty then 1 else 0
      segment.fontFamily.foreach(family =>
        surface.text.setFont(FontSpec.fromAwt(Font(family, font.getStyle, font.getSize).deriveFont(font.getSize2D)))

      CharacterRenderer.renderStringPlain(
        surface,
        x + iconWidth + iconGap,

      )

/** A floating surface's resolved paint plan. `composition` is the surface's *only* stored content representation (issue
  * #1683) -- there is no second, independently-settable `rows`/`header`/`footer`/`keyHintRow` that could disagree with
  * it and need a precedence rule to resolve, the way this type used to carry both (`TextOverlayRenderer` used to pick
  * `composition` over `rows` whenever it was set).
  *
  * The companion's `apply` still accepts `header`/`rows`/`footer`/`keyHintRow` for construction: a plain informational
  * surface with no bespoke `*SurfaceComposition` of its own is built from them, via [[RowsSurfaceComposition]], rather
  * than every caller having to build a `ResolvedSurfaceComposition` by hand.
  */
final case class TextOverlayView private (
    rect: LayoutRect,
    contentRect: Option[LayoutRect],
    borderCells: Int,
    alphaMultiplier: Float,
    title: Option[String],
    itemGapRows: Double,
    itemTargetRows: Int,
    verticalOffsetRows: Double,
    surfaceId: Option[SurfaceId],
    composition: Option[ResolvedSurfaceComposition]
):

  def resolvedContentRect: LayoutRect =
    contentRect.getOrElse(com.serenity.ui.layout.SurfaceFrameLayout(rect, borderCells).contentRect)

  def contentRowSlots: List[SurfaceContentRowSlot] =
    composition.fold(List.empty[SurfaceContentRowSlot])(RowsSurfaceComposition.contentRowSlots)

object TextOverlayView:

  /** `composition`, when given, wins outright -- there is no dual-path precedence rule to apply, only a choice of which
    * single composition to store: a caller building an overlay for content with no bespoke `*SurfaceComposition` passes
    * `header`/`rows`/`footer`/`keyHintRow` instead, and this builds the generic one via [[RowsSurfaceComposition]].
    */
  def apply(
    rect: LayoutRect,
    contentRect: Option[LayoutRect] = None,
    borderCells: Int = 1,
    alphaMultiplier: Float = 1.0f,
    title: Option[String] = None,
    header: Option[OverlayRow] = None,
    rows: List[OverlayRow] = Nil,
    footer: Option[OverlayRow] = None,
    keyHintRow: Option[OverlayRow] = None,
    itemGapRows: Double = 0.0,
    itemTargetRows: Int = 1,
    verticalOffsetRows: Double = 0.0,
    surfaceId: Option[SurfaceId] = None,
    composition: Option[ResolvedSurfaceComposition] = None
  ): TextOverlayView =
    val resolvedComposition = composition.orElse(
      Option.when(header.nonEmpty || rows.nonEmpty || footer.nonEmpty || keyHintRow.nonEmpty)(
        RowsSurfaceComposition.forResolved(
          ResolvedSurfaceContent(header = header, rows = rows, footer = footer, keyHintRow = keyHintRow),
          frameRect = rect,
          borderCells = borderCells,
          itemGapRows = itemGapRows,
          itemTargetRows = itemTargetRows,
          contentRectOverride = contentRect
        )
      )
    )
    new TextOverlayView(
      rect,
      contentRect,
      borderCells,
      alphaMultiplier,
      title,
      itemGapRows,
      itemTargetRows,
      verticalOffsetRows,
      surfaceId,
      resolvedComposition
    )

object OverlayViewModel:
  private val InactiveFloatingPanelAlphaMultiplier = 0.68f

  def fromState(state: AppState, layout: CalculatedLayout): OverlayViews =

  private def fromState(
    state: AppState,
    layout: CalculatedLayout,
    scene: Option[UiSceneSnapshot]
  ): OverlayViews =
    val aboveCursorStack = stackViews(aboveCursorSurfaceIds(state, layout), state, layout, scene)
    val aboveCursor      = aboveCursorStack.headOption

    val belowCursorStack = stackViews(layout.belowCursorOverlayStack.map(_._1), state, layout, scene)
    val belowCursor      = belowCursorStack.headOption
    val cornerStack      = stackViews(layout.cornerOverlayStack.map(_._1), state, layout, scene)
    val modal = scene.toList.flatMap(_.modal).flatMap {
      case node @ SceneNode(SceneNodeId.Surface(surfaceId), _, _, _, _, _) =>

          case None          =>
            // A blocking ModalDialog (#814) -- not a UiSurface, but shares the SceneNodeId.Surface id scheme.
            state.runtime.modalStack.find(_.id == surfaceId).flatMap(buildModalView(_, state, node.frameRect))
      case _ => None
    }

    val tabBar = state.tabBarSurface.flatMap(surface =>
      buildView(surface, state, layout.tabBarRect, collapsed = false, verticalOffsetRows = 0.0)
    )

    OverlayViews(
      aboveCursor = aboveCursor,
      aboveCursorStack = aboveCursorStack,
      belowCursor = belowCursor,
      belowCursorStack = belowCursorStack,
      cornerStack = cornerStack,
      modal = modal,
      tabBar = tabBar

  private def aboveCursorSurfaceIds(state: AppState, layout: CalculatedLayout): List[SurfaceId] =

  private def contentView(

    state: AppState,
    rect: LayoutRect,
    collapsed: Boolean = false

          // below, which it always is for these, so resolving real rows here would be dead computation on this
          // specific call site. Scoped to just this call site, not `SurfaceContentResolver.resolve`'s own dispatch:

              SurfaceContent.Diagnostics(_, _, _) | SurfaceContent.DirectoryTree(_, _, _) |
              SurfaceContent.Comments(_, _, _) =>

import cats.effect.{IO, Resource}

/** A dedicated single-thread execution context for CPU-bound Swing/Java2D paint work.
  *
  * IO.blocking hands work to Cats Effect's unbounded, cached blocking thread pool -- meant for calls that actually
  * block (file/socket I/O), not synchronous CPU-bound painting. Routing that work onto IO's default compute pool
  * instead would be worse on a low-core machine: that pool is small (sized to availableProcessors) and shared by every
  * other fiber in the app, so any multi-millisecond paint call would stall unrelated work. A dedicated single thread
  * keeps paint work off both pools, and as a side effect gives every AWT/Swing call in the render path a single
  * consistent thread instead of whichever blocking-pool thread happened to be free.
  */
object PaintExecutionContext:

  def resource: Resource[IO, ExecutionContext] =
    Resource
      .make(IO(Executors.newSingleThreadExecutor { runnable =>
        // Daemon so this CPU-bound paint worker can never keep the JVM alive once the app has otherwise shut down
        // (#1543): a lingering non-daemon thread is one way a GUI process outlives its window.
        val thread = new Thread(runnable, "serenity-paint")
        thread.setDaemon(true)
        thread
      }))(executor => IO(executor.shutdown()))
      .map(ExecutionContext.fromExecutorService)

/** A panel painted straight into a persisting surface, with no layer buffer to repaint it from: where it sat and what
  * it showed. One is kept for each of a surface's contents, since a surface alternating between two images holds the
  * panel as it was on each image's last frame.
  */
final case class PaintedPanel(
    viewportWidth: Int,
    viewportHeight: Int,
    cursorVisible: Boolean,
    frameRect: LayoutRect,
    bounds: PixelRect,
    content: Any
):

  def matches(
    viewportWidth: Int,
    viewportHeight: Int,
    cursorVisible: Boolean,
    frameRect: LayoutRect,
    layerBounds: PixelRect
  ): Boolean =
    this.viewportWidth == viewportWidth && this.viewportHeight == viewportHeight &&
      this.cursorVisible == cursorVisible && this.frameRect == frameRect && bounds == layerBounds

/** The panels a frame painted without a layer buffer, per surface owner (as [[RendererFrameState]] keys its cached
  * layers, for the same reason) and per the surface contents each was painted into.
  */
final private[renderer] class PaintedPanelRecords(cacheCapacity: AtomicInteger):

  private type Records = Map[(SurfaceId, SurfaceContentIdentity), PaintedPanel]

  private val records = new BoundedRefCache[ScreenIdentity, Records](cacheCapacity)

  def of(surface: RenderSurface): Records = records.get(surface.layerCacheOwner).getOrElse(Map.empty)

  def remember(surface: RenderSurface, surfaceId: SurfaceId, key: SurfaceContentIdentity, panel: PaintedPanel): Unit =
    val owner = surface.layerCacheOwner
    records.replaceAll(tracked =>
      tracked.updated(owner, tracked.getOrElse(owner, Map.empty).updated(surfaceId -> key, panel))
    )

  /** Drops the records of panels no longer on screen. */
  def prune(surface: RenderSurface, activeIds: Set[SurfaceId]): Unit =
    val owner = surface.layerCacheOwner
    records.replaceAll { tracked =>
      tracked
        .get(owner)
        .fold(tracked)(current =>
          tracked.updated(owner, current.filter { case ((id, _), _) => activeIds.contains(id) })
        )
    }

  /** `key`'s pixels no longer hold what these panels painted, because they were cleared or are about to be repainted.
    */
  def forget(key: SurfaceContentIdentity): Unit =
    records.replaceAll(tracked =>
      tracked.map((owner, panels) => owner -> panels.filterNot { case ((_, held), _) => held == key })
    )

/** What a [[Java2DRenderSurface]] knows about the pixels it has painted since it last cleared the viewport, so a run
  * can leave out a background fill that would repeat the clear.
  *
  * Everything is in the logical pixels the clears and runs are expressed in. Anything painted under a translation is
  * moved back to those before it is compared, and anything the tracking cannot place -- a pixel copy, a layer composite
  * -- forgets the clear altogether, which only ever costs a redundant fill, never a missing one.
  *
  * Every draw reports here and every run asks here, many times a frame, so the calls take plain ints and allocate
  * nothing. A surface is painted by one thread at a time; the edges of the last run and of everything else painted are
  * contained scratch arrays for that reason.
  */
final private[ui] class PaintedPixels:
  import PaintedPixels.*

  private val clean       = AtomicReference[Option[CleanBackground]](None)
  private val translation = AtomicReference(Translation.Zero)
  private val translucent = AtomicBoolean(false)
  private val lastRun     = new Array[Int](4)
  private val touchedBox  = new Array[Int](4)

  def setTranslucent(isTranslucent: Boolean): Unit = translucent.set(isTranslucent)

  def withTranslation(xPx: Double, yPx: Double)(render: => Unit): Unit =
    val outer = translation.get()
    translation.set(Translation(outer.xPx + xPx, outer.yPx + yPx))
    try render
    finally translation.set(outer)

  /** The viewport was cleared to `color` everywhere in `cleared`. A clear that was itself moved, clipped or blended is
    * not a clean slate.
    */
  def cleared(color: Color, cleared: scala.collection.immutable.List[PixelRect], clipped: Boolean): Unit =
    val everythingCleared = translation.get().isZero && !translucent.get() && !clipped
    clean.set(if everythingCleared then CleanBackground.cleared(color, cleared) else None)
    reset(lastRun)
    reset(touchedBox)

  /** Something other than a run painted `xPx, yPx, widthPx, heightPx` (in the current translation). */
  def touched(xPx: Int, yPx: Int, widthPx: Int, heightPx: Int): Unit =
    val moved  = translation.get()
    val left   = moved.left(xPx)
    val top    = moved.top(yPx)
    val right  = moved.right(xPx + widthPx)
    val bottom = moved.bottom(yPx + heightPx)
    if touchedBox(0) >= touchedBox(2) then store(touchedBox, left, top, right, bottom)
    else
      store(
        touchedBox,
        math.min(touchedBox(0), left),
        math.min(touchedBox(1), top),
        math.max(touchedBox(2), right),
        math.max(touchedBox(3), bottom)
      )

  def forgetClear(): Unit = clean.set(None)

  /** A run's fill over `left, top, right, bottom` (in the current translation) is redundant only over pixels still
    * holding the cleared colour: no other draw, nor the run painted just before (whose edge pixels a fractional
    * position can share), has reached them, and nothing translucent is in force to make a second fill differ from the
    * first.
    */
  def needsBackground(left: Int, top: Int, right: Int, bottom: Int, background: Color): Boolean =
    val moved = translation.get()
    val l     = moved.left(left)
    val t     = moved.top(top)
    val r     = moved.right(right)
    val b     = moved.bottom(bottom)
    translucent.get() || overlaps(lastRun, l, t, r, b) || overlaps(touchedBox, l, t, r, b) ||
    (clean.get() match
      case Some(untouched) => !untouched.covers(l, t, r, b, background)
      case None            => true)

  def ranOver(left: Int, top: Int, right: Int, bottom: Int): Unit =
    val moved = translation.get()
    store(lastRun, moved.left(left), moved.top(top), moved.right(right), moved.bottom(bottom))

  private def reset(edges: Array[Int]): Unit = store(edges, 0, 0, 0, 0)

  private def store(edges: Array[Int], left: Int, top: Int, right: Int, bottom: Int): Unit =
    edges(0) = left
    edges(1) = top
    edges(2) = right
    edges(3) = bottom

  private def overlaps(edges: Array[Int], left: Int, top: Int, right: Int, bottom: Int): Boolean =
    edges(0) < right && left < edges(2) && edges(1) < bottom && top < edges(3)

object PaintedPixels:

  /** How far the graphics has been moved from the origin the clears and runs are expressed against. */
  final private case class Translation(xPx: Double, yPx: Double):
    def isZero: Boolean = xPx == 0.0 && yPx == 0.0

    def left(edge: Int): Int   = if isZero then edge else math.floor(edge + xPx).toInt
    def top(edge: Int): Int    = if isZero then edge else math.floor(edge + yPx).toInt
    def right(edge: Int): Int  = if isZero then edge else math.ceil(edge + xPx).toInt
    def bottom(edge: Int): Int = if isZero then edge else math.ceil(edge + yPx).toInt

  private object Translation:
    val Zero: Translation = Translation(0.0, 0.0)

/** Paints each pinned/expanded/floating panel through a layer buffer cut to the panel's own bounds (#1100 stage 3,
  * #1869), and keeps those buffers in step with the panels on screen.
  */
private[renderer] object PanelLayers:

  /** Layer buffers exist only to be cached; with caching off, painting straight onto the frame is the same picture
    * without allocating a layer image per panel per frame (#1798).
    */
  def supportFor(state: AppState, context: RenderContext): Option[LayerBufferSupport] =
    context.surface.layerBuffers.filter(_ => state.persisted.config.surfaceConfig.layerCachingEnabled)

  /** Paint one panel's own content into an isolated layer buffer, reusing its last-painted pixels instead of repainting
    * when [[RendererFramePlanner.panelDirtyCheck]] says it's safe to -- the panel generalisation of the modal layer.
    *
    * The layer starts transparent, so everything the panel didn't paint stays transparent and compositing it back over
    * a frame whose content has since changed is still correct. `kept` says the frame already holds the panel's pixels
    * ([[PanelReuse.kept]] decided that when it planned), so there is nothing to paint or composite.
    *
    * No `layerBuffers` capability (TUI's `TerminalRenderSurface`), or layer caching switched off -> falls straight
    * through to `paintPanel`, painting directly into the shared surface every frame.
    */
  def paint(
    state: AppState,
    context: RenderContext,
    site: PanelSite,
    isDirty: Boolean,
    kept: Boolean
  )(paintPanel: RenderContext => Unit): Unit =
    supportFor(state, context) match
      case None => paintDirectly(context, site, kept)(paintPanel)
      case Some(support) =>
        val surface = context.surface
        val cached  = context.caches.frameState.cachedPanelLayersFor(surface).get(site.id)
        val reusable = !isDirty && cached.exists(
          _.matches(surface.viewportWidth, surface.viewportHeight, context.cursorVisible, site.rect, site.bounds)
        )
        val key = surface.persistentContentKey
        if reusable && kept then ()
        else if reusable then
          cached.foreach(c => surface.pixels.compositeLayer(c.image, site.bounds))
          key.foreach(context.caches.frameState.panelCompositedOnto(surface, site.id, _))
        else
          val capture      = LayerCapture()
          val layerSurface = support.newBoundedLayerSurface(site.bounds, capture.onFlush, cached.map(_.image))
          paintPanel(context.copy(surface = layerSurface))
          layerSurface.flush()
          capture.image.foreach { image =>
            val newlyCached = CachedPanelLayer(
              image,
              surface.viewportWidth,
              surface.viewportHeight,
              context.cursorVisible,
              site.rect,
              Some(site.bounds),
              key.toSet
            )
            context.caches.frameState.rememberPanelLayer(surface, site.id, newlyCached)
            surface.pixels.compositeLayer(image, site.bounds)
          }

  /** Whether a frame can leave a panel as its surface already holds it: through a cached layer, or, without layer
    * buffers, because the surface keeps its own pixels from frame to frame.
    */
  def canKeep(state: AppState, context: RenderContext): Boolean =
    supportFor(state, context).isDefined ||
      (context.surface.addressesPixels && context.surface.persistentContentKey.isDefined)

  /** What tells [[PanelReuse.kept]] that the pixels `key` holds are this panel as the frame would paint it. */
  def holds(
    state: AppState,
    context: RenderContext,
    key: SurfaceContentIdentity,
    frame: PanelReuse.Frame
  ): PanelSite => Boolean =
    val frameState = context.caches.frameState
    if supportFor(state, context).isDefined then
      val cached = frameState.cachedPanelLayersFor(context.surface)
      site =>
        cached.get(site.id).exists { layer =>
          layer.compositedOnto.contains(key) &&
          layer.matches(frame.viewportWidth, frame.viewportHeight, frame.cursorVisible, site.rect, site.bounds)
        }
    else
      val painted = frameState.paintedPanels.of(context.surface)
      site =>
        painted.get(site.id -> key).exists { panel =>
          panel.content == site.content &&
          panel.matches(frame.viewportWidth, frame.viewportHeight, frame.cursorVisible, site.rect, site.bounds)
        }

  /** A panel without a layer buffer is painted into the frame, unless the frame keeps what its surface holds. What was
    * painted is recorded against the pixels it went into, with the content it showed: a surface that alternates between
    * two images has a record for each, and an image keeps the panel only while the content it holds is the content now.
    */
  private def paintDirectly(context: RenderContext, site: PanelSite, kept: Boolean)(
    paintPanel: RenderContext => Unit
  ): Unit =
    if !kept then
      paintPanel(context)
      val surface = context.surface
      surface.persistentContentKey.foreach { key =>
        context.caches.frameState.paintedPanels.remember(
          surface,
          site.id,
          key,
          PaintedPanel(
            surface.viewportWidth,
            surface.viewportHeight,
            context.cursorVisible,
            site.rect,
            site.bounds,
            site.content
          )
        )
      }

  /** Drop cached panel buffers for surfaces no longer on screen this frame, scoped to `surface`'s own entry -- a
    * dismissed panel's cache would otherwise sit in that inner `Map[SurfaceId, _]` forever ([[SurfaceId]] is a plain
    * value, not an object [[RendererFrameState]] can bound the lifetime of any other way).
    */
  def forgetStale(state: AppState, panels: FramePanels, surface: RenderSurface, caches: RenderCaches): Unit =
    if state.persisted.config.surfaceConfig.layerCachingEnabled then
      caches.frameState.pruneStalePanelLayers(surface, panels.activeIds(state))
    else
      caches.frameState.forgetLayerBuffers(surface)
      caches.frameState.paintedPanels.prune(surface, panels.activeIds(state))

object PinnedPanelRenderer:

  def render(
    surface: RenderSurface,
    panel: TextPanelView,
    theme: Theme,
    config: AppConfig,
    cellMetrics: CellMetrics
  ): Unit =
    val rect            = panel.rect
    val panelBackground = theme.panel.background

    surface.setForegroundColor(theme.panel.foreground)
    surface.setBackgroundColor(panelBackground)
    for y <- rect.y until rect.bottom do surface.putString(rect.x, y, " " * rect.width)

    val textInsetPx = SurfaceTextInset.px(config)
    drawBorder(surface, panel, theme, config)
    drawTitle(surface, panel, theme, textInsetPx)
    drawComposition(surface, panel.composition, theme, panelBackground, cellMetrics, textInsetPx)

    surface.setForegroundColor(theme.foreground)
    surface.setBackgroundColor(theme.background)

  private def drawBorder(
    surface: RenderSurface,
    panel: TextPanelView,
    theme: Theme,
    config: AppConfig
  ): Unit =
    val rect = panel.rect
    if rect.width >= 2 && rect.height >= 2 then
      surface.panelOutlines.foreach(
        _.strokeRect(
          rect.x,
          rect.y,
          rect.width,
          rect.height,
          theme.border,
          config.scaledUiOutlineThicknessPx
        )
      )

  private def drawTitle(
    surface: RenderSurface,
    panel: TextPanelView,
    theme: Theme,
    textInsetPx: Double
  ): Unit =
    val titleRect = panel.titleRect
    // The title sits on the frame's top edge, so it is not padded out: padding would blank the rest of that edge.
    val title = panel.title.take(titleRect.width)
    if titleRect.width > 0 then
      surface.pixels.withPixelTranslation(textInsetPx, 0.0) {
        renderText(surface, titleRect.x, titleRect.y, title, theme.panel.foreground)
      }

  /** Paints a panel's composed plan -- box rects are already whole-cell granular (built from the same integer
    * content-rect coordinates `contentRowSlotsFor` uses), so no font-metric or sub-cell mapping is needed, unlike
    * `TextOverlayRenderer.drawComposition`'s pixel-measured floating layout.
    */
  private def drawComposition(
    surface: RenderSurface,
    composition: ResolvedSurfaceComposition,
    theme: Theme,
    panelBackground: RenderColor,
    cellMetrics: CellMetrics,
    textInsetPx: Double
  ): Unit =
    composition.paintBoxes.foreach { box =>
      box.text.foreach { text =>
        val x      = math.round(box.rect.x).toInt
        val y      = math.round(box.rect.y).toInt
        val width  = math.round(box.rect.width).toInt
        val padded = text.take(width).padTo(width, ' ')
        val (foreground, background) =
          if box.selected then (theme.highlighted.foreground, theme.highlighted.background)
          else
            (
              OverlaySegmentRowRenderer.toneForeground(box.tone, theme).getOrElse(theme.panel.foreground),
              OverlaySegmentRowRenderer.toneBackground(box.tone, theme).getOrElse(panelBackground)
            )
        surface.setForegroundColor(foreground)
        surface.setBackgroundColor(background)
        if box.selected then
          surface.enableStyle(theme.focusStyle)
          fillRowBackground(surface, cellMetrics, x, y, width, background)
        surface.pixels.withPixelTranslation(textInsetPx, 0.0) {
          renderText(surface, x, y, padded, foreground)
        }
        if box.selected then surface.disableStyle(theme.focusStyle)
      }
    }

  /** Fills one row's full content width with `color`, at its true (un-inset) position -- called only for a selected
    * row/box, ahead of its glyphs drawing shifted by `textInsetPx`. Every other row already gets this from `render`'s
    * own full-panel blank, which already paints the unselected panel background; a selected row needs its own fill
    * because that blank predates knowing which row is selected.
    */
  private def fillRowBackground(
    surface: RenderSurface,
    cellMetrics: CellMetrics,
    x: Int,
    y: Int,
    width: Int,
    color: RenderColor
  ): Unit =
    val leftPx  = cellMetrics.toPixelX(x)
    val rightPx = cellMetrics.toPixelX(x + width)
    surface.pixels.fillPixelRect(
      leftPx,
      cellMetrics.toPixelY(y),
      rightPx - leftPx,
      cellMetrics.lineHeight,
      color
    )

  private def renderText(surface: RenderSurface, x: Int, y: Int, text: String, foreground: RenderColor): Unit =
    surface.setForegroundColor(foreground)
    text.zipWithIndex.foreach((char, index) => CharacterRenderer.renderChar(surface, x + index, y, char))

/** A docked panel's resolved paint plan. `composition` is the panel's *only* stored content representation (issue
  * #1683) -- there is no second, independently-settable `rows`/`header`/`footer` that could disagree with it and need a
  * precedence rule, the way this type used to carry both. `rows`/`header`/`footer` below are read-only views derived
  * from `composition.paintBoxes`, kept only so existing call sites (and `PinnedPanelViewModel.resolve`'s own tests) can
  * still ask "what text did this panel resolve to" without reaching into paint-box internals.
  *
  * The companion's `apply` also still accepts `rows`/`header`/`footer` (as `TextPanelRow`s) for construction: a plain
  * informational panel with no bespoke `*SurfaceComposition` of its own is built from them, via
  * [[RowsSurfaceComposition]], rather than every caller having to build a `ResolvedSurfaceComposition` by hand.
  */
final case class TextPanelView private[renderer] (
    rect: LayoutRect,
    contentRect: Option[LayoutRect],
    title: String,
    surfaceId: Option[SurfaceId],
    composition: ResolvedSurfaceComposition
):
  def rows: List[TextPanelRow] =
    composition.paintBoxes.collect { case box if box.kind == SurfacePaintKind.Text => toPanelRow(box) }

  def header: Option[TextPanelRow] =
    composition.paintBoxes.collectFirst { case box if box.kind == SurfacePaintKind.Heading => toPanelRow(box) }

  def footer: Option[TextPanelRow] =
    composition.paintBoxes.collectFirst { case box if box.kind == SurfacePaintKind.Footer => toPanelRow(box) }

  def lines: List[String] = (header.toList ++ rows ++ footer.toList).map(_.plainText)

  def contentRowSlots: List[SurfaceContentRowSlot] = RowsSurfaceComposition.contentRowSlots(composition)

  def resolvedContentRect: LayoutRect =
    contentRect.getOrElse(SurfaceFrameLayout(rect).contentRect)

  def titleRect: LayoutRect =
    val content = resolvedContentRect
    LayoutRect(content.x, rect.y, content.width, 1)

  private def toPanelRow(box: SurfacePaintBox): TextPanelRow =
    TextPanelRow(plainText = box.text.getOrElse(""), selected = box.selected)

object TextPanelView:

  /** `composition`, when given, wins outright -- there is no dual-path precedence rule to apply, only a choice of which
    * single composition to store: a caller building a panel for content with no bespoke `*SurfaceComposition` passes
    * `rows`/`header`/`footer` instead, and this builds the generic one via [[RowsSurfaceComposition]].
    */
  def apply(
    rect: LayoutRect,
    contentRect: Option[LayoutRect] = None,
    title: String,
    rows: List[TextPanelRow] = Nil,
    header: Option[TextPanelRow] = None,
    footer: Option[TextPanelRow] = None,
    surfaceId: Option[SurfaceId] = None,
    composition: Option[ResolvedSurfaceComposition] = None
  ): TextPanelView =
    val resolvedComposition = composition.getOrElse(
      RowsSurfaceComposition.forResolved(
        ResolvedSurfaceContent(
          header = header.map(fromPanelRow),
          rows = rows.map(fromPanelRow),
          footer = footer.map(fromPanelRow)
        ),
        frameRect = rect,
        contentRectOverride = contentRect
      )
    )
    new TextPanelView(rect, contentRect, title, surfaceId, resolvedComposition)

  private def fromPanelRow(row: TextPanelRow): OverlayRow =
    OverlayRow(plainText = row.plainText, selected = row.selected)

object PinnedPanelViewModel:

  /** Every docked surface's panel view, including the currently-expanded one -- `state.pinnedSurfaces` (issue #817)
    * already covers it: expansion is a `Layout.maximizedWorkspaceNodeId` overlay on an otherwise still-docked surface,
    * not a separate presentation that would exclude it, so no second, expanded-only surface list is needed here.
    */
  def fromState(state: AppState, layout: CalculatedLayout): List[TextPanelView] =
    state.pinnedSurfaces.flatMap { surface =>
      EditorLayoutContract
        .panelRectFor(surface, state, layout)
        .map(rect => resolve(surface, rect, Some(state), MarkdownPreviewCache()))
    }

  def resolve(
    surface: UiSurface,
    rect: LayoutRect,
    cache: MarkdownPreviewCache = MarkdownPreviewCache()
  ): TextPanelView =
    resolve(surface, rect, None, cache)

  def resolve(
    surface: UiSurface,
    rect: LayoutRect,
    state: AppState,
    cache: MarkdownPreviewCache
  ): TextPanelView =
    resolve(surface, rect, Some(state), cache)

  private def resolve(
    surface: UiSurface,
    rect: LayoutRect,
    state: Option[AppState],
    cache: MarkdownPreviewCache
  ): TextPanelView =
    val resolved =
      surface.content match
        case SurfaceContent.MarkdownPreview(bufferId, title) =>
          val content = state.flatMap(_.persisted.buffers.get(bufferId)).map(_.document.content)
          SurfaceContentResolver.resolveBufferMarkdownPreview(title, content, rect, SurfaceRenderMode.Pinned, cache)
        case SurfaceContent.Outline(symbols, activeLocation, scroll) =>
          SurfaceContentResolver.resolve(
            SurfaceContent.Outline(symbols, activeSymbolLocation(symbols, activeLocation, state), scroll),
            rect,
            SurfaceRenderMode.Pinned
          )
        case SurfaceContent.Comments(symbols, activeLocation, scroll) =>
          SurfaceContentResolver.resolve(
            SurfaceContent.Comments(symbols, activeSymbolLocation(symbols, activeLocation, state), scroll),
            rect,
            SurfaceRenderMode.Pinned
          )
        case other =>
          SurfaceContentResolver.resolve(other, rect, SurfaceRenderMode.Pinned)
    TextPanelView(
      rect = rect,
      contentRect = Some(SurfaceFrameLayout.forContent(rect, surface.content).contentRect),
      title = resolved.title.getOrElse(""),
      surfaceId = Some(surface.id),
      composition = Some(compositionFor(surface, rect, resolved, state))
    )

  /** Every docked content kind's composition, bespoke where one already exists, otherwise the generic
    * [[RowsSurfaceComposition]] built from `resolved` -- so a pinned panel is always painted from one composed plan,
    * never a plain-rows fallback with its own, separately derived geometry (issue #1683).
    */
  private def compositionFor(
    surface: UiSurface,
    rect: LayoutRect,
    resolved: ResolvedSurfaceContent,
    state: Option[AppState]
  ): ResolvedSurfaceComposition =
    surface.content match
      case SurfaceContent.DirectoryTree(tree, selectedPath, scroll) =>
        DirectoryTreeSurfaceComposition.forTree(tree, selectedPath, scroll, rect)
      case SurfaceContent.Outline(symbols, activeLocation, scroll) =>
        OutlineSurfaceComposition.forOutline(
          symbols,
          activeSymbolLocation(symbols, activeLocation, state),
          rect,
          scroll
        )
      case SurfaceContent.Diagnostics(issues, activeLocation, scroll) =>
        DiagnosticsSurfaceComposition.forDiagnostics(issues, activeLocation, rect, scroll)
      case SurfaceContent.Comments(symbols, activeLocation, scroll) =>
        CommentsSurfaceComposition.forComments(
          symbols,
          activeSymbolLocation(symbols, activeLocation, state),
          rect,
          scroll
        )
      case content =>
        RowsSurfaceComposition.forResolved(resolved, rect, SurfaceFrameLayout.borderCellsFor(content))

  private def activeSymbolLocation(
    symbols: List[Symbol],
    fallback: Option[Location],
    state: Option[AppState]
  ): Option[Location] =
    fallback.orElse {
      state
        .flatMap(_.activeCursorPosition)
        .flatMap(cursor => DocumentNavigation.currentSymbol(symbols, cursor))
        .map(_.location)
    }

/** Identity of the pixels a [[RenderSurface]] accumulates into. A newtype over the backing object's own reference
  * identity (typically the image it draws into) rather than a bare `AnyRef`, so the render path's cache keys carry an
  * asserted domain meaning instead of an untyped object reference. Erases to the wrapped value at runtime, so equality,
  * hashing and weak-reference behaviour are exactly the backing object's own -- what a `WeakHashMap[AnyRef, _]` needs
  * to track it correctly.
  */
opaque type SurfaceContentIdentity = AnyRef

object SurfaceContentIdentity:
  def apply(value: AnyRef): SurfaceContentIdentity = value

trait RenderSurface:

  /** Identity of the pixels this surface accumulates into, when it preserves what earlier frames drew.
    *
    * `Some(key)` promises two things: the surface still holds the pixels of the last frame flushed through a surface
    * reporting the same key, and [[clearViewportExcept]] genuinely preserves the rectangles it is handed. Renderers may
    * only skip redrawing unchanged content when a key is present. `None` — the default — means every frame starts from
    * unknown pixels and everything must be drawn.
    */
  def persistentContentKey: Option[SurfaceContentIdentity] = None

  /** Whose cached modal/panel layers this surface may reuse. Successive frames of one window must agree on it even
    * though each frame is a fresh surface, or every frame starts a new cache entry (#1798).
    */
  def layerCacheOwner: ScreenIdentity = ScreenIdentity(this)

  /** Whether this surface's pixel coordinates are real device pixels, finer than a cell. The renderer keeps and
    * repaints sub-cell bands of chrome and rows only on a surface that does; on one whose "pixel" is a cell it repaints
    * each chrome whole.
    */
  def addressesPixels: Boolean = false

  def setForegroundColor(color: RenderColor): Unit
  def setBackgroundColor(color: RenderColor): Unit
  def getBackgroundColor: RenderColor

  def clearViewport(color: RenderColor): Unit =
    setBackgroundColor(color)
    fillRect(0, 0, viewportWidth, viewportHeight, ' ')

  /** Clear the viewport to `color` while leaving the given logical-pixel rectangles untouched.
    *
    * The default clears everything, which is why callers must check [[persistentContentKey]] first: a surface without a
    * persistent key preserves nothing, so its caller has to redraw the content it would otherwise have skipped.
    */
  def clearViewportExcept(color: RenderColor, @unused preserved: List[PixelRect]): Unit =
    clearViewport(color)

  def putString(x: Int, y: Int, s: String): Unit
  def fillRect(x: Int, y: Int, width: Int, height: Int, char: Char): Unit
  def enableStyle(style: TextStyle): Unit
  def disableStyle(style: TextStyle): Unit
  def devicePixelScaleX: Double = 1.0
  def devicePixelScaleY: Double = 1.0

  /** Character- and pixel-run text drawing. Required: every real surface must draw text. */
  def text: TextDrawing

  /** Pixel-addressed rect fills, image blits, and pixel-space translation. Required: every real surface must draw
    * carets and images.
    */
  def pixels: PixelDrawing

  /** Alpha compositing, when this surface supports it. `None` means callers must skip the effect rather than assume it
    * happened.
    */
  def effects: Option[Effects] = None

  /** Copying pixels the surface already holds, when it can. `None` means a row that moved has to be drawn again. */
  def pixelCopying: Option[PixelCopying] = None

  /** Lifting pixels off the surface and putting them back, when it can. `None` means nothing may be drawn over pixels
    * that persist.
    */
  def pixelLifting: Option[PixelLifting] = None

  /** Panel borders and rectangular clipping, when this surface supports them. `None` means callers must fall back to
    * drawing without that chrome rather than assume it happened.
    */

  def withFallbackFamily(@unused family: String, @unused ligaturesOn: Boolean = false)(draw: => Unit): Unit = draw

/** Pixel-addressed rect fills, image blits, and pixel-space coordinate translation. Like [[TextDrawing]], every real
  * surface implements this -- carets and Markdown preview images are drawn through it unconditionally -- so
  * [[RenderSurface.pixels]] exposes it directly rather than as an `Option`.
  */
trait PixelDrawing:
  def fillPixelRect(xPx: Int, yPx: Int, widthPx: Int, heightPx: Int, color: RenderColor): Unit

  /** Fill each rect of `rects`, four ints (x, y, width, height) apiece, with `color`: the picture [[fillPixelRect]]
    * makes for each in turn. A surface where every fill is a call into native code draws them as one.
    */
  def fillPixelRects(rects: IArray[Int], color: RenderColor): Unit =
    (0 until rects.length by 4).foreach(i => fillPixelRect(rects(i), rects(i + 1), rects(i + 2), rects(i + 3), color))

  def drawImage(image: RenderImage, x: Int, y: Int, width: Int, height: Int): Unit

  /** Composite a whole-surface layer image (a modal/panel layer buffer, produced by
    * [[LayerBufferSupport.newLayerSurface]]) back onto this surface, covering it exactly.
    *
    * Distinct from [[drawImage]] because a layer buffer is already at this surface's own backing resolution: it must be
    * blitted one-for-one, not scaled through the cell grid and device transform that [[drawImage]]'s cell-addressed
    * geometry applies. Routing it through `drawImage(image, 0, 0, viewportWidth, viewportHeight)` snaps the destination
    * to `floor(logicalSize / cellSize) * cellSize` and then re-scales by the device factor, so a layer comes back very
    * slightly smaller than it left -- imperceptible in one frame, but the command runner re-composites on every
    * navigation keystroke, so the shrink compounds into a visible "zoom out" of the editor pane behind it until a full
    * clean repaint resets it.
    *
    * The only real surface with a layer-buffer capability is [[Java2DRenderSurface]], which overrides this to blit 1:1
    * in device pixels; every other surface either advertises no layer buffers at all (a terminal) or is a headless test
    * double. The default records the composite as an ordinary full-surface `drawImage` so those doubles keep observing
    * one blit per layer paint, without needing this surface's backing resolution.
    */
  def compositeFullSurfaceLayer(image: RenderImage): Unit =
    drawImage(image, 0, 0, image.widthPx, image.heightPx)

  /** Composite a layer buffer made by [[LayerBufferSupport.newBoundedLayerSurface]] for `bounds` back where it was
    * painted. A surface whose bounded layers are whole-surface ones (the default) composites them as such.
    */
  def compositeLayer(image: RenderImage, @unused bounds: PixelRect): Unit =
    compositeFullSurfaceLayer(image)

  /** Translate drawing in device-independent logical pixels for fractional-cell floating geometry. */
  def withPixelTranslation(xPx: Double, yPx: Double)(render: => Unit): Unit

/** Moving pixels the surface already holds, rather than drawing them again. Genuinely optional: a surface that cannot
  * (a cell-addressed terminal, or a raster surface whose device scale is not a whole number, where logical pixels do

trait PixelCopying:

    * destination may overlap: the copy reads every source pixel as it was before it.

  def copyPixels(from: PixelRect, dyPx: Int): Unit

/** What a rect of a surface's pixels held, to be put back exactly as they were. */
trait LiftedPixels

/** Drawing something over pixels that persist from frame to frame, and taking it off again: the pixels beneath are
  * lifted first and put back before the next frame builds on them. Genuinely optional, so
  * [[RenderSurface.pixelLifting]] exposes it as an `Option`.
  */
trait PixelLifting:

  /** The pixels of `area`, in logical pixels, with a device pixel's margin so that a fill of the same area is covered
    * whatever the scale rounds it to. `spare` is pixels lifted before and since given back, whose storage is reused
    * when it fits.
    */
  def lift(area: PixelRect, spare: Option[LiftedPixels]): LiftedPixels

  /** Overwrites the pixels `lifted` came from with what they held. */
  def putBack(lifted: LiftedPixels): Unit

/** Alpha compositing. Genuinely optional: a surface that can't do it (or a headless test double) simply skips the fade
  * rather than degrading a required drawing operation, so [[RenderSurface.effects]] exposes it as an `Option` and
  * callers decide whether skipping the effect is safe.
  */
trait Effects:
  def setAlpha(alpha: Float): Unit

/** A fresh, independently-paintable surface shaped exactly like the surface this capability came from -- same cell
  * metrics, font, logical size and device scale -- for a layer (a pinned panel, a modal, a floating overlay) to own its
  * own persisted buffer instead of painting straight into the shared frame surface (#1100 stage 2). Genuinely optional:
  * a surface with no natural notion of an offscreen sub-buffer (a cell-addressed terminal, which has no sub-cell pixel
  * buffering the same way a raster surface does) simply has no capability to expose, so [[RenderSurface.layerBuffers]]
  * exposes this as an `Option` and callers fall back to painting the layer directly into the shared surface, same as
  * before this capability existed.
  */
trait LayerBufferSupport:

  /** A new surface painting into a blank, fully transparent buffer the same shape as the surface this capability came
    * from. `onFlush` receives the finished image once the caller's `flush()` completes -- compositing it onto the frame
    * surface (e.g. via `RenderSurface.pixels.compositeFullSurfaceLayer`) is the caller's job, not this surface's; a
    * layer surface never publishes itself anywhere on its own.
    *
    * Everything the layer's owner didn't paint stays transparent. That is what makes compositing the whole layer back
    * over a frame whose content has since changed correct (#1798) -- a layer seeded with a full copy of the frame would
    * paste that stale copy back over everything.
    *
    * `recycled` is a previous layer image the caller is done with; it is cleared and painted into instead of
    * allocating, when its size still matches.
    */
  def newLayerSurface(onFlush: RenderImage => Unit, recycled: Option[RenderImage] = None): RenderSurface

  /** As [[newLayerSurface]], for a layer that paints only inside `bounds` (logical pixels): the surface still addresses
    * the whole window, so a renderer paints at its usual coordinates, but the image it hands back covers `bounds` alone
    * and goes back through [[PixelDrawing.compositeLayer]]. A capability with no smaller buffer to offer returns the
    * whole-surface layer, which that composite handles too.
    */
  def newBoundedLayerSurface(
    @unused bounds: PixelRect,
    onFlush: RenderImage => Unit,
    recycled: Option[RenderImage] = None
  ): RenderSurface =
    newLayerSurface(onFlush, recycled)

/** A caret shape a real terminal's own cursor can be styled as via DECSCUSR (`CSI Ps SP q`). */
enum HardwareCursorShape:
  case Block, Underline, Bar

/** A DECSCUSR-expressible caret style: shape plus whether the terminal should blink it itself.
  *
  * There is no cursor-shape setting in [[com.serenity.config.CursorConfig]] today (only
  * [[com.serenity.config.CursorMode]]'s blink or steady mode) -- callers that delegate the caret to the terminal
  * (#1170) always ask for a block, the shape every terminal defaults to. `decscusrParam` is kept as a total function of
  * shape/blink regardless, so a future per-buffer shape setting has somewhere to plug in without touching the
  * escape-emission code.
  */
final case class HardwareCursorStyle(shape: HardwareCursorShape, blinking: Boolean):

  /** The DECSCUSR parameter for this shape/blink pair, per xterm's ctlseqs: `CSI Ps SP q` where `Ps` is 1/2 =
    * blinking/steady block, 3/4 = blinking/steady underline, 5/6 = blinking/steady bar.
    */
  def decscusrParam: Int =
    shape match
      case HardwareCursorShape.Block     => if blinking then 1 else 2
      case HardwareCursorShape.Underline => if blinking then 3 else 4
      case HardwareCursorShape.Bar       => if blinking then 5 else 6

object HardwareCursorStyle:

  def forMode(mode: CursorMode): HardwareCursorStyle =
    HardwareCursorStyle(HardwareCursorShape.Block, blinking = mode == CursorMode.Blink)

/** Paints a chapter note's overview, faded, on the blank rows under an empty chapter. A separate pass drawn after each
  * pane's own content, so it replaces nothing: the rows are blank in the document and the ghost exists only in the
  * frame. It reuses [[RendererPaneContent]]'s row arithmetic so a ghost sits exactly where the row's own text would.
  *
  * Single-column plain panes only: the inline Markdown lens and the multi-column page layout paint their own rows, and
  * `BufferRenderAnnotations.ghostsByLine` is empty for the former.
  */
object RendererChapterGhosts:

  /** Paints the ghosts on the rows `dirtyRowsByPane` names for each pane; a pane it does not name paints every row. */
  def render(
    state: AppState,
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    dirtyRowsByPane: Map[PaneId, Set[Int]]
  ): Unit =
    state.persisted.layout.editorPanes.foreach { (paneId, pane) =>
      for
        paneLayout <- renderPlan.paneLayouts.get(paneId)
        snapshot   <- renderPlan.snapshots.get(paneId)
        bufferId   <- pane.bufferId
        buffer     <- state.persisted.buffers.get(bufferId)
        ghosts     <- renderPlan.annotations.get(bufferId).map(_.ghostsByLine).filter(_.nonEmpty)
        if renderPlan.columnSnapshotsFor(paneId).isEmpty
      do renderPane(buffer, paneLayout.contentRect, snapshot, ghosts, dirtyRowsByPane.get(paneId), state, context)
    }

  /** The ghost each of `snapshot`'s rows paints, if any: a ghost sits on the first row of a blank line. */
  def rowGhosts(snapshot: TextLayoutSnapshot, ghosts: Map[Int, String]): Vector[Option[String]] =
    if ghosts.isEmpty then Vector.empty
    else
      snapshot.visualLines.map(visualLine =>
        ghosts.get(visualLine.bufferLine).filter(_ => visualLine.startColumn == 0 && visualLine.text.isBlank)
      )

  /** The rows whose ghost differs between two frames' [[rowGhosts]], a row past either end having none. */
  def changedRows(previous: Vector[Option[String]], current: Vector[Option[String]]): Set[Int] =
    (0 until math.max(previous.size, current.size)).filter { row =>
      previous.lift(row).flatten != current.lift(row).flatten
    }.toSet

  /** Whether `bufferId` has a chapter note a ghost is derived from. A ghost depends on a note's text and on which
    * chapters are empty, which no buffer damage reports.
    */
  def paneShowsGhosts(state: AppState, bufferId: BufferId): Boolean =
    state.runtime.chapterGhostsVisible &&
      state.persisted.buffers
        .get(bufferId)
        .exists(buffer => !buffer.hidden && buffer.annotations.notes.keys.exists(hasChapterKey))

  private def hasChapterKey(key: NoteKey): Boolean =
    key match
      case NoteKey.Chapter(_) => true
      case NoteKey.Keyword(_) => false

  private def renderPane(
    buffer: Buffer,
    rect: LayoutRect,
    snapshot: TextLayoutSnapshot,
    ghosts: Map[Int, String],
    dirtyRows: Option[Set[Int]],
    state: AppState,
    context: RenderContext
  ): Unit =
    context.surface.text.setFont(FontSpec.fromAwt(context.fontForBuffer(buffer)))
    val ghostOfRow = rowGhosts(snapshot, ghosts)
    snapshot.visualLines.zipWithIndex.foreach { (visualLine, index) =>
      val ghost   = ghostOfRow.lift(index).flatten.filter(_ => dirtyRows.forall(_.contains(index)))
      val screenX = rect.x + RendererPaneContent.visualLineCellOffset(visualLine, context)
      val visible =
        RendererPaneContent.visualLineFits(rect, index, context, snapshot) &&
          RendererPaneContent.visualLineVisible(rect, index, context, snapshot) &&
          screenX >= 0 && screenX < rect.right
      if visible then
        ghost.foreach(paintLine(_, rect, screenX, index, visualLine, snapshot, state.persisted.theme, context))
    }

  private def paintLine(
    ghost: String,
    rect: LayoutRect,
    screenX: Int,
    index: Int,
    visualLine: TextVisualLine,
    snapshot: TextLayoutSnapshot,
    theme: com.serenity.ui.theme.Theme,
    context: RenderContext
  ): Unit =
    val text = ghost.take(math.max(0, rect.right - screenX))
    if text.nonEmpty then
      val surface = context.surface
      surface.setForegroundColor(theme.placeholder)
      surface.setBackgroundColor(theme.background)
      if RendererPaneSetup.usesMeasuredDrawing(snapshot, context) then
        val xOriginPx       = context.cellMetrics.toPixelX(rect.x).toFloat
        val contentRightXPx = context.cellMetrics.toPixelX(rect.right).toFloat
        surface.text.drawRunPx(
          xOriginPx,
          RendererPaneContent.visualLineTopPx(rect, index, context, snapshot),
          contentRightXPx - xOriginPx,
          RendererPaneContent.rowHeightPxFor(visualLine, snapshot),
          RendererPaneContent.rowAscentPxFor(visualLine, snapshot),
          text
        )
      else
        CharacterRenderer.renderString(
          surface,
          screenX,
          RendererPaneContent.textRowMetrics(rect, context, snapshot).screenRow(index),
          text
        )
      surface.setForegroundColor(theme.foreground)
package com.serenity.ui.renderer

/** Which parts of one chrome element a frame repaints: every part, or only those named. */
final case class Repaint[A] private (everything: Boolean, parts: Set[A]):
  def includes(part: A): Boolean = everything || parts.contains(part)

  def nonEmpty: Boolean = everything || parts.nonEmpty

object Repaint:
  def all[A]: Repaint[A]                 = Repaint(everything = true, Set.empty[A])
  def only[A](parts: Set[A]): Repaint[A] = Repaint(everything = false, parts)

/** The chrome a frame paints -- spacers, pane headers, line numbers and the pinned status row -- as opposed to the
  * document rows. A persisting surface still holds what the chrome it does not repaint looked like.
  *
  * `columnRails` names, for a multi-column pane, the pane rows ([[ColumnPaneRows]]) whose line-number rail bands the
  * frame repaints; a pane it does not name repaints its rails whole.
  */
final case class ChromeRepaint(
    spacers: Boolean,
    statusRow: Boolean,
    headers: Repaint[PaneId],
    lineNumberRows: Repaint[Int],
    columnRails: Map[PaneId, Repaint[Int]] = Map.empty
):
  def paintsAny: Boolean =
    spacers || statusRow || headers.nonEmpty || lineNumberRows.nonEmpty || columnRails.values.exists(_.nonEmpty)

  def railRows(paneId: PaneId): Repaint[Int] = columnRails.getOrElse(paneId, Repaint.all[Int])

object ChromeRepaint:
  val Whole: ChromeRepaint = ChromeRepaint(spacers = true, statusRow = true, Repaint.all, Repaint.all)

/** The rects chrome is painted into. Chrome kept from an earlier frame is only valid while they hold still.
  *
  * A moved pane rect also repaints every pane row: the tab strip appearing or going away shifts them with no damage a
  * `DamageProducer` transition could report, as it knows the state but not the layout.
  */
final case class ChromeGeometry(
    spacers: List[LayoutRect],
    statusRow: Option[LayoutRect],
    lineNumbers: List[LayoutRect],
    panes: Map[PaneId, EditorPaneLayout]
)

object ChromeGeometry:

  def of(renderPlan: EditorPaneRenderPlan): ChromeGeometry =
    val contract = renderPlan.layoutContract
    ChromeGeometry(
      spacers = List(contract.leftSpacerRect, contract.rightSpacerRect),
      statusRow = contract.gutterRect,
      lineNumbers = contract.lineNumberRect.toList ++ contract.rightLineNumberRect.toList,
      panes = renderPlan.paneLayouts
    )

/** Decides which chrome a frame must repaint, and which pixels of it the frame keeps.
  *
  * Chrome is a function of state that [[Damage]] already reports: the status row of [[Damage.Chrome]], a header of
  * [[Damage.PaneChrome]], a line number of its own row's damage, and everything else of a layout, theme or config
  * change, which is [[Damage.Everything]]. So a frame needs to repaint only what its damage names, and the rest stays
  * where the persisting surface already holds it.
  */
private[renderer] object RendererChromePlan:

  /** What the frame's panes and this surface's last frame say beyond the damage itself. */
  final case class Signals(
      geometryChanged: Boolean,
      rowsOverflow: Boolean,
      surfaceOverChrome: Boolean,
      lastRowMoved: Set[Int]
  )

  /** The chrome this frame repaints: [[repaintFor]] over [[Signals]] read off the frame's panes and surfaces. */
  def plan(
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    panes: Map[PaneId, PaneFrameRecord],
    persistenceKey: SurfaceContentIdentity,
    damage: Damage,
    dirtyRowsByPane: Map[PaneId, Set[Int]],
    geometryChanged: Boolean
  ): ChromeRepaint =
    repaintFor(
      renderPlan.layoutContract,
      damage,
      dirtyRowsByPane,
      Signals(
        geometryChanged || !context.surface.addressesPixels,
        rowsOverflow(context, renderPlan, panes, persistenceKey),
        surfacesMayTouch(context, renderPlan, persistenceKey, damage, chromeRegions(context, renderPlan)),
        lastRowIfRowCountChanged(context, renderPlan.layoutContract.activePaneId, panes, persistenceKey)
      )
    )

  /** Everything is repainted when the chrome's rects moved or `damage` is `Everything`, and also when it names a
    * surface that may have left pixels over chrome (one that is, or was, over it, or whose place is not known): a panel
    * that opened, moved or closed there is a change nothing here tracks.
    *
    * `rowsOverflow` says glyphs reach past a pane's content rect, into the spacers and a right-hand counter. Chrome a
    * row's ink may have reached is repainted, as it is cleared of that ink only by being repainted.
    */
  def repaintFor(
    contract: EditorLayoutContract,
    damage: Damage,
    dirtyRowsByPane: Map[PaneId, Set[Int]],
    signals: Signals
  ): ChromeRepaint =
    if signals.geometryChanged || Damage.isEverything(damage) || signals.surfaceOverChrome then ChromeRepaint.Whole
    else
      ChromeRepaint(
        spacers = signals.rowsOverflow,
        statusRow = Damage.touchesChrome(damage),
        headers = Repaint.only(Damage.paneChromeIds(damage)),
        lineNumberRows =
          if signals.rowsOverflow && contract.rightLineNumberRect.isDefined then Repaint.all[Int]
          else lineNumberRowsToRepaint(contract.activePaneId, dirtyRowsByPane, signals.lastRowMoved),
        columnRails = dirtyRowsByPane.map((paneId, rows) => paneId -> Repaint.only(rows))
      )

  /** Whether a surface `damage` names may have left pixels in `regions`: it is, or was last frame, over one, or its
    * place is not known.
    */
  def surfacesMayTouch(
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    persistenceKey: SurfaceContentIdentity,
    damage: Damage,
    regions: List[PixelRect]
  ): Boolean =
    val contract = renderPlan.layoutContract
    val previous = context.caches.frameState.previousFloatingSurfaceRectsFor(persistenceKey)
    Damage.surfaceIds(damage).exists { surfaceId =>
      val current =
        (if surfaceId == UiSurface.TabBarSurfaceId then contract.tabBarRect.toList
         else contract.overlayRect(surfaceId).toList ++ contract.panelRect(surfaceId).toList)
          .map(pixelRectOf(context.cellMetrics))
      val places = current ++ previous.get(surfaceId).toList
      places.isEmpty || places.exists(place => regions.exists(_.intersects(place)))
    }

  /** How far, in logical pixels, a row's glyphs can reach past the left and right of its band: antialiasing of a glyph
    * with a negative side bearing. A row is kept with that margin, or the clear would take it.
    */
  val BleedPx: Int = 1

  def withBleed(context: RenderContext)(band: PixelRect): PixelRect =
    if context.surface.addressesPixels then band.copy(xPx = band.xPx - BleedPx, widthPx = band.widthPx + 2 * BleedPx)
    else band

  /** Every place chrome is painted. */
  private[renderer] def regions(context: RenderContext, renderPlan: EditorPaneRenderPlan): List[PixelRect] =
    chromeRegions(context, renderPlan)

  private def chromeRegions(context: RenderContext, renderPlan: EditorPaneRenderPlan): List[PixelRect] =
    val contract = renderPlan.layoutContract
    val panes = renderPlan.paneLayouts.values.toList.flatMap(layout =>
      List(layout.headerRect, layout.topSpacerRect, layout.bottomSpacerRect)
    )
    val columnRails = renderPlan.paneLayouts.toList.flatMap { (paneId, layout) =>
      renderPlan
        .columnSnapshotsFor(paneId)
        .filter(_.gutterWidthCells > 0)
        .map(ColumnRails.railRect(layout.contentRect, _))
    }
    (List(contract.leftSpacerRect, contract.rightSpacerRect) ++ contract.gutterRect.toList ++
      contract.lineNumberRect.toList ++ contract.rightLineNumberRect.toList ++ columnRails ++ panes)
      .map(pixelRectOf(context.cellMetrics))

  /** The active pane's last row when the pane showed another number of rows in the frame this surface last drew: the
    * counter below the last row belongs to it, and what lay there then is no longer what lies there now.
    */
  def lastRowIfRowCountChanged(
    context: RenderContext,
    activePaneId: Option[PaneId],
    panes: Map[PaneId, PaneFrameRecord],
    persistenceKey: SurfaceContentIdentity
  ): Set[Int] =
    for
      paneId   <- activePaneId.toSet
      record   <- panes.get(paneId).toSet
      previous <- context.caches.frameState.previousSnapshotsFor(persistenceKey).get(paneId).toSet
      if previous.visualLines.length != record.rowRects.length
      lastRow <- record.rowRects.indices.lastOption.toSet
    yield lastRow

  /** Whether any pane has a row whose glyphs reach past its content rect, now or in the frame this surface last drew.
    */
  def rowsOverflow(
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    panes: Map[PaneId, PaneFrameRecord],
    persistenceKey: SurfaceContentIdentity
  ): Boolean =
    panes.values.exists(_.overflowingRows.nonEmpty) ||
      context.caches.frameState.previousSnapshotsFor(persistenceKey).exists { (paneId, snapshot) =>
        renderPlan.paneLayouts
          .get(paneId)
          .exists(layout => RendererPaneSetup.overflowingRows(layout.contentRect, context, snapshot).nonEmpty)
      }

  /** A line number sits beside its row and repaints with it. An active pane the frame redraws without row records (a
    * markdown lens, an empty buffer) has no rows to pair them with, so its numbers repaint whole.
    */
  private def lineNumberRowsToRepaint(
    activePaneId: Option[PaneId],
    dirtyRowsByPane: Map[PaneId, Set[Int]],
    lastRowMoved: Set[Int]
  ): Repaint[Int] =
    activePaneId.flatMap(dirtyRowsByPane.get) match
      case Some(dirtyRows) => Repaint.only(dirtyRows ++ lastRowMoved)
      case None            => Repaint.all[Int]

  /** Paints the chrome `repaint` names, in the order a whole frame paints it. */
  def paint(state: AppState, context: RenderContext, renderPlan: EditorPaneRenderPlan, repaint: ChromeRepaint): Unit =
    if repaint.spacers then RendererPaneSetup.renderSpacerColumns(state, context, renderPlan.layoutContract)
    RendererGutter.renderLineNumbers(state, context, renderPlan, repaint.lineNumberRows, repaint.railRows)
    if repaint.statusRow then RendererGutter.renderGutter(state, context, renderPlan.layoutContract)

  /** The pixel bands `repaint` leaves alone, to be handed to `clearViewportExcept` so they survive the clear. */
  def preservedRects(
    state: AppState,
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    panes: Map[PaneId, PaneFrameRecord],
    repaint: ChromeRepaint
  ): List[PixelRect] =
    val contract = renderPlan.layoutContract
    val editorPaneLayouts =
      state.persisted.layout.orderedPaneIds.filter(state.persisted.layout.editorPanes.contains).flatMap { paneId =>
        renderPlan.paneLayouts.get(paneId).map(paneId -> _)
      }
    val toPixels = pixelRectOf(context.cellMetrics)
    val spacers =
      if repaint.spacers then Nil
      else
        List(
          trimmedTowardsContent(context, toPixels(contract.leftSpacerRect), contentOnRight = true),
          trimmedTowardsContent(context, toPixels(contract.rightSpacerRect), contentOnRight = false)
        ) ++ editorPaneLayouts.flatMap((_, layout) => List(layout.topSpacerRect, layout.bottomSpacerRect)).map(toPixels)
    val statusRow = contract.gutterRect.filter(_ => !repaint.statusRow).toList
    val headers = editorPaneLayouts.collect {
      case (paneId, layout) if !repaint.headers.includes(paneId) =>
        contract.paneHeaderRect(paneId).getOrElse(layout.headerRect)
    }
    spacers ++ (statusRow ++ headers).map(toPixels) ++
      keptLineNumberBands(state, context, renderPlan, panes, repaint.lineNumberRows) ++
      keptColumnRailBands(state, context, renderPlan, repaint)

  /** A side spacer without the margin ([[BleedPx]]) beside the text: a glyph's antialiasing can land a pixel there, and
    * only the row's own redraw clears it. The margin beside a kept row is kept with the row's band; beside a row drawn
    * again, or one that is gone, it is cleared, where a kept spacer would hold the last frame's fringe.
    */
  private def trimmedTowardsContent(context: RenderContext, spacer: PixelRect, contentOnRight: Boolean): PixelRect =
    if !context.surface.addressesPixels then spacer
    else if contentOnRight then spacer.copy(widthPx = (spacer.widthPx - BleedPx).max(0))
    else spacer.copy(xPx = spacer.xPx + BleedPx, widthPx = (spacer.widthPx - BleedPx).max(0))

  /** The rail bands of every multi-column pane's rows that `repaint` leaves alone. */
  private def keptColumnRailBands(
    state: AppState,
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    repaint: ChromeRepaint
  ): List[PixelRect] =
    if !RendererGutter.paintsColumnRails(state) then Nil
    else
      for
        paneId <- state.persisted.layout.editorPanes.keys.toList
        layout <- renderPlan.paneLayouts.get(paneId).toList
        columns = renderPlan.columnSnapshotsFor(paneId)
        (placement, rows) <- columns.zip(ColumnPaneRows.localRepaints(columns, repaint.railRows(paneId)))
        if placement.gutterWidthCells > 0
        band <- ColumnRails.keptBands(
          context,
          ColumnRails.railRect(layout.contentRect, placement),
          placement.snapshot,
          rows
        )
      yield band

  /** The line-number bands of the active pane's rows that `rows` leaves alone, in both counters. */
  private def keptLineNumberBands(
    state: AppState,
    context: RenderContext,
    renderPlan: EditorPaneRenderPlan,
    panes: Map[PaneId, PaneFrameRecord],
    rows: Repaint[Int]
  ): List[PixelRect] =
    val contract = renderPlan.layoutContract
    for
      paneId <- state.persisted.layout.activeEditorPaneId.toList
      record <- panes.get(paneId).toList
      if state.chrome.lineNumbers
      rect <- contract.lineNumberRect.toList ++ contract.rightLineNumberRect.toList
      band <- RendererGutter.keptRowBands(state, context, record.snapshot, rect, rows)
    yield band

  def pixelRectOf(cellMetrics: CellMetrics)(rect: LayoutRect): PixelRect =
    val leftPx = cellMetrics.toPixelX(rect.x)
    val topPx  = cellMetrics.toPixelY(rect.y)
    PixelRect(leftPx, topPx, cellMetrics.toPixelX(rect.right) - leftPx, cellMetrics.toPixelY(rect.bottom) - topPx)
package com.serenity.ui.renderer

import com.serenity.config.AppConfig
import com.serenity.state.models.*
import com.serenity.ui.color.RenderColor
import com.serenity.ui.layout.*
import com.serenity.ui.theme.Theme

/** The package's caret authority: it paints the caret glyph(s) for a buffer's cursors, and owns the pane-clipping
  * geometry every caret and measured run in the package is placed by. [[measuredRunWidthWithin]] is public because
  * [[RendererHighlights]] clips selection and diagnostic backgrounds to the same right edge a caret is clipped to, and
  * a highlight that clipped differently would bleed past the pane the caret stops at.
  */
object RendererCursorGlyphs:

  /** Paint every visible cursor in `buffer` and report the pixel rect each one occupies, so a caller can bound a
    * repaint to cover them.
    */
  def renderCursors(
    buffer: Buffer,
    rect: LayoutRect,
    theme: Theme,
    config: AppConfig,
    context: RenderContext,
    snapshot: TextLayoutSnapshot,
    paintsCursor: CursorPosition => Boolean = _ => true
  ): List[PixelRect] =

    buffer.editing.cursors.toList.zipWithIndex.filter((cursor, _) => paintsCursor(cursor.position)).flatMap {
      (cursor, cursorIndex) =>
        val isPrimaryCursor = cursorIndex == 0
        val shouldRenderCursor =
          context.cursorVisible || (buffer.editing.cursors.size > 1 && !isPrimaryCursor)
        val rowMetrics = RendererPaneContent.textRowMetrics(rect, context, snapshot)
        caretPlacement(cursor.position, rect, context.cellMetrics, snapshot, rowMetrics) match
          case Some((visualLine, caret)) if shouldRenderCursor =>
            if RendererPaneContent.visualLineVisible(rect, visualLine, context, snapshot)
            then
              context.surface.pixels.fillPixelRect(
                caret.xPx,
                caret.yPx,
                caret.widthPx,
                caret.heightPx,
                cursorColorFor(config, theme, context, isPrimaryCursor)
              )
              List(caret)
            else Nil
          case _ => Nil
    }

  /** Where a caret at `position` is drawn within `rect`, and on which visual row: the pixel rect it fills, clipped to
    * the pane. Shared by painting and by anything that must hang off the painted caret (a floating surface's anchor),
    * so the two cannot disagree about the caret's row or column.
    */
  def caretPlacement(
    position: CursorPosition,
    rect: LayoutRect,
    cellMetrics: CellMetrics,
    snapshot: TextLayoutSnapshot,
    rowMetrics: TextRowMetrics
  ): Option[(Int, PixelRect)] =
    calculateCursorVisualPosition(position, snapshot).flatMap {
      case (visualLine, xPx) =>
        val caretWidthPx = math.max(2, math.round(cellMetrics.charWidth * 0.12f))
        val screenXPx    = cellMetrics.toPixelX(rect.x) + math.round(xPx)
        val screenYPx    = rowMetrics.cursorTopPx(visualLine)
        // Caret is as tall as the row it sits on, so on a heading line it grows with the heading (#1542 prose scale).
        val caretHeightPx = rowMetrics.rowHeightPx(visualLine)
        caretWithin(rect, cellMetrics, screenXPx, caretWidthPx).map((caretXPx, widthPx) =>
          visualLine -> PixelRect(caretXPx, screenYPx, widthPx, caretHeightPx)
        )
    }

  /** The (visual-row index, caret x) the caret is drawn at. Row selection is delegated to
    * [[com.serenity.state.models.NavigationGeometry.visualRowIndexFor]] -- the same lookup vertical navigation uses --
    * so the caret is never painted on a different wrapped row than Up/Down moves from at a wrap boundary (the "stuck
    * until nudged left/right" bug). Previously this used `collectFirst` (the *earlier* row at a boundary) while
    * navigation used `.lastOption` (the later row), so the two disagreed.
    */
  private def calculateCursorVisualPosition(
    cursor: CursorPosition,
    snapshot: TextLayoutSnapshot
  ): Option[(Int, Float)] =
    snapshot.navigationGeometry.visualRowIndexFor(cursor).map { visualIndex =>
      val line = snapshot.visualLines(visualIndex)
      val xPx  = line.xForColumn(cursor.column).getOrElse(line.widthPx)
      (visualIndex, xPx)
    }

  /** The column a caret belongs to. A wrapped line that continues into the next column leaves a cursor on the boundary
    * column matching both: the end of the earlier column's last row and the start of the later column's first. The
    * cursor's [[RowAffinity]] settles it exactly as it does between two rows of one snapshot -- downstream (the
    * default) is the later column, upstream the earlier -- so only one column paints it, and a terminal's single
    * hardware cursor is never placed from the wrong one.
    */
  def owningColumn(
    columnPlacements: Vector[ColumnSnapshotPlacement],
    position: CursorPosition
  ): Option[ColumnSnapshotPlacement] =
    val holding = columnPlacements.filter(_.snapshot.navigationGeometry.visualRowIndexFor(position).isDefined)
    position.rowAffinity match
      case RowAffinity.Upstream   => holding.headOption
      case RowAffinity.Downstream => holding.lastOption

  def measuredRunWidthWithin(
    rect: LayoutRect,
    context: RenderContext,
    startXPx: Float,
    endXPx: Float
  ): Option[Float] =
    val rightXPx = context.cellMetrics.toPixelX(rect.right).toFloat
    Option.when(startXPx < rightXPx)(math.max(0.0f, math.min(endXPx, rightXPx) - startXPx)).filter(_ > 0.0f)

  def caretWithin(
    rect: LayoutRect,
    cellMetrics: CellMetrics,
    desiredXPx: Int,
    desiredWidthPx: Int
  ): Option[(Int, Int)] =
    val leftXPx  = cellMetrics.toPixelX(rect.x)
    val rightXPx = cellMetrics.toPixelX(rect.right)
    val widthPx  = math.min(math.max(1, desiredWidthPx), rightXPx - leftXPx)
    Option.when(widthPx > 0)(desiredXPx.max(leftXPx).min(rightXPx - widthPx) -> widthPx)

  def cursorColorFor(
    config: AppConfig,
    theme: Theme,
    context: RenderContext,
    isPrimaryCursor: Boolean
  ): RenderColor =
    val activeColor =
      context.cursorColorOverride.getOrElse(config.cursorColors.active.fold(theme.cursor)(RenderColor.fromAwt))
    if isPrimaryCursor then activeColor
    else config.cursorColors.inactive.fold(activeColor)(RenderColor.fromAwt)
package com.serenity.ui.renderer

import java.util.concurrent.atomic.AtomicReference

import com.serenity.config.CursorMode

/** Cursor-only redraws (blink ticks that don't need a full layout) and cursor-overlay frames (a base frame painted
  * without the caret, then the caret drawn on top so blinking never has to redraw the base). Both flavours reuse the
  * cached [[PreparedScene]] from the last full layout via [[resolveCursorRenderPlan]] when it is still valid.
  */
object RendererCursorOverlay:

  /** The render plan a cursor-only redraw needs: the cached one from the last full layout, if it is still valid for
    * these fonts/metrics/viewport, or a freshly prepared one otherwise. Shared by every cursor-only entry point (Swing
    * and surface-generic alike) so the cache is consulted exactly the same way regardless of shell.
    */
  private def resolveCursorRenderPlan(
    state0: AppState,
    authoritativeScene: UiSceneSnapshot,
    surface: RenderSurface,
    viewportSize: ViewportSize,
    cursorVisible: Boolean,
    cursorColor: Option[RenderColor],
    codeFont: java.awt.Font,
    textFont: java.awt.Font,
    uiFont: java.awt.Font,
    cellMetrics: CellMetrics,
    uiMetrics: CellMetrics,
    caches: RenderCaches
  ): (CalculatedLayout, EditorPaneRenderPlan) =
    caches.frameState
      .preparedSceneFor(surface)
      .filter(_.matches(authoritativeScene, codeFont, textFont, uiFont, cellMetrics, uiMetrics, viewportSize))
      .map(value => value.scene.calculatedLayout -> value.renderPlan)
      .getOrElse {
        val next = RendererFramePlanner.prepareScene(
          state0,
          surface,
          viewportSize,
          authoritativeScene,
          cursorVisible,
          cursorColor,
          codeFont,
          textFont,
          uiFont,
          cellMetrics,
          uiMetrics,
          caches
        )
        caches.frameState.rememberPreparedScene(surface, next)
        next.scene.calculatedLayout -> next.renderPlan
      }

  /** Draw just the cursor glyphs for `renderPlan` into `surface` and flush. Shared tail of every cursor-only /
    * cursor-overlay entry point: the only thing that differs between them is which surface the cursor lands on and how
    * that surface's owning shell was told a fresh layout was needed.
    */
  private def paintCursorsOnly(
    state0: AppState,
    surface: RenderSurface,
    layout: CalculatedLayout,
    renderPlan: EditorPaneRenderPlan,
    cursorVisible: Boolean,
    cursorColor: Option[RenderColor],
    codeFont: java.awt.Font,
    textFont: java.awt.Font,
    uiFont: java.awt.Font,
    cellMetrics: CellMetrics,
    uiMetrics: CellMetrics,
    caches: RenderCaches
  ): List[PixelRect] =
    val context = RenderContext(
      surface,
      layout,
      cursorVisible,
      cursorColor,
      codeFont,
      textFont,
      uiFont,
      cellMetrics,
      uiMetrics,
      caches
    )
    val cursorRects = RendererPaneContent.renderEditorCursors(state0, context, renderPlan)
    presentHardwareCursor(surface, cursorVisible, cursorRects, cellMetrics, state0.persisted.config.cursorMode)
    surface.flush()
    cursorRects

  /** #1170: on a surface that can delegate the caret to a real hardware/terminal cursor, position and style it from the
    * primary caret's rect instead of leaving it to be painted as surface content. A surface with no hardware cursor to
    * delegate to is entirely unaffected: this is a no-op there.
    */
  private def presentHardwareCursor(
    surface: RenderSurface,
    cursorVisible: Boolean,
    cursorRects: List[PixelRect],
    cellMetrics: CellMetrics,
    cursorMode: CursorMode
  ): Unit =
    surface.hardwareCursor.foreach { hardwareCursor =>
      cursorRects.headOption match
        case Some(rect) if cursorVisible =>
          hardwareCursor.present(
            cellMetrics.toCol(rect.xPx),
            cellMetrics.toRow(rect.yPx),
            HardwareCursorStyle.forMode(cursorMode)
          )
        case _ =>
          hardwareCursor.hide()
    }

  /** Surface-generic form of [[renderCursorOnly]]: redraws just the cursor glyphs into `surface`, reusing the cached
    * render plan from the last full layout when it is still valid. `false` only for a start-page-only state, which has
    * no cursor to draw.
    */
  def renderCursorOnly(
    state: AppState,
    cursorVisible: Boolean,
    surface: RenderSurface,
    viewportSize: ViewportSize,
    codeFont: FontSpec,
    textFont: FontSpec,
    uiFont: FontSpec,
    cellMetrics: CellMetrics,
    uiMetrics: CellMetrics,
    cursorColor: Option[RenderColor],
    caches: RenderCaches
  ): Boolean =
    // #1105/#1215: a surface reporting no FontRenderContext (a terminal) has no real font rendering to measure
    // against at all -- its own declared cellMetrics must reach this scene's cell-based layout, and must force it,
    // rather than being silently discarded by the font's own measured-vs-cell auto-detection. A surface that does
    // report one (every GUI canvas, and any other surface-generic caller with real font rendering) keeps this scene's
    // long-standing auto-detected behaviour untouched.
    val cellMetricsOverride = Option.when(surface.text.fontRenderContext.isEmpty)(cellMetrics)
    RendererEntryPoints.withSceneIfNeeded(
      state,
      caches.authoritativeScene
        .forState(state, viewportSize, codeFont.toAwt, textFont.toAwt, cellMetrics = cellMetricsOverride)
    )(_ => false) { authoritativeScene =>
      val (layout, renderPlan) = resolveCursorRenderPlan(
        state,
        authoritativeScene,
        surface,
        viewportSize,
        cursorVisible,
        cursorColor,
        codeFont.toAwt,
        textFont.toAwt,
        uiFont.toAwt,
        cellMetrics,
        uiMetrics,
        caches
      )
      val _ = paintCursorsOnly(
        state,
        surface,
        layout,
        renderPlan,
        cursorVisible,
        cursorColor,
        codeFont.toAwt,
